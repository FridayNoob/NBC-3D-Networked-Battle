## 二、功能概述

美术/策划在 Battle 场景里摆几个 `BattleObstacleMarker`（圆或方），点菜单写回 `BattleObstacle.csv`；服务端读表后，英雄与怪物的移动都会被这些形状挡住，斜着撞墙会贴着滑开。

## 三、接口与入参出参

```csharp
public readonly struct ObstacleRow            // 一行障碍：id/dungeonId/kind/xMm/zMm/radiusMm/halfXMm/halfZMm/name
public static BattleField BattleField.From(IReadOnlyList<ObstacleRow> rows, int dungeonId);
public bool BattleField.IsBlocked(int xMm, int zMm);   // 圆用平方距离，矩形用 |dx|/|dz| 比半轴
public void BattleEntity.ApplyMove(int moveX, int moveY, int speedMmPerTick, int halfExtentMm, BattleField? field = null);
```

约定：`kind` 0 = 圆（只看 `radiusMm`），1 = 轴对齐矩形（只看 `halfXMm`/`halfZMm`）；半径或半轴 ≤ 0 的行跳过并计入 `Skipped`；`dungeonId` 不匹配的行不参与。

## 四、逐段讲解

1. `ServerTables.TryLoad` 多读一张 `BattleObstacle`，用 `Int`/`IntOr`/`StrOr` 逐列取，读不到就用兜底值并记一次「缺列」告警，最后按 `id` 排序，保证 `Skipped` 之外的顺序稳定。
2. `BattleField.From` 按副本过滤、按形状校验、把圆和矩形分别塞进两个数组；`IsBlocked` 只做整数比较，没有开方也没有浮点，30Hz 下每帧问几百次也不心疼。
3. `BattleEntity.ApplyMove` 先算出本帧位移（斜向按 `1000/1414` 缩放，和原来一样不超速），**先记下「起点是不是已经在障碍里」**，再挪位置。
4. 起点不在障碍里才调 `ApplyObstacles`：把这一帧的位移退回，先试 X 轴，X 通过了再试 Z 轴（Z 用已经采纳后的 X 去判），两个轴各试一次，于是贴墙会滑。
5. 「起点在障碍里就放行」是为了避免「障碍是后加的 / 出生点摆错」导致单位永久卡死：已经在里面的人可以走出来，但走进去的人进不去。

```csharp
// 这个文件里有：
//   BattleField（类）：把 BattleObstacle 表的行编译成"哪些坐标不能走"，并提供整数判定

using System;
using System.Collections.Generic;

namespace NBC.Server.Game;

/// <summary>
/// 一张副本战场的障碍判定表：读表后只留下"能不能站"，不涉及任何物理。
/// <para>
/// 主要数据成员：圆障碍（圆心 + 半径）与轴对齐矩形（中心 + 半宽半高），全部是整数毫米。
/// 主要外部方法：`From`（按副本过滤并校验）、`IsBlocked`（判断一个坐标是否被挡）、`IsEmpty`。
/// 谁在用它：`DungeonBattle` 建世界时造一份，`BattleEntity.ApplyMove` 每次位移前问它。
/// </para>
/// </summary>
public sealed class BattleField
{
    /// <summary>没有障碍的战场（空表 / 副本没有配障碍时用它，行为与改动前一致）。</summary>
    public static readonly BattleField Empty = new(Array.Empty<Circle>(), Array.Empty<Box>());

    /// <summary>一个圆障碍：圆心与半径，单位毫米。</summary>
    private readonly struct Circle
    {
        /// <summary>圆心 X（毫米）。</summary>
        public readonly int Xmm;

        /// <summary>圆心 Z（毫米）。</summary>
        public readonly int Zmm;

        /// <summary>半径（毫米）。</summary>
        public readonly int RadiusMm;

        /// <summary>记一个圆。</summary>
        /// <param name="xMm">圆心 X（毫米）。</param>
        /// <param name="zMm">圆心 Z（毫米）。</param>
        /// <param name="radiusMm">半径（毫米）。</param>
        public Circle(int xMm, int zMm, int radiusMm)
        {
            Xmm = xMm;
            Zmm = zMm;
            RadiusMm = radiusMm;
        }
    }

    /// <summary>一个轴对齐矩形障碍：中心与半宽半高，单位毫米。</summary>
    private readonly struct Box
    {
        /// <summary>中心 X（毫米）。</summary>
        public readonly int Xmm;

        /// <summary>中心 Z（毫米）。</summary>
        public readonly int Zmm;

        /// <summary>半宽（毫米，X 方向）。</summary>
        public readonly int HalfXMm;

        /// <summary>半高（毫米，Z 方向）。</summary>
        public readonly int HalfZMm;

        /// <summary>记一个矩形。</summary>
        /// <param name="xMm">中心 X（毫米）。</param>
        /// <param name="zMm">中心 Z（毫米）。</param>
        /// <param name="halfXMm">半宽（毫米）。</param>
        /// <param name="halfZMm">半高（毫米）。</param>
        public Box(int xMm, int zMm, int halfXMm, int halfZMm)
        {
            Xmm = xMm;
            Zmm = zMm;
            HalfXMm = halfXMm;
            HalfZMm = halfZMm;
        }
    }

    /// <summary>圆障碍列表（顺序与表里编号一致，判定结果因此可复现）。</summary>
    private readonly Circle[] m_circles;

    /// <summary>矩形障碍列表。</summary>
    private readonly Box[] m_boxes;

    /// <summary>记一份判定表（外部统一走 `From`，这里只给内部构造用）。</summary>
    /// <param name="circles">圆障碍。</param>
    /// <param name="boxes">矩形障碍。</param>
    private BattleField(Circle[] circles, Box[] boxes)
    {
        m_circles = circles;
        m_boxes = boxes;
        Count = circles.Length + boxes.Length;
    }

    /// <summary>生效的障碍条数。</summary>
    public int Count { get; }

    /// <summary>因为数值非法被跳过的条数（半径或半宽半高 ≤ 0）。</summary>
    public int Skipped { get; private set; }

    /// <summary>一条障碍都没有时为真，此时调用方可以直接走老路径。</summary>
    public bool IsEmpty => Count == 0;

    /// <summary>把表里的行按副本过滤并校验，编译成判定表。</summary>
    /// <param name="rows">`BattleObstacle` 表的全部行。</param>
    /// <param name="dungeonId">只取这个副本的行。</param>
    /// <returns>判定表；没有可用行时返回 `Empty`。</returns>
    /// <remarks>
    /// 校验规则：圆要求半径 &gt; 0；矩形要求半宽与半高都 &gt; 0。不合法的行跳过并计数，
    /// 由调用方（`DungeonBattle`）在日志里报出来，避免刷屏。
    /// </remarks>
    public static BattleField From(IReadOnlyList<ObstacleRow> rows, int dungeonId)
    {
        if (rows == null || rows.Count == 0)
        {
            return Empty;
        }

        var circles = new List<Circle>();
        var boxes = new List<Box>();
        int skipped = 0;

        for (int i = 0; i < rows.Count; i++)
        {
            ObstacleRow row = rows[i];

            if (row.DungeonId != dungeonId)
            {
                continue;
            }

            if (row.Kind == 0)
            {
                if (row.RadiusMm <= 0)
                {
                    skipped++;
                    continue;
                }

                circles.Add(new Circle(row.Xmm, row.Zmm, row.RadiusMm));
            }
            else if (row.Kind == 1)
            {
                if (row.HalfXMm <= 0 || row.HalfZMm <= 0)
                {
                    skipped++;
                    continue;
                }

                boxes.Add(new Box(row.Xmm, row.Zmm, row.HalfXMm, row.HalfZMm));
            }
            else
            {
                skipped++;
            }
        }

        if (circles.Count == 0 && boxes.Count == 0)
        {
            return Empty;
        }

        return new BattleField(circles.ToArray(), boxes.ToArray()) { Skipped = skipped };
    }

    /// <summary>判断某个坐标是否落在障碍内部（含边界）。</summary>
    /// <param name="xMm">X（毫米）。</param>
    /// <param name="zMm">Z（毫米）。</param>
    /// <returns>被挡住返回 true；场地边界不在这里判（由调用方的钳位负责）。</returns>
    public bool IsBlocked(int xMm, int zMm)
    {
        for (int i = 0; i < m_circles.Length; i++)
        {
            Circle circle = m_circles[i];
            long dx = xMm - circle.Xmm;
            long dz = zMm - circle.Zmm;
            long r = circle.RadiusMm;

            // 平方比较，避免开方；用 long 防止 20000 级别的平方溢出 int
            if (dx * dx + dz * dz <= r * r)
            {
                return true;
            }
        }

        for (int i = 0; i < m_boxes.Length; i++)
        {
            Box box = m_boxes[i];

            if (Math.Abs(xMm - box.Xmm) <= box.HalfXMm && Math.Abs(zMm - box.Zmm) <= box.HalfZMm)
            {
                return true;
            }
        }

        return false;
    }
}
```

```csharp
// 这个文件里有：
//   BattleObstacleMarker（MonoBehaviour）：在编辑场景里描述一个战场障碍（圆或轴对齐矩形），
//   用 Gizmos 画出来，供编辑器菜单写回 Configs/Design/BattleObstacle.csv

using UnityEngine;

namespace NBC.Game.View
{
    // 主要功能：把"这个位置有一根柱子/一块石头"变成可写回配置的数据（形状 + 尺寸，单位毫米）。
    // 主要数据成员：Kind（0 圆 / 1 矩形）、RadiusMm、HalfXMm、HalfZMm。
    // 主要外部方法：无。它自己不参与运行时逻辑，只给编辑器菜单扫。
    // 谁在用它：Editor 的 BattleObstacleWriter（写回 CSV）；Scene 视图的 Gizmos 显示。
    /// <summary>
    /// 场景里的一个障碍标记。位置就是障碍中心（只看 X/Z，Y 只影响 Gizmos 画在哪），
    /// 尺寸填毫米，与服务端 `BattleField` 的判定口径一致。
    /// <para>注意：它不参与运行时碰撞 —— 判定在服务端，客户端只显示快照位置。</para>
    /// </summary>
    public sealed class BattleObstacleMarker : MonoBehaviour
    {
        /// <summary>形状：0 = 圆（用 RadiusMm），1 = 轴对齐矩形（用 HalfXMm / HalfZMm）。</summary>
        [Tooltip("0 = 圆（填半径），1 = 矩形（填半宽半高）")]
        [Range(0, 1)]
        public int Kind;

        /// <summary>圆的半径（毫米）。Kind = 0 时生效。</summary>
        [Tooltip("圆的半径（毫米），Kind = 0 时生效")]
        [Min(0)]
        public int RadiusMm = 800;

        /// <summary>矩形的半宽（毫米，X 方向）。Kind = 1 时生效。</summary>
        [Tooltip("矩形的半宽（毫米，X 方向），Kind = 1 时生效")]
        [Min(0)]
        public int HalfXMm = 1000;

        /// <summary>矩形的半高（毫米，Z 方向）。Kind = 1 时生效。</summary>
        [Tooltip("矩形的半高（毫米，Z 方向），Kind = 1 时生效")]
        [Min(0)]
        public int HalfZMm = 500;

        /// <summary>本标记在场景里的中心（米，含 Y；服务端只用 X/Z）。</summary>
        public Vector3 Center => transform.position;

        /// <summary>
        /// 在 Scene 视图画出障碍范围：红色线框。
        /// <para>只在编辑器里画（Gizmos 不进包），所以玩家在游戏里看不到 —— 要看画布效果请用真实模型。</para>
        /// </summary>
        private void OnDrawGizmos()
        {
            Gizmos.color = new Color(1f, 0.25f, 0.2f, 0.9f);

            Vector3 center = transform.position;

            if (Kind == 0)
            {
                Gizmos.DrawWireSphere(center, RadiusMm * 0.001f);
                return;
            }

            Gizmos.DrawWireCube(center,
                new Vector3(HalfXMm * 0.002f, 0.05f, HalfZMm * 0.002f));
        }

        /// <summary>选中时把地面轮廓画成半透明，方便对齐位置。</summary>
        private void OnDrawGizmosSelected()
        {
            Gizmos.color = new Color(1f, 0.25f, 0.2f, 0.15f);
            Vector3 center = transform.position;

            if (Kind == 0)
            {
                Gizmos.DrawSphere(center, RadiusMm * 0.001f);
                return;
            }

            Gizmos.DrawCube(center, new Vector3(HalfXMm * 0.002f, 0.05f, HalfZMm * 0.002f));
        }
    }
}
```

### 5.3 `Client/Assets/_Project/Editor/BattleObstacleWriter.cs`

```csharp
// 这个文件里有：
//   BattleObstacleWriter（编辑器菜单）：把场景里的 BattleObstacleMarker 写回
//   Configs/Design/BattleObstacle.csv，并在 Console 报出条数与校验问题

using System.Collections.Generic;
using System.IO;
using System.Text;
using NBC.Game.View;
using UnityEditor;
using UnityEngine;

namespace NBC.Editor
{
    // 主要功能：把"场景里摆好的障碍标记"转成服务端能读的 CSV（§0.4：界面里调，写回配置源）。
    // 主要数据成员：CsvPath / CsvHeader（表头三行 + 规则行，与 ConfigKit 约定一致）。
    // 主要外部方法：WriteObstacles（菜单入口）。
    // 谁在用它：负责人从菜单 Tools/NBC/战场/写回障碍到 CSV 调用。
    /// <summary>
    /// 障碍写回工具：扫当前打开场景里的所有 <see cref="BattleObstacleMarker"/>，
    /// 按名字排序后覆盖写 `Configs/Design/BattleObstacle.csv`。
    /// <para>注意：写回后必须重跑 ConfigKit 并重启服务端才生效（与 Monster 表一致）。</para>
    /// </summary>
    internal static class BattleObstacleWriter
    {
        /// <summary>CSV 相对工程根的位置。</summary>
        private const string CsvPath = "Configs/Design/BattleObstacle.csv";

        /// <summary>默认副本编号（本计划只有 1001 一个副本；以后从 Dungeon 表取）。</summary>
        private const int DungeonId = 1001;

        /// <summary>障碍离出生点太近的告警半径（毫米）。</summary>
        private const int SpawnGuardMm = 1500;

        /// <summary>提示里用的出生点（与服务端 `RoomBattleService` 一致：(-500, 0) 毫米）。</summary>
        private static readonly Vector2 SpawnMm = new Vector2(-500f, 0f);

        /// <summary>
        /// 菜单入口：扫场景 → 校验 → 写 CSV → Console 与弹窗汇报。
        /// <para>步骤：1) 找标记并排序（编号稳定，便于 diff）2) 逐条校验尺寸与出生点距离
        /// 3) 拼 CSV 文本 4) 写文件 5) 汇报。</para>
        /// </summary>
        [MenuItem("Tools/NBC/战场/写回障碍到 CSV", priority = 210)]
        private static void WriteObstacles()
        {
            BattleObstacleMarker[] markers =
                Object.FindObjectsOfType<BattleObstacleMarker>(true);

            if (markers.Length == 0)
            {
                EditorUtility.DisplayDialog("写回障碍",
                    "当前场景里没有任何 BattleObstacleMarker。\n\n" +
                    "请先建一个空物体、挂上 BattleObstacleMarker 当障碍。", "知道了");
                return;
            }

            var sorted = new List<BattleObstacleMarker>(markers);
            sorted.Sort((left, right) =>
                string.CompareOrdinal(left.gameObject.name, right.gameObject.name));

            var problems = new List<string>();
            var rows = new StringBuilder();

            for (int i = 0; i < sorted.Count; i++)
            {
                BattleObstacleMarker marker = sorted[i];
                Vector3 center = marker.Center;
                int xMm = Mathf.RoundToInt(center.x * 1000f);
                int zMm = Mathf.RoundToInt(center.z * 1000f);

                if (marker.Kind == 0 && marker.RadiusMm <= 0)
                {
                    problems.Add($"{marker.gameObject.name}：圆障碍半径必须 > 0");
                }

                if (marker.Kind == 1 && (marker.HalfXMm <= 0 || marker.HalfZMm <= 0))
                {
                    problems.Add($"{marker.gameObject.name}：矩形障碍的半宽与半高都必须 > 0");
                }

                float dx = xMm - SpawnMm.x;
                float dz = zMm - SpawnMm.y;

                if (Mathf.Sqrt(dx * dx + dz * dz) < SpawnGuardMm)
                {
                    problems.Add($"{marker.gameObject.name}：离出生点 (-500, 0) 不到 {SpawnGuardMm}mm，" +
                                 "出生时可能被卡住");
                }

                rows.Append(i + 1).Append(',')
                    .Append(DungeonId).Append(',')
                    .Append(marker.Kind).Append(',')
                    .Append(xMm).Append(',')
                    .Append(zMm).Append(',')
                    .Append(marker.Kind == 0 ? marker.RadiusMm : 0).Append(',')
                    .Append(marker.Kind == 1 ? marker.HalfXMm : 0).Append(',')
                    .Append(marker.Kind == 1 ? marker.HalfZMm : 0).Append(',')
                    .Append(marker.gameObject.name)
                    .Append('\n');
            }

            string header =
                "id,dungeonId,kind,xMm,zMm,radiusMm,halfXMm,halfZMm,name\n" +
                "编号,副本编号,形状(0圆1方),中心X（毫米）,中心Z（毫米）,半径（毫米）,半宽（毫米）,半高（毫米）,备注\n" +
                "int,int,int,int,int,int,int,int,string\n" +
                "key,\"range(1,999999)\",\"range(0,1)\",,,\"range(0,20000)\",\"range(0,20000)\",\"range(0,20000)\",\"len(0,32)\"\n";

            string path = Path.Combine(GetProjectRoot(), CsvPath);
            File.WriteAllText(path, header + rows, new UTF8Encoding(false));

            Debug.Log($"[障碍] 已写回 {sorted.Count} 条到 {CsvPath}（副本 {DungeonId}）");

            foreach (string problem in problems)
            {
                Debug.LogWarning("[障碍] " + problem);
            }

            string detail = problems.Count == 0
                ? $"已写回 {sorted.Count} 条障碍。\n\n下一步：重启服务端才生效。"
                : $"已写回 {sorted.Count} 条，但有 {problems.Count} 个问题（见 Console）：\n\n" +
                  string.Join("\n", problems);

            EditorUtility.DisplayDialog("写回障碍", detail, "知道了");
        }

        /// <summary>取工程根目录（`Assets` 的上一级）。</summary>
        /// <returns>绝对路径，末尾不带分隔符。</returns>
        private static string GetProjectRoot()
            => Directory.GetParent(Application.dataPath)!.FullName;
    }
}
```

## 六、调用示例

1. 在 Battle 场景里新建一个空物体，挂 `BattleObstacleMarker`，`Kind = 0`、`RadiusMm = 800`，摆在你想挡人的位置。
2. 点菜单 `Tools/NBC/战场/写回障碍到 CSV`，Console 会打「`[障碍] 已写回 N 条 …`」。
3. 重编服务端并重启（表在 `AttachTables` 时读），进副本往那个圆走——会停在圆边并贴边滑开。

## 七、风险与幻觉自查

| 风险 | 自查结果 |
