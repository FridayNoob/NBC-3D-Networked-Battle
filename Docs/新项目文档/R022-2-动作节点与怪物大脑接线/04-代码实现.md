## 二、功能概述

怪物每帧：刷新仇恨范围内最近英雄 → Tick 行为树 → 把 `ctx.Chasing` 写回实体 → 记状态名。
7 个动作里 `CastSkill` 仍是占位（恒失败 + 首次告警），其余全部可用。

## 三、接口与入参出参

| 成员 | 入参 | 出参 |
| --- | --- | --- |
| `BtActionOps.FindNearestHero(ctx)` | 上下文 | 仇恨内最近英雄 / null |
| `BtActionOps.MoveTowards/MoveHome/TryAttack(ctx)` | 上下文 | 无 / bool |
| `BtOps.SquaredDistanceTo(ctx)` | 上下文 | `long` 距离平方 |
| `BtOps.WithinRange(ctx, rangeMm)` | 上下文 + 射程 | bool（圆形，含边界） |
| `MonsterBrain.Think(world)` | 世界 | 无（写 `StateName`） |
| `MonsterRow`（8 参构造） | id/name/hp/attack/aggro/attackRange/speed/leash | 只读行 |

## 四、关键修复：两把尺子

**现象**：接线后探针出现「英雄在 1500mm 内攻击却被判射程外（2256mm > 2000mm）」。
**根因**：`DungeonBattle.DistanceMm` 是**切比雪夫距离**（max(|dx|,|dz|)），只用于 AI 的"离家/仇恨"判断；
而攻击判定用 `BattleRules.WithinBasicAttackRange`（**圆形**，欧氏距离平方）。
我在 R022-1 的 `InAttackRange` 里用了切比雪夫 → AI 认为"到 1800 就贴脸"，实际欧氏 2256，攻击永远打不出。
**修法**：`BtOps` 新增 `SquaredDistanceTo` / `WithinRange`（圆形、整数、含边界），
`InAttackRange`、`Chase`（90% 贴脸缓冲）、`BasicAttack` 三处统一改用它。

### `Ai/Bt/Actions.cs`

```csharp
// 这个文件里有：
//   BtActionOps（静态类）：动作节点用的世界操作（找目标、移动、回位、普攻）。
//   ChaseAction / BasicAttackAction / CastSkillAction / ReturnHomeAction / PatrolAction / IdleAction / WaitAction（类）：7 个动作节点。
//
// 动作节点是唯一允许修改 ctx.Chasing 与世界的节点类型；条件节点保持只读。

using System;
using System.Collections.Generic;

namespace NBC.Server.Game.Ai.Bt
{
    // 主要功能：动作节点能用的世界操作 —— 找目标、朝目标走、回家、普攻。
    // 主要数据成员：无（全静态）。
    // 主要外部方法：FindNearestHero / MoveTowards / MoveHome / TryAttack。
    // 谁在用它：7 个动作节点、MonsterBrain（每帧刷新目标）。
    // 注意：所有规则（射程、冷却、钳位）仍由 DungeonBattle 判，这里只负责发起。
    public static class BtActionOps
    {
        // 功能：找仇恨范围内最近的活英雄。
        // 入参：ctx 上下文
        // 返回：英雄；范围内没有返回 null
        // 步骤：取最近英雄 → 超过 Profile.AggroRangeMm 就丢弃
        // 上游：MonsterBrain.Think
        // 下游：DungeonBattle.NearestAliveHero / DistanceMm
        public static BattleEntity FindNearestHero(BtContext ctx)
        {
            if (ctx?.World == null || ctx.Self == null)
            {
                return null;
            }

            BattleEntity hero = ctx.World.NearestAliveHero(ctx.Self);

            if (hero == null)
            {
                return null;
            }

            int range = ctx.Profile.AggroRangeMm;

            if (range > 0 && DungeonBattle.DistanceMm(ctx.Self, hero) > range)
            {
                return null;
            }

            return hero;
        }

        // 功能：朝目标移动一帧。
        // 入参：ctx 上下文
        // 返回：无
        // 步骤：目标空就返回 → 调世界移动
        // 上游：ChaseAction
        // 下游：DungeonBattle.MoveTowards
        public static void MoveTowards(BtContext ctx)
        {
            if (ctx?.Self == null || ctx.Target == null)
            {
                return;
            }

            ctx.World.MoveTowards(ctx.Self, ctx.Target);
        }

        // 功能：朝出生点移动一帧。
        // 入参：ctx 上下文
        // 返回：无
        // 步骤：调世界按点移动（世界负责钳位）
        // 上游：ReturnHomeAction
        // 下游：DungeonBattle.MoveTowardsPoint
        public static void MoveHome(BtContext ctx)
        {
            if (ctx?.Self == null)
            {
                return;
            }

            ctx.World.MoveTowardsPoint(ctx.Self, ctx.HomeXmm, ctx.HomeZmm);
        }

        // 功能：对目标发起一次普攻（世界判射程与冷却）。
        // 入参：ctx 上下文
        // 返回：true = 真的打出去了
        // 步骤：目标空就失败 → 调世界普攻
        // 上游：BasicAttackAction
        // 下游：DungeonBattle.TryBasicAttack
        public static bool TryAttack(BtContext ctx)
        {
            if (ctx?.Self == null || ctx.Target == null)
            {
                return false;
            }

            return ctx.World.TryBasicAttack(ctx.Self, ctx.Target.Id, out _);
        }
    }

    // 主要功能：追击 —— 设置 Chasing 并朝目标走一帧，贴到射程内就成功。
    // 主要数据成员：无。
    // 主要外部方法：Tick。
    // 谁在用它：wolf / boss 的「有目标」分支。
    // 注意：**没贴脸时必须返回 Running**；返回 Success 会让 Selector 吃掉后面的攻击分支。
    [BtNode("Chase")]
    public sealed class ChaseAction : BtNode
    {
        // 类型名
        public override string TypeName => "Chase";

        // 功能：追一帧。
        // 入参：ctx 上下文
        // 返回：贴到射程内 → Success（并清 Chasing）；否则 Running
        // 步骤：设 Chasing → 判距离 → 没到就走一帧
        // 上游：父节点
        // 下游：BtActionOps.MoveTowards
        public override BtStatus Tick(BtContext ctx)
        {
            if (ctx?.Self == null || ctx.Target == null)
            {
                return BtStatus.Failure;
            }

            int range = ctx.Profile.AttackRangeMm > 0 ? ctx.Profile.AttackRangeMm : BtOps.DefaultAttackRangeMm;

            // 贴脸缓冲：停在射程的 90% 处，别卡在"2000/2001"的边界上来回抖
            int stopRange = range - range / 10;

            if (stopRange < 1)
            {
                stopRange = range;
            }

            if (BtOps.WithinRange(ctx, stopRange))
            {
                ctx.Chasing = false;
                ctx.LastAction = "追击";
                return BtStatus.Success;
            }

            ctx.Chasing = true;
            ctx.LastAction = "追击";
            BtActionOps.MoveTowards(ctx);
            return BtStatus.Running;
        }
    }

    // 主要功能：普攻 —— 多段时按帧分段，每段仍受世界普攻冷却约束。
    // 主要数据成员：m_hits（总段数）、m_intervalTicks（段最小间隔）、m_done（已打出的段数）、m_waitTicks。
    // 主要外部方法：Configure、Tick。
    // 谁在用它：wolf（1 段）/ boss（3 段）。
    // 注意：世界冷却（15 tick）大于段间隔时，实际节奏由世界冷却决定 —— 不重复实现伤害规则。
    [BtNode("BasicAttack")]
    public sealed class BasicAttackAction : BtNode
    {
        // 总段数（默认 1）
        private int m_hits = 1;

        // 段最小间隔（逻辑帧，默认 5）
        private int m_intervalTicks = 5;

        // 已打出的段数
        private int m_done;

        // 距下一段的等待帧数
        private int m_waitTicks;

        // 类型名
        public override string TypeName => "BasicAttack";

        // 功能：读 hits 与 intervalTicks。
        // 入参：source 参数来源
        // 返回：无
        // 步骤：读两个整数，非法值夹到 >=1
        // 上游：R022-3 建树
        // 下游：Tick
        public override void Configure(IBtParamSource source)
        {
            if (source == null)
            {
                return;
            }

            if (source.TryGetInt("hits", out int hits) && hits >= 1)
            {
                m_hits = hits;
            }

            if (source.TryGetInt("intervalTicks", out int interval) && interval >= 1)
            {
                m_intervalTicks = interval;
            }
        }

        // 功能：打一段或等冷却。
        // 入参：ctx 上下文
        // 返回：全部打完 → Success；还差几段 → Running；没目标/超射程 → Failure
        // 步骤：无目标或超射程就重置并失败 → 等待帧未到返回 Running → 尝试普攻
        // 上游：父节点
        // 下游：BtActionOps.TryAttack
        public override BtStatus Tick(BtContext ctx)
        {
            if (ctx?.Self == null || ctx.Target == null)
            {
                ResetCombo();
                return BtStatus.Failure;
            }

            if (!BtOps.WithinRange(ctx, GetRange(ctx)))
            {
                ResetCombo();
                return BtStatus.Failure;
            }

            if (m_waitTicks > 0)
            {
                m_waitTicks--;
                ctx.LastAction = "普攻间隔";
                return BtStatus.Running;
            }

            if (BtActionOps.TryAttack(ctx))
            {
                m_done++;
                m_waitTicks = m_intervalTicks;
                ctx.LastAction = m_hits > 1 ? "普攻 " + m_done + "/" + m_hits : "普攻";
            }
            else
            {
                // 冷却还没好：本帧没打出去，下一帧再试（不重置段数）
                ctx.LastAction = "等普攻冷却";
                return BtStatus.Running;
            }

            if (m_done >= m_hits)
            {
                ResetCombo();
                return BtStatus.Success;
            }

            return BtStatus.Running;
        }

        // 功能：当前射程（表里的 attackRangeMm）。
        // 入参：ctx 上下文
        // 返回：射程（毫米）
        // 步骤：表值 <=0 时回落默认 2000
        // 上游：Tick
        // 下游：无
        private static int GetRange(BtContext ctx)
        {
            return ctx.Profile.AttackRangeMm > 0 ? ctx.Profile.AttackRangeMm : BtOps.DefaultAttackRangeMm;
        }

        // 功能：清零多段状态。
        // 入参：无
        // 返回：无
        // 步骤：段数与等待都归零
        // 上游：Tick
        // 下游：无
        private void ResetCombo()
        {
            m_done = 0;
            m_waitTicks = 0;
        }
    }

    // 主要功能：放技能 —— 服务端技能系统未实现前的占位，恒失败。
    // 主要数据成员：m_skillId、s_warned（只告警一次）。
    // 主要外部方法：Configure、Tick。
    // 谁在用它：暂时没有树用它（BOSS 树里不放），留给技能系统落地后接。
    // 注意：恒 Failure 是刻意的；一旦技能系统可用，只改这里。
    [BtNode("CastSkill")]
    public sealed class CastSkillAction : BtNode
    {
        // 技能编号
        private int m_skillId;

        // 是否已告警
        private static bool s_warned;

        // 类型名
        public override string TypeName => "CastSkill";

        // 功能：读 skillId。
        // 入参：source 参数来源
        // 返回：无
        // 步骤：读不到就用 0
        // 上游：R022-3 建树
        // 下游：Tick
        public override void Configure(IBtParamSource source)
        {
            if (source != null && source.TryGetInt("skillId", out int skillId))
            {
                m_skillId = skillId;
            }
        }

        // 功能：恒失败（技能系统未做）。
        // 入参：ctx 上下文
        // 返回：Failure
        // 步骤：首次调用打一条告警
        // 上游：父节点
        // 下游：无
        public override BtStatus Tick(BtContext ctx)
        {
            if (!s_warned)
            {
                s_warned = true;
                Console.WriteLine("[BT] 服务端还没有技能系统：CastSkill(skillId=" + m_skillId + ") 恒为失败。");
            }

            return BtStatus.Failure;
        }
    }

    // 主要功能：回位 —— 朝出生点走，进了容差就成功。
    // 主要数据成员：无（容差用常量 200mm）。
    // 主要外部方法：Tick。
    // 谁在用它：boss 的 HomeFar 分支、普通怪的 leash。
    // 注意：容差 200mm 与 BOSS 的回位半径一致；不要写成 0（会永远差一点点）。
    [BtNode("ReturnHome")]
    public sealed class ReturnHomeAction : BtNode
    {
        // 到家容差（毫米）
        private const int HomeToleranceMm = 200;

        // 类型名
        public override string TypeName => "ReturnHome";

        // 功能：回一帧。
        // 入参：ctx 上下文
        // 返回：到家 → Success；否则 Running
        // 步骤：算离家距离 → 够近就成功并清 Chasing → 否则走一帧
        // 上游：父节点
        // 下游：BtActionOps.MoveHome
        public override BtStatus Tick(BtContext ctx)
        {
            if (ctx?.Self == null)
            {
                return BtStatus.Failure;
            }

            if (BtOps.DistanceToHome(ctx) <= HomeToleranceMm)
            {
                ctx.Chasing = false;
                ctx.LastAction = "回位到位";
                return BtStatus.Success;
            }

            ctx.Chasing = true;
            ctx.LastAction = "回位";
            BtActionOps.MoveHome(ctx);
            return BtStatus.Running;
        }
    }

    // 主要功能：巡逻 —— 清掉 Chasing，交给实体自身的巡逻位移。
    // 主要数据成员：无。
    // 主要外部方法：Tick。
    // 谁在用它：wolf 的默认分支。
    // 注意：不清 Chasing 的话，实体的 Step 会跳过巡逻，表现为"站着不动"。
    [BtNode("Patrol")]
    public sealed class PatrolAction : BtNode
    {
        // 类型名
        public override string TypeName => "Patrol";

        // 功能：允许巡逻。
        // 入参：ctx 上下文
        // 返回：恒 Success
        // 步骤：清 Chasing + 记动作名
        // 上游：父节点
        // 下游：无
        public override BtStatus Tick(BtContext ctx)
        {
            if (ctx != null)
            {
                ctx.Chasing = false;
                ctx.LastAction = "巡逻";
            }

            return BtStatus.Success;
        }
    }

    // 主要功能：站桩 —— 清掉 Chasing，且不允许巡逻。
    // 主要数据成员：无。
    // 主要外部方法：Tick。
    // 谁在用它：boss 的默认分支、PatrolWhenIdle=false 的怪。
    // 注意：与 Patrol 的区别只有"实体是否自己巡逻"，这里靠 Profile.PatrolWhenIdle 决定是否额外拦住。
    [BtNode("Idle")]
    public sealed class IdleAction : BtNode
    {
        // 类型名
        public override string TypeName => "Idle";

        // 功能：站住。
        // 入参：ctx 上下文
        // 返回：恒 Success
        // 步骤：清 Chasing + 记动作名
        // 上游：父节点
        // 下游：无
        public override BtStatus Tick(BtContext ctx)
        {
            if (ctx != null)
            {
                ctx.Chasing = false;
                ctx.LastAction = "站桩";
            }

            return BtStatus.Success;
        }
    }

    // 主要功能：等待 —— 按逻辑帧计时。
    // 主要数据成员：m_seconds（秒）、m_framesLeft（剩余帧）。
    // 主要外部方法：Configure、Tick。
    // 谁在用它：技能后摇、段间隔等（当前默认树未用）。
    // 注意：30Hz 换算，向上取整；每帧只减一，不用真实时间。
    [BtNode("Wait")]
    public sealed class WaitAction : BtNode
    {
        // 每逻辑帧的帧率（与服务端一致）
        private const int TickRate = 30;

        // 总帧数（由秒数换算）
        private int m_totalFrames = TickRate;

        // 剩余帧数
        private int m_framesLeft;

        // 类型名
        public override string TypeName => "Wait";

        // 功能：读秒数并换算帧数。
        // 入参：source 参数来源
        // 返回：无
        // 步骤：读 seconds → 乘 30 向上取整 → 夹到至少 1 帧
        // 上游：R022-3 建树
        // 下游：Tick
        public override void Configure(IBtParamSource source)
        {
            if (source == null || !source.TryGetFloat("seconds", out float seconds))
            {
                return;
            }

            int frames = (int)Math.Ceiling(seconds * TickRate);
            m_totalFrames = frames < 1 ? 1 : frames;
        }

        // 功能：计时。
        // 入参：ctx 上下文
        // 返回：到点 → Success；否则 Running
        // 步骤：本段开始时装载剩余帧 → 每帧减一
        // 上游：父节点
        // 下游：无
        public override BtStatus Tick(BtContext ctx)
        {
            if (m_framesLeft <= 0)
            {
                m_framesLeft = m_totalFrames;
            }

            m_framesLeft--;

            if (ctx != null)
            {
                ctx.LastAction = "等待";
            }

            return m_framesLeft > 0 ? BtStatus.Running : BtStatus.Success;
        }
    }
}
```

### `Ai/Bt/BtTreeBuilder.cs`

```csharp
// 这个文件里有：
//   BtParams（类）：服务端用的参数来源实现（R022-3 会换成从 JSON 读；当前由代码填）。
//   BtTreeBuilder（静态类）：用代码搭 wolf / boss 两套默认树。
//
// 树的结构与策划在编辑器里搭的一致；R022-3 落地后这里改成"读 JSON 建树"，本文件会缩掉。

using System.Collections.Generic;

namespace NBC.Server.Game.Ai.Bt
{
    // 主要功能：把字符串键值喂给节点读参数（Configure 阶段用）。
    // 主要数据成员：m_values（键 → 文本）。
    // 主要外部方法：Set、TryGetInt、TryGetBool、TryGetFloat。
    // 谁在用它：BtTreeBuilder、R022-3 的 JSON 解析。
    // 注意：只在建树阶段用，Tick 阶段节点不再读它。
    public sealed class BtParams : IBtParamSource
    {
        // 键 → 文本值
        private readonly Dictionary<string, string> m_values = new Dictionary<string, string>();

        // 功能：填一个参数。
        // 入参：key 键；value 文本值
        // 返回：自身（便于链式书写）
        // 步骤：写字典
        // 上游：BtTreeBuilder
        // 下游：节点的 Configure
        public BtParams Set(string key, string value)
        {
            m_values[key] = value;
            return this;
        }

        // 功能：读整数。
        // 入参：key 键；value 值
        // 返回：true = 存在且可解析
        // 步骤：查字典 + int.TryParse
        // 上游：节点 Configure
        // 下游：无
        public bool TryGetInt(string key, out int value)
        {
            value = 0;
            return m_values.TryGetValue(key, out string raw) && int.TryParse(raw, out value);
        }

        // 功能：读布尔。
        // 入参：key 键；value 值
        // 返回：true = 存在且可解析
        // 步骤：查字典 + bool.TryParse
        // 上游：节点 Configure
        // 下游：无
        public bool TryGetBool(string key, out bool value)
        {
            value = false;
            return m_values.TryGetValue(key, out string raw) && bool.TryParse(raw, out value);
        }

        // 功能：读浮点（固定用 InvariantCulture，避免小数点随地区变化）。
        // 入参：key 键；value 值
        // 返回：true = 存在且可解析
        // 步骤：查字典 + float.TryParse(InvariantCulture)
        // 上游：节点 Configure
        // 下游：无
        public bool TryGetFloat(string key, out float value)
        {
            value = 0f;
            return m_values.TryGetValue(key, out string raw)
                   && float.TryParse(raw, System.Globalization.NumberStyles.Float,
                       System.Globalization.CultureInfo.InvariantCulture, out value);
        }
    }

    // 主要功能：搭默认树 —— 普通怪（wolf）与 BOSS（boss）各一棵。
    // 主要数据成员：WolfAiKind / BossAiKind（种类名）。
    // 主要外部方法：Build(aiKind)。
    // 谁在用它：DungeonBattle 建怪时、MonsterBrain 构造。
    // 注意：树的形状要与客户端编辑器里搭的一致；R022-3 用 JSON 取代本文件。
    public static class BtTreeBuilder
    {
        // 普通怪的种类名
        public const string WolfAiKind = "wolf";

        // BOSS 的种类名
        public const string BossAiKind = "boss";

        // 功能：按种类搭树。
        // 入参：aiKind 种类名（未知当作 wolf）
        // 返回：根节点
        // 步骤：boss → BuildBoss；其它 → BuildWolf
        // 上游：MonsterBrain 构造
        // 下游：各节点构造函数
        public static BtNode Build(string aiKind)
        {
            return aiKind == BossAiKind ? BuildBoss() : BuildWolf();
        }

        // 功能：普通怪：死了停 → 能打就打（1 段）→ 有目标就追 → 巡逻。
        // 入参：无
        // 返回：根节点
        // 步骤：逐层 AddChild
        // 上游：Build
        // 下游：条件/动作节点
        private static BtNode BuildWolf()
        {
            BtSelector root = new BtSelector();
            root.AddChild(new IsDeadCondition());

            BtSequence attack = new BtSequence();
            attack.AddChild(new HasTargetCondition());
            attack.AddChild(Configured(new InAttackRangeCondition(), new BtParams()));
            attack.AddChild(Configured(new AttackReadyCondition(), new BtParams().Set("skillId", "0")));
            attack.AddChild(Configured(new BasicAttackAction(), new BtParams().Set("hits", "1").Set("intervalTicks", "5")));
            root.AddChild(attack);

            BtSequence chase = new BtSequence();
            chase.AddChild(new HasTargetCondition());
            chase.AddChild(new ChaseAction());
            root.AddChild(chase);

            root.AddChild(new PatrolAction());
            return root;
        }

        // 功能：BOSS：死了停 → 能打就打（3 段）→ 有目标就追 → 离家远就回位 → 站桩。
        // 入参：无
        // 返回：根节点
        // 步骤：逐层 AddChild
        // 上游：Build
        // 下游：条件/动作节点
        private static BtNode BuildBoss()
        {
            BtSelector root = new BtSelector();
            root.AddChild(new IsDeadCondition());

            BtSequence attack = new BtSequence();
            attack.AddChild(new HasTargetCondition());
            attack.AddChild(Configured(new InAttackRangeCondition(), new BtParams()));
            attack.AddChild(Configured(new AttackReadyCondition(), new BtParams().Set("skillId", "0")));
            attack.AddChild(Configured(new BasicAttackAction(), new BtParams().Set("hits", "3").Set("intervalTicks", "5")));
            root.AddChild(attack);

            BtSequence chase = new BtSequence();
            chase.AddChild(new HasTargetCondition());
            chase.AddChild(new ChaseAction());
            root.AddChild(chase);

            BtSequence home = new BtSequence();
            home.AddChild(new HomeFarCondition());
            home.AddChild(new ReturnHomeAction());
            root.AddChild(home);

            root.AddChild(new IdleAction());
            return root;
        }

        // 功能：给节点灌参数（小工具，避免每处写两行）。
        // 入参：node 节点；parameters 参数
        // 返回：同一个节点
        // 步骤：调 Configure
        // 上游：BuildWolf / BuildBoss
        // 下游：节点.Configure
        private static BtNode Configured(BtNode node, BtParams parameters)
        {
            node.Configure(parameters);
            return node;
        }
    }
}
```

### `Ai/Bt/MonsterBrain.cs`

```csharp
// 这个文件里有：
//   MonsterBrain（类）：一只怪的大脑 —— 每逻辑帧刷新目标、Tick 行为树、维护状态名。
//
// 目标检测放在大脑里而不是树节点：树里的"检测节点"一旦返回成功，就会让 Selector 吃掉整帧（R022 的老坑）。

namespace NBC.Server.Game.Ai.Bt
{
    // 主要功能：把"世界 + 行为树"接起来 —— 每帧刷新最近英雄、Tick 根节点、把状态写回实体。
    // 主要数据成员：EntityId / HomeXmm / HomeZmm / StateName / m_ctx / m_root。
    // 主要外部方法：Think(world)。
    // 谁在用它：DungeonBattle 的每帧循环（`_brains`）。
    // 注意：`m_ctx` 每只怪一个、跨帧复用；`Self.Chasing` 每帧由 `ctx.Chasing` 覆盖，避免动作切换后残留。
    public sealed class MonsterBrain
    {
        // 大脑驱动的实体编号
        public int EntityId { get; }

        // 出生点 X（毫米）
        public int HomeXmm { get; }

        // 出生点 Z（毫米）
        public int HomeZmm { get; }

        // 当前状态名（日志/调试用：追击 / 普攻 / 回位 / 巡逻 / 站桩）
        public string StateName => m_ctx.LastAction ?? "站桩";

        // 状态切换次数（探针用）
        public int TransitionCount { get; private set; }

        // 复用的上下文
        private readonly BtContext m_ctx;

        // 树根
        private readonly BtNode m_root;

        // 上一次的状态名（统计切换次数）
        private string m_lastStateName;

        // 功能：建大脑。
        // 入参：self 实体；aiKind 种类名（wolf / boss）；profile 该怪型的 AI 参数
        // 返回：无
        // 步骤：记实体与出生点 → 建树 → 建上下文
        // 上游：DungeonBattle.BuildDungeon
        // 下游：BtTreeBuilder.Build
        public MonsterBrain(BattleEntity self, string aiKind, MonsterAiProfile profile)
        {
            EntityId = self.Id;
            HomeXmm = self.PosXmm;
            HomeZmm = self.PosZmm;
            m_root = BtTreeBuilder.Build(aiKind);
            m_ctx = new BtContext
            {
                Self = self,
                HomeXmm = HomeXmm,
                HomeZmm = HomeZmm,
                Profile = profile
            };
        }

        // 功能：跑一帧。
        // 入参：world 战斗世界
        // 返回：无
        // 步骤：挂世界 → 刷新目标（仇恨范围内最近英雄）→ 清本帧动作 → Tick 树 → 回写 Chasing → 记状态
        // 上游：DungeonBattle.Tick
        // 下游：BtNode.Tick
        public void Think(DungeonBattle world)
        {
            if (world == null)
            {
                return;
            }

            m_ctx.World = world;
            m_ctx.Tick++;
            m_ctx.Target = BtActionOps.FindNearestHero(m_ctx);
            m_ctx.Chasing = false;
            m_ctx.LastAction = null;

            m_root.Tick(m_ctx);

            m_ctx.Self.Chasing = m_ctx.Chasing;

            string current = StateName;

            if (current != m_lastStateName)
            {
                m_lastStateName = current;
                TransitionCount++;
            }
        }
    }
}
```
