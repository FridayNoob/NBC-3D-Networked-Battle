## 二、功能概述

SceneFlow.Instance.LoadAsync(GameScene.World)：资源层就绪走 YooAsset，否则用 SceneManager 兜底；带回进度、错误、防重入。

## 三、接口与入参出参

| 成员 | 入参 | 出参 |
| --- | --- | --- |
| GameSceneAddress.AddressOf(scene) | 场景枚举 | 地址（文件名）；非法值抛异常 |
| SceneFlow.LoadAsync(scene, onDone) | 场景枚举、回调 | 回调 (bool, string) |
| SceneFlow.IsLoading / Progress / LastError | — | 状态 |

## 四、逐段讲解

1. **地址表**：五个场景 → Login/World/PVP_Room/Battle/Test；查不到直接抛，防止静默黑屏。
2. **单例**：第一次访问创建 [SceneFlow] 并 DontDestroyOnLoad，切场景时它跟着活。
3. **加载**：IsLoading 挡重复点击 → 协程取地址 → 资源层可用走 AssetManager.LoadSceneAsync，轮询进度，完成后 Dispose；不可用走 SceneManager。
4. **调试键**：F1~F5 调 LoadAsync，已挂在 Login 与 World 两个场景里。

### Game/GameFlow/GameSceneAddress.cs

`csharp
// 这个文件里有：
//   GameScene（枚举）：正式流程里的场景编号。
//   GameSceneAddress（静态类）：枚举 → YooAsset 地址 / Unity 场景名。
//
// 地址规则必须与 YooAsset 收集器一致：collector = AddressByFileName、SupportExtensionless = 1
// ⇒ 地址就是"文件名去掉 .unity"，例如 Assets/_Project/Scenes/World.unity 的地址是 World。

using System;
using System.Collections.Generic;

namespace NBC.Game.GameFlow
{
    // 正式流程里的场景（Test 只给开发用）
    public enum GameScene
    {
        // 登录 / 注册
        Login = 0,

        // 主世界（登录后默认进这里）
        World = 1,

        // PVP 决斗场景
        PvpRoom = 2,

        // 多人联机副本
        Battle = 3,

        // 开发测试场景
        Test = 4,
    }

    // 主要功能：把 GameScene 换成"YooAsset 地址"与"Unity 场景名"。
    // 主要数据成员：Addresses（枚举 → 文件名）。
    // 主要外部方法：AddressOf / NameOf。
    // 谁在用它：SceneFlow（加载）、SceneFlowDebugKeys（调试快捷键）。
    // 注意：改名场景必须同步改这里，否则加载会报"地址不存在"。
    public static class GameSceneAddress
    {
        // 枚举 → 场景文件名（不含扩展名），也是 YooAsset 地址
        private static readonly Dictionary<GameScene, string> Addresses = new Dictionary<GameScene, string>
        {
            { GameScene.Login, "Login" },
            { GameScene.World, "World" },
            { GameScene.PvpRoom, "PVP_Room" },
            { GameScene.Battle, "Battle" },
            { GameScene.Test, "Test" },
        };

        // 功能：取 YooAsset 地址。
        // 入参：scene 场景编号
        // 返回：地址字符串
        // 步骤：查表；查不到直接抛（配置错要立刻暴露，不能静默加载错场景）
        // 上游：SceneFlow
        // 下游：无
        public static string AddressOf(GameScene scene)
        {
            if (Addresses.TryGetValue(scene, out string address))
            {
                return address;
            }

            throw new ArgumentOutOfRangeException(nameof(scene), scene, "[GameSceneAddress] 没有登记这个场景");
        }

        // 功能：取 Unity 场景名（SceneManager 兜底用）。
        // 入参：scene 场景编号
        // 返回：场景名（与地址相同，因为地址就是文件名）
        // 步骤：复用 AddressOf
        // 上游：SceneFlow
        // 下游：SceneManager.LoadSceneAsync
        public static string NameOf(GameScene scene)
        {
            return AddressOf(scene);
        }
    }
}
`

### Game/GameFlow/SceneFlow.cs

`csharp
// 这个文件里有：
//   SceneFlow（MonoBehaviour 单例）：唯一的场景加载入口 —— 走 YooAsset 资源层，失败退回 SceneManager。
//   SceneFlowDebugKeys（MonoBehaviour）：开发用快捷键（F1~F5）直接切场景。
//
// 为什么要单例：加载过程是异步的，需要有东西开着协程；业务只调 SceneFlow.Instance.LoadAsync(...)。

using System;
using System.Collections;
using NBC.Framework.Asset;
using UnityEngine;
using UnityEngine.SceneManagement;

namespace NBC.Game.GameFlow
{
    // 主要功能：切场景（优先 YooAsset，资源层没就绪时退回 SceneManager），并防止重复点。
    // 主要数据成员：IsLoading（是否正在加载）、Progress（0..1）。
    // 主要外部方法：LoadAsync（加载并回调结果）、Instance（拿单例）。
    // 谁在用它：Login 的按钮、调试快捷键、以后的传送门/副本入口。
    // 注意：YooAsset 地址错了会抛异常，这里接住并回报错误，不让它变成静默黑屏。
    public sealed class SceneFlow : MonoBehaviour
    {
        // 单例（第一次访问时创建，切场景不销毁）
        private static SceneFlow s_instance;

        // 是否正在加载
        public bool IsLoading { get; private set; }

        // 最近一次加载进度（0..1）
        public float Progress { get; private set; }

        // 最近一次错误（空串表示没出错）
        public string LastError { get; private set; } = string.Empty;

        // 功能：拿单例（没有就建一个常驻对象）。
        // 入参：无
        // 返回：单例
        // 步骤：先找场景里的 → 没有就新建并 DontDestroyOnLoad
        // 上游：业务调用
        // 下游：Awake
        public static SceneFlow Instance
        {
            get
            {
                if (s_instance != null)
                {
                    return s_instance;
                }

                SceneFlow existing = FindObjectOfType<SceneFlow>();

                if (existing != null)
                {
                    s_instance = existing;
                    return s_instance;
                }

                GameObject go = new GameObject("[SceneFlow]");
                s_instance = go.AddComponent<SceneFlow>();
                return s_instance;
            }
        }

        // 功能：避免重复单例。
        // 入参：无
        // 返回：无
        // 步骤：已有别的实例就销毁自己
        // 上游：Unity
        // 下游：无
        private void Awake()
        {
            if (s_instance != null && s_instance != this)
            {
                Destroy(gameObject);
                return;
            }

            s_instance = this;
            DontDestroyOnLoad(gameObject);
        }

        // 功能：异步加载一个场景。
        // 入参：scene 场景编号；onDone 完成回调（参数：成功？、错误信息）
        // 返回：无（协程跑，回调通知）
        // 步骤：
        //   1) 正在加载就直接告诉调用方"忙"
        //   2) 资源层就绪 ⇒ 走 AssetManager.LoadSceneAsync（YooAsset）
        //   3) 没就绪 ⇒ 退回 SceneManager.LoadSceneAsync（编辑器直接运行也安全）
        // 上游：Login 按钮 / SceneFlowDebugKeys
        // 下游：AssetManager.LoadSceneAsync、SceneManager.LoadSceneAsync
        public void LoadAsync(GameScene scene, Action<bool, string> onDone = null)
        {
            if (IsLoading)
            {
                onDone?.Invoke(false, "正在加载中，请稍候");
                return;
            }

            StartCoroutine(LoadRoutine(scene, onDone));
        }

        // 功能：加载协程（YooAsset 优先）。
        // 入参：scene 场景编号；onDone 回调
        // 返回：迭代器
        // 步骤：
        //   1) 资源层可用 ⇒ 拆场景句柄，轮询进度，完成后释放句柄
        //   2) 资源层不可用 ⇒ 用 SceneManager 加载
        //   3) 任何异常都记进 LastError 并回报
        // 上游：LoadAsync
        // 下游：GameSceneAddress、SceneManager
        private IEnumerator LoadRoutine(GameScene scene, Action<bool, string> onDone)
        {
            IsLoading = true;
            Progress = 0f;
            LastError = string.Empty;

            string address = GameSceneAddress.AddressOf(scene);
            bool ok = false;
            string error = string.Empty;

            if (AssetManager.Instance != null && AssetManager.Instance.IsInitialized)
            {
                ISceneHandle handle = null;

                try
                {
                    handle = AssetManager.Instance.LoadSceneAsync(address, LoadSceneMode.Single);
                }
                catch (Exception ex)
                {
                    error = "YooAsset 加载失败：" + ex.Message;
                }

                if (handle != null)
                {
                    while (!handle.IsDone)
                    {
                        Progress = handle.Progress;
                        yield return null;
                    }

                    Progress = 1f;
                    ok = string.IsNullOrEmpty(handle.Error);
                    error = handle.Error ?? string.Empty;
                    handle.Dispose();
                }
            }
            else
            {
                // 资源层没初始化（例如直接打开 World 场景调试）：用 Unity 自带加载，保证流程能走
                AsyncOperation operation = SceneManager.LoadSceneAsync(GameSceneAddress.NameOf(scene), LoadSceneMode.Single);

                if (operation == null)
                {
                    error = "SceneManager 加载失败（场景是否在 Build Settings 里？）";
                }
                else
                {
                    while (!operation.isDone)
                    {
                        Progress = operation.progress;
                        yield return null;
                    }

                    Progress = 1f;
                    ok = true;
                }
            }

            LastError = error;
            IsLoading = false;

            if (!ok)
            {
                Debug.LogError("[SceneFlow] 加载 " + address + " 失败：" + error);
            }

            onDone?.Invoke(ok, error);
        }
    }

    // 主要功能：开发期快捷键切场景（F1 Login / F2 World / F3 PVP / F4 Battle / F5 Test）。
    // 主要数据成员：无。
    // 主要外部方法：Unity 的 Update。
    // 谁在用它：挂在任意场景里手动测试；正式版本移除。
    // 注意：只做"按键 → SceneFlow.LoadAsync"，不含任何业务判断。
    public sealed class SceneFlowDebugKeys : MonoBehaviour
    {
        // 是否启用快捷键（关掉就不会误触）
        [SerializeField]
        private bool m_enabled = true;

        // 功能：每帧查快捷键。
        // 入参：无
        // 返回：无
        // 步骤：F1~F5 分别加载五个场景
        // 上游：Unity
        // 下游：SceneFlow.LoadAsync
        private void Update()
        {
            if (!m_enabled)
            {
                return;
            }

            if (Input.GetKeyDown(KeyCode.F1)) { SceneFlow.Instance.LoadAsync(GameScene.Login); }
            if (Input.GetKeyDown(KeyCode.F2)) { SceneFlow.Instance.LoadAsync(GameScene.World); }
            if (Input.GetKeyDown(KeyCode.F3)) { SceneFlow.Instance.LoadAsync(GameScene.PvpRoom); }
            if (Input.GetKeyDown(KeyCode.F4)) { SceneFlow.Instance.LoadAsync(GameScene.Battle); }
            if (Input.GetKeyDown(KeyCode.F5)) { SceneFlow.Instance.LoadAsync(GameScene.Test); }
        }
    }
}
`

// 这个文件里有：
//   GameSceneAddressTests（NUnit 测试）：场景枚举 → 地址 / 场景名的映射必须和实际文件一致。
//
// 它挡的是"改了场景文件名忘了改映射"——那种错在运行期表现为黑屏，很难查。

using System;
using NBC.Game.GameFlow;
using NUnit.Framework;

namespace NBC.Tests.EditMode
{
    // 主要功能：校验五个场景的地址映射与异常分支。
    // 主要数据成员：无。
    // 主要外部方法：Addresses / Test / Unknown。
    // 谁在用它：Unity Test Runner（EditMode）。
    public sealed class GameSceneAddressTests
    {
        // 功能：五个场景的地址必须等于文件名。
        // 入参：场景枚举、期望文件名
        // 返回：无
        // 步骤：逐个断言
        // 上游：Unity Test Runner
        // 下游：GameSceneAddress.AddressOf
        [TestCase(GameScene.Login, "Login")]
        [TestCase(GameScene.World, "World")]
        [TestCase(GameScene.PvpRoom, "PVP_Room")]
        [TestCase(GameScene.Battle, "Battle")]
        [TestCase(GameScene.Test, "Test")]
        public void Addresses(GameScene scene, string expected)
        {
            Assert.AreEqual(expected, GameSceneAddress.AddressOf(scene));
            Assert.AreEqual(expected, GameSceneAddress.NameOf(scene));
        }

        // 功能：未登记的枚举值必须抛异常（不能静默加载错场景）。
        // 入参：无
        // 返回：无
        // 步骤：传一个越界值，断言抛 ArgumentOutOfRangeException
        // 上游：Unity Test Runner
        // 下游：GameSceneAddress.AddressOf
        [Test]
        public void Unknown()
        {
            Assert.Throws<ArgumentOutOfRangeException>(() => GameSceneAddress.AddressOf((GameScene)99));
        }
    }
}
