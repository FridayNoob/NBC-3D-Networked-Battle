## 二、功能概述

编辑期：选中狼预制体，Scene 视图出现红圈（攻击 2000）、白圈（检测 4000）、蓝竖线（接地 300）。
运行期：LineRenderer 在 Game 视图画同样三条线，跟着狼走。
Inspector：显示「表值 / 显示值」，覆盖时红字提示；点按钮写回 `Monster.csv` 三列。

## 三、接口与入参出参

| 成员 | 入参 | 出参 |
| --- | --- | --- |
| `MonsterAiDebugView.Bind(int)` | Monster 主键 | 无（记下怪型） |
| `DisplayAttackRangeMm` / `DisplayAggroRangeMm` / `DisplayGroundLineMm` | — | 毫米（覆盖优先，否则表值） |
| `IsOverriding` | — | bool |
| `ClearOverrides()` | — | 无 |
| `MonsterAiCsvWriter.TryWrite(id, attack, aggro, ground, out msg)` | 4 个值 | bool + 提示文本 |

## 四、逐段讲解

1. **组件**：三个锚点（`Attack`/`Aggro`/`Ground`，缺失自动建）→ `OnDrawGizmos` 画三色 → `LateUpdate` 用 LineRenderer 画同样三条（顶点数组复用，不每帧 new）。
2. **显示值**：`覆盖 > 0 ? 覆盖 : MonsterConfig.Get(id).xxx`，保证默认与服务端同源。
3. **Inspector**：三行对比 + 覆盖红字 + 写回/清空两个按钮；写回成功后自动清空覆盖。
4. **CSV 写回**：引号感知切分（保住 `skillIds` 里的逗号），只替换三列；一路往上找仓库根的 CSV（Unity 工程在 `Client/` 子目录，踩过路径坑）。
5. **绑定**：`SnapshotSceneBinder` 在 `EnsureSlot` 之后（有 `e` 的作用域）调 `Bind(e.ConfigId)`。

### `Game/View/MonsterAiDebugView.cs`

```csharp
// 这个文件里有：
//   MonsterAiDebugView（组件）：把怪的三条判定画出来（攻击红 / 玩家检测白 / 接地蓝），并允许在 Inspector 覆盖长度。
//
// 真值来自 MonsterConfig（与服务端同源的生成配置）；覆盖值只在"还没写回"时用于预览，
// 写回由 MonsterAiDebugViewEditor 负责把三列写进 Configs/Design/Monster.csv。

using NBC.Game.Config;
using UnityEngine;

namespace NBC.Game.View
{
    // 主要功能：编辑期用 Gizmos、运行期用 LineRenderer，把攻击/检测/接地三条判定画在怪身上。
    // 主要数据成员：m_config（表）、m_monsterConfigId（写回哪一行）、三个覆盖值、三个子 Transform。
    // 主要外部方法：Bind（运行时由 Binder 调用）、Display* 只读属性、IsOverriding。
    // 谁在用它：怪物预制体（Forest Wolf）上的组件；SnapshotSceneBinder 负责 Bind。
    // 注意：覆盖值只影响"显示"，服务端判定永远读表；所以覆盖未写回时 Inspector 必须提示不一致。
    public sealed class MonsterAiDebugView : MonoBehaviour
    {
        // 三个子 Transform 的固定名字（缺失时自动创建）
        public const string AttackAnchorName = "Attack";
        public const string AggroAnchorName = "Aggro";
        public const string GroundAnchorName = "Ground";

        // 毫米 → 米
        private const float MmToUnit = 0.001f;

        // 圆的采样点数（越大越圆，48 足够）
        private const int CircleSegments = 48;

        // 线宽（米）
        private const float LineWidth = 0.02f;

        // 攻击判定颜色（红）
        public static readonly Color AttackColor = new Color(1f, 0.2f, 0.2f, 0.9f);

        // 玩家检测颜色（白）
        public static readonly Color AggroColor = new Color(1f, 1f, 1f, 0.8f);

        // 接地颜色（蓝）
        public static readonly Color GroundColor = new Color(0.25f, 0.55f, 1f, 0.9f);

        // 怪物配置表（Client/Assets/_Project/Configs/MonsterConfig.asset）
        [SerializeField]
        private MonsterConfig m_config;

        // 写回时用哪一行（Monster 表主键，6001 = 野狼）
        [SerializeField]
        private int m_monsterConfigId = 6001;

        // 攻击判定半径覆盖（毫米）；0 = 用表值
        [SerializeField]
        private int m_attackRangeMmOverride;

        // 玩家检测半径覆盖（毫米）；0 = 用表值
        [SerializeField]
        private int m_aggroRangeMmOverride;

        // 接地线长度覆盖（毫米）；0 = 用表值
        [SerializeField]
        private int m_groundLineMmOverride;

        // 攻击判定中心
        [SerializeField]
        private Transform m_attackAnchor;

        // 玩家检测中心
        [SerializeField]
        private Transform m_aggroAnchor;

        // 接地点
        [SerializeField]
        private Transform m_groundAnchor;

        // 运行期的三条线（懒创建）
        private LineRenderer m_attackLine;
        private LineRenderer m_aggroLine;
        private LineRenderer m_groundLine;

        // 圆顶点缓存（复用，避免每帧分配）
        private Vector3[] m_circleBuffer;

        // 怪物配置编号（可运行时改）
        public int MonsterConfigId
        {
            get { return m_monsterConfigId; }
            set { m_monsterConfigId = value; }
        }

        // 攻击半径覆盖值（毫米，0 = 用表值）
        public int AttackRangeMmOverride
        {
            get { return m_attackRangeMmOverride; }
            set { m_attackRangeMmOverride = value; }
        }

        // 检测半径覆盖值（毫米，0 = 用表值）
        public int AggroRangeMmOverride
        {
            get { return m_aggroRangeMmOverride; }
            set { m_aggroRangeMmOverride = value; }
        }

        // 接地线覆盖值（毫米，0 = 用表值）
        public int GroundLineMmOverride
        {
            get { return m_groundLineMmOverride; }
            set { m_groundLineMmOverride = value; }
        }

        // 表里的攻击半径（毫米）；读不到返回 0
        public int TableAttackRangeMm
        {
            get
            {
                Config_Monster row = ResolveRow();
                return row == null ? 0 : row.attackRangeMm;
            }
        }

        // 表里的检测半径（毫米）
        public int TableAggroRangeMm
        {
            get
            {
                Config_Monster row = ResolveRow();
                return row == null ? 0 : row.aggroRangeMm;
            }
        }

        // 表里的接地线长（毫米）
        public int TableGroundLineMm
        {
            get
            {
                Config_Monster row = ResolveRow();
                return row == null ? 0 : row.groundLineMm;
            }
        }

        // 显示用攻击半径：覆盖优先，否则表值
        public int DisplayAttackRangeMm
        {
            get { return m_attackRangeMmOverride > 0 ? m_attackRangeMmOverride : TableAttackRangeMm; }
        }

        // 显示用检测半径
        public int DisplayAggroRangeMm
        {
            get { return m_aggroRangeMmOverride > 0 ? m_aggroRangeMmOverride : TableAggroRangeMm; }
        }

        // 显示用接地线长
        public int DisplayGroundLineMm
        {
            get { return m_groundLineMmOverride > 0 ? m_groundLineMmOverride : TableGroundLineMm; }
        }

        // 是否有覆盖值（与表值不同）
        public bool IsOverriding
        {
            get
            {
                return (m_attackRangeMmOverride > 0 && m_attackRangeMmOverride != TableAttackRangeMm)
                       || (m_aggroRangeMmOverride > 0 && m_aggroRangeMmOverride != TableAggroRangeMm)
                       || (m_groundLineMmOverride > 0 && m_groundLineMmOverride != TableGroundLineMm);
            }
        }

        // 功能：运行时由 Binder 告知这是哪个怪型（决定读哪一行表）。
        // 入参：monsterConfigId Monster 表主键
        // 返回：无
        // 步骤：编号合法就记下
        // 上游：SnapshotSceneBinder.EnsureSlot
        // 下游：ResolveRow
        public void Bind(int monsterConfigId)
        {
            if (monsterConfigId > 0)
            {
                m_monsterConfigId = monsterConfigId;
            }
        }

        // 功能：清掉三个覆盖值（写回成功后调用，让显示回到表值）。
        // 入参：无
        // 返回：无
        // 步骤：三个覆盖值归零
        // 上游：MonsterAiDebugViewEditor
        // 下游：Display* 属性
        public void ClearOverrides()
        {
            m_attackRangeMmOverride = 0;
            m_aggroRangeMmOverride = 0;
            m_groundLineMmOverride = 0;
        }

        // 功能：在编辑期画三色 Gizmos。
        // 入参：无
        // 返回：无
        // 步骤：取三个锚点 → 红圈/白圈/蓝竖线
        // 上游：Unity 编辑器
        // 下游：DrawCircle
        private void OnDrawGizmos()
        {
            EnsureAnchors();

            Gizmos.color = AttackColor;
            DrawCircle(m_attackAnchor, DisplayAttackRangeMm);

            Gizmos.color = AggroColor;
            DrawCircle(m_aggroAnchor, DisplayAggroRangeMm);

            Gizmos.color = GroundColor;

            if (m_groundAnchor != null)
            {
                float length = DisplayGroundLineMm * MmToUnit;
                Gizmos.DrawLine(m_groundAnchor.position, m_groundAnchor.position + Vector3.down * length);
            }
        }

        // 功能：运行期每帧刷新三条 LineRenderer。
        // 入参：无
        // 返回：无
        // 步骤：确保线和缓冲 → 更新两个圆 → 更新接地竖线
        // 上游：Unity 每帧
        // 下游：LineRenderer.SetPositions
        private void LateUpdate()
        {
            if (!Application.isPlaying)
            {
                return;
            }

            EnsureAnchors();
            EnsureRenderers();

            UpdateCircle(m_attackLine, m_attackAnchor, DisplayAttackRangeMm);
            UpdateCircle(m_aggroLine, m_aggroAnchor, DisplayAggroRangeMm);

            if (m_groundLine != null && m_groundAnchor != null)
            {
                float length = DisplayGroundLineMm * MmToUnit;
                m_groundLine.positionCount = 2;
                m_groundLine.SetPosition(0, m_groundAnchor.position);
                m_groundLine.SetPosition(1, m_groundAnchor.position + Vector3.down * length);
            }
        }

        // 功能：读表里的这一行（读不到返回 null）。
        // 入参：无
        // 返回：配置行；无表/无主键返回 null
        // 步骤：m_config.TryGet
        // 上游：Display* 属性
        // 下游：无
        private Config_Monster ResolveRow()
        {
            if (m_config == null)
            {
                return null;
            }

            Config_Monster row;
            return m_config.TryGet(m_monsterConfigId, out row) ? row : null;
        }

        // 功能：确保三个锚点存在（缺失就自动创建）。
        // 入参：无
        // 返回：无
        // 步骤：逐个检查，缺失则建空物体并挂到自身下
        // 上游：OnDrawGizmos / LateUpdate
        // 下游：CreateAnchor
        private void EnsureAnchors()
        {
            if (m_attackAnchor == null)
            {
                m_attackAnchor = CreateAnchor(AttackAnchorName, new Vector3(0f, 0f, 0.5f));
            }

            if (m_aggroAnchor == null)
            {
                m_aggroAnchor = CreateAnchor(AggroAnchorName, Vector3.zero);
            }

            if (m_groundAnchor == null)
            {
                m_groundAnchor = CreateAnchor(GroundAnchorName, Vector3.zero);
            }
        }

        // 功能：创建一个锚点（运行期也安全）。
        // 入参：name 名字；localPosition 本地位置
        // 返回：新 Transform
        // 步骤：查同名子物体 → 没有就建
        // 上游：EnsureAnchors
        // 下游：无
        private Transform CreateAnchor(string name, Vector3 localPosition)
        {
            Transform existing = transform.Find(name);

            if (existing != null)
            {
                return existing;
            }

            GameObject go = new GameObject(name);
            go.transform.SetParent(transform, false);
            go.transform.localPosition = localPosition;
            go.transform.localRotation = Quaternion.identity;
            return go.transform;
        }

        // 功能：懒创建三条 LineRenderer。
        // 入参：无
        // 返回：无
        // 步骤：各自取或加组件 → 设置材质/宽度/颜色/世界坐标
        // 上游：LateUpdate
        // 下游：ConfigureLine
        private void EnsureRenderers()
        {
            if (m_attackLine == null && m_attackAnchor != null)
            {
                m_attackLine = ConfigureLine(m_attackAnchor, AttackColor);
            }

            if (m_aggroLine == null && m_aggroAnchor != null)
            {
                m_aggroLine = ConfigureLine(m_aggroAnchor, AggroColor);
            }

            if (m_groundLine == null && m_groundAnchor != null)
            {
                m_groundLine = ConfigureLine(m_groundAnchor, GroundColor);
            }
        }

        // 功能：给锚点配一条 LineRenderer。
        // 入参：anchor 锚点；color 颜色
        // 返回：LineRenderer
        // 步骤：取或加组件 → 材质/宽度/世界空间/颜色 → 关闭阴影与光照
        // 上游：EnsureRenderers
        // 下游：无
        private static LineRenderer ConfigureLine(Transform anchor, Color color)
        {
            LineRenderer line = anchor.GetComponent<LineRenderer>();

            if (line == null)
            {
                line = anchor.gameObject.AddComponent<LineRenderer>();
            }

            line.useWorldSpace = true;
            line.loop = false;
            line.widthMultiplier = LineWidth;
            line.numCornerVertices = 0;
            line.numCapVertices = 0;
            line.shadowCastingMode = UnityEngine.Rendering.ShadowCastingMode.Off;
            line.receiveShadows = false;
            line.material = new Material(Shader.Find("Sprites/Default"));
            line.startColor = color;
            line.endColor = color;
            return line;
        }

        // 功能：把一条线更新成圆。
        // 入参：line 线；anchor 圆心；radiusMm 半径（毫米）
        // 返回：无
        // 步骤：缓冲不足就重建 → 逐点算圆 → SetPositions
        // 上游：LateUpdate
        // 下游：无
        private void UpdateCircle(LineRenderer line, Transform anchor, int radiusMm)
        {
            if (line == null || anchor == null)
            {
                return;
            }

            int count = CircleSegments + 1;

            if (m_circleBuffer == null || m_circleBuffer.Length != count)
            {
                m_circleBuffer = new Vector3[count];
            }

            float radius = radiusMm * MmToUnit;
            Vector3 center = anchor.position;

            for (int i = 0; i < count; i++)
            {
                float angle = (360f / CircleSegments) * i * Mathf.Deg2Rad;
                m_circleBuffer[i] = center + new Vector3(Mathf.Cos(angle) * radius, 0f, Mathf.Sin(angle) * radius);
            }

            line.positionCount = count;
            line.SetPositions(m_circleBuffer);
        }

        // 功能：画一个圆（Gizmos 版）。
        // 入参：anchor 圆心；radiusMm 半径（毫米）
        // 返回：无
        // 步骤：逐段 DrawLine 连成圆
        // 上游：OnDrawGizmos
        // 下游：无
        private static void DrawCircle(Transform anchor, int radiusMm)
        {
            if (anchor == null || radiusMm <= 0)
            {
                return;
            }

            float radius = radiusMm * MmToUnit;
            Vector3 center = anchor.position;
            Vector3 previous = center + new Vector3(radius, 0f, 0f);

            for (int i = 1; i <= CircleSegments; i++)
            {
                float angle = (360f / CircleSegments) * i * Mathf.Deg2Rad;
                Vector3 next = center + new Vector3(Mathf.Cos(angle) * radius, 0f, Mathf.Sin(angle) * radius);
                Gizmos.DrawLine(previous, next);
                previous = next;
            }
        }
    }
}
```

### `Editor/MonsterAiDebugViewEditor.cs`

```csharp
// 这个文件里有：
//   MonsterAiDebugViewEditor（编辑器类）：MonsterAiDebugView 的自定义 Inspector —— 对比表值与显示值、一键写回 CSV。
//
// 【仅编辑器】写回只改 CSV，不自动跑重生成（遵守 AGENTS.md 第 3 节：禁止一键脚本）。

using NBC.Game.View;
using UnityEditor;
using UnityEngine;

namespace NBC.EditorTools
{
    // 主要功能：把"表值 / 显示值 / 是否一致"摊开给负责人看，并提供「写回 Monster.csv」。
    // 主要数据成员：无（只用 target）。
    // 主要外部方法：OnInspectorGUI。
    // 谁在用它：Inspector 里选中带 MonsterAiDebugView 的怪物预制体时。
    // 注意：覆盖值未写回时必须在顶部显示红字，否则又会变成"显示 3 米、实际 2 米"。
    [CustomEditor(typeof(MonsterAiDebugView))]
    public sealed class MonsterAiDebugViewEditor : UnityEditor.Editor
    {
        // 功能：画自定义 Inspector。
        // 入参：无
        // 返回：无
        // 步骤：
        //   1) 画默认字段
        //   2) 三行对比（攻击 / 检测 / 接地）
        //   3) 有覆盖就红字提示
        //   4) 两个按钮：写回、清空覆盖
        // 上游：Unity Inspector
        // 下游：MonsterAiCsvWriter.TryWrite
        public override void OnInspectorGUI()
        {
            MonsterAiDebugView view = (MonsterAiDebugView)target;

            DrawDefaultInspector();
            EditorGUILayout.Space();

            EditorGUILayout.LabelField("判定长度对比（毫米）", EditorStyles.boldLabel);
            DrawCompareRow("攻击（红）", view.TableAttackRangeMm, view.DisplayAttackRangeMm);
            DrawCompareRow("玩家检测（白）", view.TableAggroRangeMm, view.DisplayAggroRangeMm);
            DrawCompareRow("接地线（蓝）", view.TableGroundLineMm, view.DisplayGroundLineMm);
            EditorGUILayout.Space();

            if (view.IsOverriding)
            {
                EditorGUILayout.HelpBox(
                    "显示值 ≠ 服务端值：现在画的圈只影响显示，服务端判定仍用表里的值。\n点「写回 Monster.csv」才会把它变成真值。",
                    MessageType.Warning);
            }
            else
            {
                EditorGUILayout.HelpBox("显示值与服务端一致（都来自 Monster.csv）。", MessageType.Info);
            }

            if (GUILayout.Button("写回 Monster.csv（攻击 / 检测 / 接地）", GUILayout.Height(28f)))
            {
                bool ok = MonsterAiCsvWriter.TryWrite(
                    view.MonsterConfigId,
                    view.DisplayAttackRangeMm,
                    view.DisplayAggroRangeMm,
                    view.DisplayGroundLineMm,
                    out string message);

                if (ok)
                {
                    view.ClearOverrides();
                    EditorUtility.SetDirty(view);
                    EditorUtility.DisplayDialog("已写回", message, "知道了");
                }
                else
                {
                    EditorUtility.DisplayDialog("写回失败", message, "知道了");
                }
            }

            if (GUILayout.Button("清空覆盖（回到表值）"))
            {
                view.ClearOverrides();
                EditorUtility.SetDirty(view);
            }
        }

        // 功能：画一行「名称 + 表值 + 显示值」。
        // 入参：label 名称；tableValue 表值；displayValue 显示值
        // 返回：无
        // 步骤：水平排三个字段
        // 上游：OnInspectorGUI
        // 下游：无
        private static void DrawCompareRow(string label, int tableValue, int displayValue)
        {
            EditorGUILayout.BeginHorizontal();
            EditorGUILayout.LabelField(label, GUILayout.Width(110f));
            EditorGUILayout.LabelField("表 " + tableValue, GUILayout.Width(90f));
            EditorGUILayout.LabelField("显示 " + displayValue, GUILayout.Width(90f));

            if (tableValue != displayValue)
            {
                GUILayout.Label("≠", EditorStyles.boldLabel, GUILayout.Width(20f));
            }

            EditorGUILayout.EndHorizontal();
        }
    }
}
```

### `Editor/MonsterAiCsvWriter.cs`

```csharp
// 这个文件里有：
//   MonsterAiCsvWriter（静态类）：把某只怪的三个判定长度写回 Configs/Design/Monster.csv。
//
// 【仅编辑器】只改三列数值，其它列（含带逗号的 skillIds）原样保留；写完提示重生成命令，不自动调用 CLI。

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;
using UnityEngine;

namespace NBC.EditorTools
{
    // 主要功能：按 Monster 主键定位 CSV 行，改 aggroRangeMm / attackRangeMm / groundLineMm 三列。
    // 主要数据成员：CsvRelativePath、RegenerateCommand。
    // 主要外部方法：TryWrite。
    // 谁在用它：MonsterAiDebugViewEditor 的「写回」按钮。
    // 注意：解析按引号感知切分（skillIds 里有逗号），未触碰的列连引号一起原样写回。
    public static class MonsterAiCsvWriter
    {
        // CSV 相对项目根的路径（真值源）
        public const string CsvRelativePath = "Configs/Design/Monster.csv";

        // 写回后要执行的重生成命令（只提示，不代跑）
        public const string RegenerateCommand =
            "Tools\\ConfigKit\\src\\ConfigKit.Cli\\bin\\Debug\\net8.0\\NBC.ConfigKit.Cli.exe " +
            "--source Configs\\Design --out Client\\Assets\\_Project\\Game\\Config\\Generated";

        // 列名常量（与 CSV 表头一致）
        private const string ColumnAggro = "aggroRangeMm";
        private const string ColumnAttack = "attackRangeMm";
        private const string ColumnGround = "groundLineMm";

        // 功能：写回三列。
        // 入参：monsterId 主键；attackMm 攻击半径；aggroMm 检测半径；groundMm 接地线长
        // 返回：true = 写入成功；false = 失败且 message 说明原因
        // 步骤：
        //   1) 校验三个值都 > 0
        //   2) 读 CSV（保留原换行风格）
        //   3) 解析表头找列号；找数据行
        //   4) 引号感知切分 → 改三格 → 拼回
        //   5) 写文件（UTF-8 无 BOM）并返回"下一步"
        // 上游：MonsterAiDebugViewEditor
        // 下游：File.WriteAllText
        public static bool TryWrite(int monsterId, int attackMm, int aggroMm, int groundMm, out string message)
        {
            if (monsterId <= 0)
            {
                message = "怪物编号非法：" + monsterId;
                return false;
            }

            if (attackMm <= 0 || aggroMm <= 0 || groundMm <= 0)
            {
                message = "三个长度都必须大于 0（当前：" + attackMm + " / " + aggroMm + " / " + groundMm + "）";
                return false;
            }

            string path = ResolveCsvPath();

            if (!File.Exists(path))
            {
                message = "找不到配置表：" + path;
                return false;
            }

            string text = File.ReadAllText(path);
            string newline = text.Contains("\r\n") ? "\r\n" : "\n";
            string[] lines = text.Replace("\r\n", "\n").Split('\n');

            if (lines.Length < 5)
            {
                message = "配置表格式不对（至少要 4 行表头 + 1 行数据）";
                return false;
            }

            List<string> header = SplitFields(lines[0]);
            int aggroIndex = header.IndexOf(ColumnAggro);
            int attackIndex = header.IndexOf(ColumnAttack);
            int groundIndex = header.IndexOf(ColumnGround);

            if (aggroIndex < 0 || attackIndex < 0 || groundIndex < 0)
            {
                message = "表头缺少列：需要 " + ColumnAggro + " / " + ColumnAttack + " / " + ColumnGround;
                return false;
            }

            int rowIndex = -1;

            for (int i = 4; i < lines.Length; i++)
            {
                if (string.IsNullOrWhiteSpace(lines[i]))
                {
                    continue;
                }

                List<string> fields = SplitFields(lines[i]);

                if (fields.Count > 0 && fields[0].Trim() == monsterId.ToString(CultureInfo.InvariantCulture))
                {
                    rowIndex = i;
                    break;
                }
            }

            if (rowIndex < 0)
            {
                message = "CSV 里没有编号为 " + monsterId + " 的行";
                return false;
            }

            List<string> cells = SplitFields(lines[rowIndex]);

            if (cells.Count <= groundIndex)
            {
                message = "该行列数不足（" + cells.Count + " 列），无法写回";
                return false;
            }

            cells[aggroIndex] = aggroMm.ToString(CultureInfo.InvariantCulture);
            cells[attackIndex] = attackMm.ToString(CultureInfo.InvariantCulture);
            cells[groundIndex] = groundMm.ToString(CultureInfo.InvariantCulture);
            lines[rowIndex] = string.Join(",", cells);

            File.WriteAllText(path, string.Join(newline, lines), new UTF8Encoding(false));

            message = "已写回 Monster.csv（编号 " + monsterId + "：" + aggroMm + " / " + attackMm + " / " + groundMm + "）"
                      + "\n\n下一步（要你手动执行）：\n1) 重生成配置：\n" + RegenerateCommand
                      + "\n2) 重启服务端（否则判定半径仍是旧值）";
            return true;
        }

        // 功能：拼出 CSV 的绝对路径。
        // 入参：无
        // 返回：绝对路径
        // 步骤：Application.dataPath 的上一级（项目根）+ 相对路径
        // 上游：TryWrite
        // 下游：无
        private static string ResolveCsvPath()
        {
            // CSV 在**仓库根**（Unity 工程在 Client/ 子目录），所以要一层层往上找，
            // 不能只取 Application.dataPath 的上一级（那会得到 Client/Configs/...）
            DirectoryInfo dir = new DirectoryInfo(Application.dataPath);
            string relative = CsvRelativePath.Replace('/', Path.DirectorySeparatorChar);

            for (int i = 0; i < 5 && dir != null; i++)
            {
                string candidate = Path.Combine(dir.FullName, relative);

                if (File.Exists(candidate))
                {
                    return candidate;
                }

                dir = dir.Parent;
            }

            // 兜底：按 Unity 工程上一级拼（正常情况下上面的循环已经命中）
            return Path.Combine(Directory.GetParent(Application.dataPath).FullName, relative);
        }

        // 功能：按逗号切分一行，但**双引号里的逗号不算分隔符**；切出的字段保留原有引号。
        // 入参：line 一行文本
        // 返回：字段列表
        // 步骤：逐字符扫描 → 引号内不切 → 引号字符原样保留
        // 上游：TryWrite
        // 下游：无
        private static List<string> SplitFields(string line)
        {
            List<string> fields = new List<string>();
            StringBuilder buffer = new StringBuilder();
            bool inQuotes = false;

            for (int i = 0; i < line.Length; i++)
            {
                char c = line[i];

                if (c == '"')
                {
                    inQuotes = !inQuotes;
                    buffer.Append(c);
                    continue;
                }

                if (c == ',' && !inQuotes)
                {
                    fields.Add(buffer.ToString());
                    buffer.Clear();
                    continue;
                }

                buffer.Append(c);
            }

            fields.Add(buffer.ToString());
            return fields;
        }
    }
}
```
