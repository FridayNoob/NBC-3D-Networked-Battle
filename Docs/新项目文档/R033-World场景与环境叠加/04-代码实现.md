## 二、功能概述

World 起场景 → 异步叠加 SampleScene → 关掉它的相机/耳朵/平行光 → 打印关闭数量；相机可用右键/滚轮/WASD 观察。

## 四、逐段讲解

1. **加载**：SceneManager.LoadSceneAsync(name, Additive)，	ry/catch 接住异常转成 LastError；场景名为空直接报错。
2. **校验**：GetSceneByName → IsValid && isLoaded 双条件（加载完但无效的情况我见过）。
3. **关冲突**：按根物体遍历 Camera / AudioListener / Light(仅 Directional)；**只关不删**。
4. **相机**：球坐标（yaw/pitch/distance）+ WASD 平移焦点；pitch 夹 -20~80 防翻顶；右键转视角避免与将来左键攻击冲突。

### Game/GameFlow/WorldEnvLoader.cs

`csharp
// 这个文件里有：
//   WorldEnvLoader（MonoBehaviour）：把"环境子场景"叠加加载进来，并关掉它自带的相机/灯光/耳朵。
//
// 为什么要有它：环境包（StylizedEnvironment）自带一套相机 + 平行光 + AudioListener，
// 直接叠加会双相机重复渲染、双平行光过曝、两个 AudioListener 让 3D 音效定位错乱。

using System.Collections;
using UnityEngine;
using UnityEngine.SceneManagement;

namespace NBC.Game.GameFlow
{
    // 主要功能：叠加加载环境子场景，并把冲突的相机/灯光/音频监听器关掉。
    // 主要数据成员：m_envSceneName（环境场景名，必须在 Build Settings 里）、IsLoaded、DisabledCount。
    // 主要外部方法：LoadAsync（异步加载并处理冲突）。
    // 谁在用它：World 场景上的一个空物体（[WorldEnvLoader]）。
    // 注意：环境场景是美术包自带的，不要改它；冲突只用代码在运行时关掉。
    public sealed class WorldEnvLoader : MonoBehaviour
    {
        // 要叠加的环境场景名（必须与 Build Settings 里的场景同名）
        [SerializeField]
        private string m_envSceneName = "SampleScene";

        // 是否已经加载完成
        public bool IsLoaded { get; private set; }

        // 这次关掉了几个冲突组件（相机 / 灯光 / 音频监听器）
        public int DisabledCount { get; private set; }

        // 最近一次错误（空串表示没问题）
        public string LastError { get; private set; } = string.Empty;

        // 功能：场景启动后自动叠加环境。
        // 入参：无
        // 返回：无
        // 步骤：起协程加载
        // 上游：Unity（Start）
        // 下游：LoadRoutine
        private void Start()
        {
            StartCoroutine(LoadRoutine());
        }

        // 功能：异步叠加加载环境场景，加载完关掉冲突组件。
        // 入参：无
        // 返回：迭代器
        // 步骤：
        //   1) 场景名空 ⇒ 直接报错退出
        //   2) Additive 加载，轮询进度
        //   3) 成功后遍历该场景根物体，禁用相机/灯光/音频监听器
        // 上游：Start
        // 下游：DisableConflicts
        private IEnumerator LoadRoutine()
        {
            if (string.IsNullOrWhiteSpace(m_envSceneName))
            {
                LastError = "环境场景名为空";
                Debug.LogError("[WorldEnvLoader] 环境场景名为空，检查 Inspector");
                yield break;
            }

            AsyncOperation operation = null;

            try
            {
                operation = SceneManager.LoadSceneAsync(m_envSceneName, LoadSceneMode.Additive);
            }
            catch (System.Exception ex)
            {
                LastError = ex.Message;
            }

            if (operation == null)
            {
                if (string.IsNullOrEmpty(LastError))
                {
                    LastError = "环境场景加载失败（是否没加进 Build Settings？）";
                }

                Debug.LogError("[WorldEnvLoader] 加载 " + m_envSceneName + " 失败：" + LastError);
                yield break;
            }

            while (!operation.isDone)
            {
                yield return null;
            }

            Scene envScene = SceneManager.GetSceneByName(m_envSceneName);

            if (!envScene.IsValid() || !envScene.isLoaded)
            {
                LastError = "场景加载完成后仍无效：" + m_envSceneName;
                Debug.LogError("[WorldEnvLoader] " + LastError);
                yield break;
            }

            DisabledCount = DisableConflicts(envScene);
            IsLoaded = true;
            Debug.Log("[WorldEnvLoader] 已叠加环境 " + m_envSceneName + "，关掉冲突组件 " + DisabledCount + " 个");
        }

        // 功能：关掉环境场景里与我们冲突的组件。
        // 入参：envScene 已加载的环境场景
        // 返回：关掉的数量
        // 步骤：遍历根物体（含未激活）→ 找 Camera / Light / AudioListener → 禁用
        // 上游：LoadRoutine
        // 下游：无
        // 注意：只关不删；素材包文件保持原样，换包也不受影响
        private static int DisableConflicts(Scene envScene)
        {
            int count = 0;
            GameObject[] roots = envScene.GetRootGameObjects();

            for (int i = 0; i < roots.Length; i++)
            {
                Camera[] cameras = roots[i].GetComponentsInChildren<Camera>(true);

                for (int c = 0; c < cameras.Length; c++)
                {
                    if (cameras[c].enabled)
                    {
                        cameras[c].enabled = false;
                        count++;
                    }
                }

                AudioListener[] listeners = roots[i].GetComponentsInChildren<AudioListener>(true);

                for (int a = 0; a < listeners.Length; a++)
                {
                    if (listeners[a].enabled)
                    {
                        listeners[a].enabled = false;
                        count++;
                    }
                }

                Light[] lights = roots[i].GetComponentsInChildren<Light>(true);

                for (int l = 0; l < lights.Length; l++)
                {
                    if (lights[l].enabled && lights[l].type == LightType.Directional)
                    {
                        // 只关平行光：点光/聚光（火把、灯）留着当氛围，不会过曝
                        lights[l].enabled = false;
                        count++;
                    }
                }
            }

            return count;
        }
    }
}

### Game/GameFlow/WorldOrbitCamera.cs

`csharp
// 这个文件里有：
//   WorldOrbitCamera（MonoBehaviour）：World 场景的观察相机 —— 右键拖动转视角、滚轮缩放、WASD 平移。
//
// 它只是"看环境"用的工具相机；玩家跟随相机等角色系统落地后，它会退成调试相机。

using UnityEngine;

namespace NBC.Game.GameFlow
{
    // 主要功能：绕着一个焦点转的轨道相机，方便在没有角色的情况下查看场景。
    // 主要数据成员：m_pivot（焦点）、m_distance（距离，米）、m_yaw/m_pitch（角度）、m_panSpeed。
    // 主要外部方法：Unity 的 LateUpdate。
    // 谁在用它：World 场景的 Main Camera。
    // 注意：用右键转视角（左键留给以后的角色操作），避免和将来的点击/攻击冲突。
    public sealed class WorldOrbitCamera : MonoBehaviour
    {
        // 观察焦点（世界的哪个点在屏幕中心）
        [SerializeField]
        private Vector3 m_pivot = new Vector3(0f, 1.5f, 0f);

        // 相机到焦点的距离（米，2~60）
        [SerializeField]
        private float m_distance = 12f;

        // 水平角（度）
        [SerializeField]
        private float m_yaw = 0f;

        // 俯仰角（度，-20~80）
        [SerializeField]
        private float m_pitch = 25f;

        // 右键拖动灵敏度（度/像素）
        [SerializeField]
        private float m_lookSensitivity = 0.25f;

        // 滚轮缩放速度（米/格）
        [SerializeField]
        private float m_zoomSpeed = 2f;

        // WASD 平移速度（米/秒）
        [SerializeField]
        private float m_panSpeed = 8f;

        // 功能：每帧根据输入更新相机位姿。
        // 入参：无
        // 返回：无
        // 步骤：
        //   1) 右键拖动改 yaw/pitch（pitch 夹住，避免翻过顶）
        //   2) 滚轮改距离（夹住 2~60）
        //   3) WASD 平移焦点（按当前朝向的水平方向）
        //   4) 按球坐标算相机位置并看向焦点
        // 上游：Unity
        // 下游：无
        private void LateUpdate()
        {
            if (Input.GetMouseButton(1))
            {
                m_yaw += Input.GetAxis("Mouse X") * m_lookSensitivity * 10f;
                m_pitch -= Input.GetAxis("Mouse Y") * m_lookSensitivity * 10f;
                m_pitch = Mathf.Clamp(m_pitch, -20f, 80f);
            }

            float scroll = Input.GetAxis("Mouse ScrollWheel");

            if (Mathf.Abs(scroll) > 0.001f)
            {
                m_distance = Mathf.Clamp(m_distance - scroll * m_zoomSpeed * 5f, 2f, 60f);
            }

            float horizontal = Input.GetAxis("Horizontal");
            float vertical = Input.GetAxis("Vertical");

            if (Mathf.Abs(horizontal) > 0.001f || Mathf.Abs(vertical) > 0.001f)
            {
                Quaternion flatRotation = Quaternion.Euler(0f, m_yaw, 0f);
                Vector3 move = flatRotation * new Vector3(horizontal, 0f, vertical);
                m_pivot += move * m_panSpeed * Time.deltaTime;
            }

            Quaternion rotation = Quaternion.Euler(m_pitch, m_yaw, 0f);
            Vector3 offset = rotation * new Vector3(0f, 0f, -m_distance);
            transform.position = m_pivot + offset;
            transform.rotation = rotation;
        }
    }
}
`
