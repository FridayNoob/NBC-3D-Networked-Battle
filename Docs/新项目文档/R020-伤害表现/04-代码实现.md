## 二、功能概述

怪物：头顶血条随血量缩短；掉血飘「-N」（按伤害变色变粗）；播 `Take Damage`；死亡播 `Die` 并停留 1 秒再消失。
玩家：被打也飘字 + 播 `HIT L`/`HIT R`；左下角 HUD 显示服务端真实血量（删掉了「空格扣假血」）。

## 三、接口与入参出参

| 成员 | 入参 | 出参 | 说明 |
| --- | --- | --- | --- |
| `DamagePopup.Show(Vector3, int)` | 世界坐标、伤害 | 无 | 伤害 ≤ 0 直接忽略 |
| `MonsterHealthBar.Bind(int maxHp)` | 满血值 | 无 | 建底条 + 填充条 |
| `MonsterHealthBar.SetRatio(float)` | 0..1 | 无 | 改填充宽度并左对齐 |
| `PlayerAnimation.PlayHit()` | — | 无 | `HIT L` / `HIT R` 交替 |
| `RuntimeNetDriver.HeroHp / HeroMaxHp` | — | 血量 | HUD 每帧读 |
| `SnapshotSceneBinder.ApplyCombatFeedback(slot, entity)` | 槽位、快照 | 无 | 血条/飘字/受击/死亡 |

## 四、逐段讲解

1. **血量从快照差值来**：快照每帧带 `Hp`，槽位记 `LastHp`；`Hp` 变小就是伤害。首帧 `HpKnown=false` 只记基线 —— 这是「新怪不飘满血数字」的关键。
2. **死亡要留时间**：`Alive` 变 false 时先 `SetTrigger("Die")` 并记 `DiedAt`，`SetActive` 用 `visible = 存活 || 死后 1 秒内`；否则死亡动画根本看不到。
3. **血条不用 Canvas**：两个 `SpriteRenderer` + 一张运行时生成的 1×1 白 Sprite（全项目共享一张），billboard 朝相机。
4. **飘字分档**：1~19 白/30 号、20~49 橙/38 号、50+ 红/48 号；每条随机 ±0.25 米偏移，避免连击时叠成一坨。
5. **玩家侧放在 driver**：driver 已经持有快照入口与玩家引用，binder 只认识槽位（本地英雄不在槽位里）。
6. **HUD 用轮询**：`HpBarDemoBehaviour.Update` 读 `driver.HeroHp/HeroMaxHp`，简单且没有解绑风险；顺带删掉假的 `m_damagePerPress`。

```csharp
// 这个文件里有：
//   DamagePopup（静态类）：在世界坐标处飘一个伤害数字（自己上浮 + 淡出 + 销毁）。
//   DamagePopupView（组件）：挂在飘字物体上，负责那 1.2 秒的动画。

using TMPro;
using UnityEngine;

namespace NBC.Game.UI
{
    // 主要功能：把「伤害数字」做成一次性的世界空间文本，按伤害大小变色变粗大，1.2 秒后自己销毁。
    // 主要数据成员：无（全部是方法内局部状态）。
    // 主要外部方法：Show（唯一入口）。
    // 谁在用它：SnapshotSceneBinder（怪物掉血）、RuntimeNetDriver（玩家掉血）。
    // 注意：不做对象池 —— 同屏数字个位数，Instantiate + Destroy 足够，且不容易泄漏。
    public static class DamagePopup
    {
        // 飘字存活时间，秒
        private const float LifetimeSeconds = 1.2f;

        // 上浮速度，米/秒
        private const float RiseMetersPerSecond = 1.2f;

        // 伤害分档的阈值（单位：点）；小于第一档是最小的白色数字
        private const int MediumDamage = 20;
        private const int HeavyDamage = 50;

        // 每条飘字的随机水平偏移上限，米；避免连续命中时数字完全叠在一起
        private const float ScatterRadius = 0.25f;

        // 是否已经为「缺默认字体」告过警；只报一次，避免刷屏
        private static bool m_fontWarned;

        // 功能：在世界坐标处飘一个伤害数字（内部显示成 "-amount"）。
        // 入参：worldPosition 世界坐标（一般取受击者头顶）；amount 伤害点数，必须为正
        // 返回：无
        // 步骤：
        //   1) 伤害 <= 0 直接忽略（治疗/无效帧不飘）
        //   2) 没有 TMP 默认字体就只告警一次并放弃（否则会创建出看不见的文本）
        //   3) 建物体 + TextMeshPro，按伤害分档设颜色与字号
        //   4) 挂上 DamagePopupView，由它负责上浮淡出与销毁
        // 上游：SnapshotSceneBinder（怪物受击）、RuntimeNetDriver（玩家受击）
        // 下游：DamagePopupView
        // 注意：位置用世界坐标；飘字是纯表现，不参与任何战斗结算
        public static void Show(Vector3 worldPosition, int amount)
        {
            if (amount <= 0)
            {
                return;
            }

            if (TMP_Settings.defaultFontAsset == null)
            {
                if (!m_fontWarned)
                {
                    m_fontWarned = true;
                    Debug.LogWarning("[伤害飘字] TMP 没有默认字体资产，伤害数字不会显示（请检查 TMP Settings）。");
                }

                return;
            }

            GameObject go = new GameObject("伤害飘字");
            go.transform.position = worldPosition + new Vector3(
                Random.Range(-ScatterRadius, ScatterRadius),
                0f,
                Random.Range(-ScatterRadius, ScatterRadius));

            TextMeshPro text = go.AddComponent<TextMeshPro>();
            text.text = "-" + amount;
            text.alignment = TextAlignmentOptions.Center;
            text.fontSize = ResolveFontSize(amount);
            text.color = ResolveColor(amount);
            text.raycastTarget = false;

            go.AddComponent<DamagePopupView>().Initialize(RiseMetersPerSecond, LifetimeSeconds);
        }

        // 功能：按伤害大小给字号。
        // 入参：amount 伤害点数
        // 返回：字号（点）
        // 步骤：分档判断，越重越大
        // 上游：Show
        // 下游：TextMeshPro.fontSize
        private static float ResolveFontSize(int amount)
        {
            if (amount >= HeavyDamage)
            {
                return 48f;
            }

            return amount >= MediumDamage ? 38f : 30f;
        }

        // 功能：按伤害大小给颜色。
        // 入参：amount 伤害点数
        // 返回：颜色
        // 步骤：小伤害偏白、中伤害橙、重伤害红
        // 上游：Show
        // 下游：TextMeshPro.color
        private static Color ResolveColor(int amount)
        {
            if (amount >= HeavyDamage)
            {
                return new Color(1f, 0.30f, 0.30f);      // 红
            }

            return amount >= MediumDamage
                ? new Color(1f, 0.70f, 0.28f)            // 橙
                : new Color(1f, 0.96f, 0.84f);           // 近白
        }
    }

    // 主要功能：驱动一个飘字物体的上浮与淡出，到期自毁。
    // 主要数据成员：m_riseSpeed（米/秒）、m_lifetime（秒）、m_elapsed（已存在时间）、m_text（自己身上的 TMP）。
    // 主要外部方法：Initialize（由 DamagePopup.Show 调）。
    // 谁在用它：DamagePopup.Show。
    // 注意：不需要 Update 也能活，但用了 Transform + 颜色两件事，放 Update 里最直白。
    public sealed class DamagePopupView : MonoBehaviour
    {
        // 上浮速度，米/秒
        private float m_riseSpeed = 1.2f;

        // 总存活时间，秒
        private float m_lifetime = 1.2f;

        // 已经存活了多久，秒
        private float m_elapsed;

        // 自己身上的文本；淡出要改它的 alpha
        private TextMeshPro m_text;

        // 功能：初始化（由 Show 调，构造时没有参数）。
        // 入参：riseSpeed 上浮速度（米/秒）；lifetime 存活时间（秒）
        // 返回：无
        // 步骤：记参数 + 抓一次 TMP 引用
        // 上游：DamagePopup.Show
        // 下游：Update
        public void Initialize(float riseSpeed, float lifetime)
        {
            m_riseSpeed = riseSpeed;
            m_lifetime = lifetime > 0.01f ? lifetime : 0.01f;
            m_text = GetComponent<TextMeshPro>();
        }

        // 功能：每帧上浮 + 后半程淡出，时间到自毁。
        // 入参：无（引擎每帧调）
        // 返回：无
        // 步骤：
        //   1) 累加时间
        //   2) 上浮（只动 Y）
        //   3) 从一半时间开始线性淡出到 0
        //   4) 超过存活时间就 Destroy 自己
        // 上游：引擎
        // 下游：Destroy
        private void Update()
        {
            m_elapsed += Time.deltaTime;
            transform.position += Vector3.up * (m_riseSpeed * Time.deltaTime);

            if (m_text != null)
            {
                float half = m_lifetime * 0.5f;
                float alpha = m_elapsed <= half ? 1f : Mathf.Clamp01(1f - ((m_elapsed - half) / half));
                Color color = m_text.color;
                color.a = alpha;
                m_text.color = color;
            }

            if (m_elapsed >= m_lifetime)
            {
                Destroy(gameObject);
            }
        }
    }
}
```

```csharp
// 这个文件里有：
//   MonsterHealthBar（组件）：怪物头顶的血条（底 + 填充两个 Sprite，始终朝向相机）。

using UnityEngine;

namespace NBC.Game.UI
{
    // 主要功能：给怪物槽位画一条头顶血条 —— 比例由外部喂，掉血就短，死了随槽位一起隐藏。
    // 主要数据成员：m_fill（填充条 Transform）、m_maxHp（满血值）、m_ratio（当前比例）。
    // 主要外部方法：Bind（建条）、SetRatio（喂比例）。
    // 谁在用它：SnapshotSceneBinder 建槽位时 Bind，每帧 SetRatio。
    // 注意：不用 UI Canvas —— 两个 SpriteRenderer 更便宜、不依赖 UI 程序集，也不需要 EventSystem。
    public sealed class MonsterHealthBar : MonoBehaviour
    {
        // 血条离地面高度，米（按狼约 1 米高定的；其它体型靠父节点缩放也能用）
        private const float HeightMeters = 1.6f;

        // 血条宽度，米
        private const float BarWidth = 0.9f;

        // 血条高度，米
        private const float BarHeight = 0.12f;

        // 填充条（改它的 localScale.x 表示比例）
        private Transform m_fill;

        // 满血值；<=0 时按 1 处理，避免除零
        private int m_maxHp = 1;

        // 当前比例（0..1）
        private float m_ratio = 1f;

        // 全项目共用一张 1×1 白色 Sprite（只为血条生成一次）
        private static Sprite s_whiteSprite;

        // 是否已经为「找不到相机」告过警；只报一次
        private static bool s_cameraWarned;

        // 功能：建出血条（底黑 + 填充红），并设成满血。
        // 入参：maxHp 满血值（<=0 按 1 处理）
        // 返回：无
        // 步骤：
        //   1) 记满血值
        //   2) 建底条（深灰、不透明）
        //   3) 建填充条（红），左对齐缩放
        //   4) 比例设 1
        // 上游：SnapshotSceneBinder.EnsureSlot
        // 下游：SetRatio、Update（朝向相机）
        // 注意：血条挂在槽位根上，**不要**挂到带刚体的子物体上（否则会跟着物理抖）
        public void Bind(int maxHp)
        {
            m_maxHp = maxHp > 0 ? maxHp : 1;

            Sprite sprite = GetWhiteSprite();

            // 底条：居中，整条宽
            GameObject background = new GameObject("血条底");
            background.transform.SetParent(transform, false);
            background.transform.localPosition = new Vector3(0f, HeightMeters, 0f);
            background.transform.localScale = new Vector3(BarWidth, BarHeight, 1f);

            SpriteRenderer back = background.AddComponent<SpriteRenderer>();
            back.sprite = sprite;
            back.color = new Color(0.08f, 0.08f, 0.08f, 0.85f);
            back.sortingOrder = 100;

            // 填充条：原点在左端，靠 scale.x 表示比例
            GameObject fill = new GameObject("血条填充");
            fill.transform.SetParent(transform, false);
            fill.transform.localPosition = new Vector3(0f, HeightMeters, 0f);
            fill.transform.localScale = new Vector3(BarWidth, BarHeight * 0.7f, 1f);

            SpriteRenderer bar = fill.AddComponent<SpriteRenderer>();
            bar.sprite = sprite;
            bar.color = new Color(0.85f, 0.18f, 0.18f, 1f);
            bar.sortingOrder = 101;

            m_fill = fill.transform;
            SetRatio(1f);
        }

        // 功能：设置血条比例。
        // 入参：ratio 0..1（外部保证；这里仍会钳位）
        // 返回：无
        // 步骤：
        //   1) 钳到 0..1，没变化就直接返回（避免每帧写 Transform）
        //   2) 改填充条 scale.x，并把左端对齐：位置左移 (1-ratio) * 宽度 / 2
        // 上游：SnapshotSceneBinder 每帧
        // 下游：Transform.localScale / localPosition
        public void SetRatio(float ratio)
        {
            float clamped = Mathf.Clamp01(ratio);

            if (Mathf.Abs(clamped - m_ratio) < 0.001f || m_fill == null)
            {
                m_ratio = clamped;
                return;
            }

            m_ratio = clamped;

            Vector3 scale = m_fill.localScale;
            m_fill.localScale = new Vector3(BarWidth * clamped, scale.y, scale.z);

            Vector3 position = m_fill.localPosition;
            m_fill.localPosition = new Vector3(-(BarWidth * (1f - clamped)) * 0.5f, position.y, position.z);
        }

        // 功能：让血条始终正对相机（billboard）。
        // 入参：无（引擎每帧调）
        // 返回：无
        // 步骤：没有相机就告警一次并跳过；有就把自己的旋转设成相机的旋转
        // 上游：引擎
        // 下游：Transform.rotation
        private void Update()
        {
            Camera camera = Camera.main;

            if (camera == null)
            {
                if (!s_cameraWarned)
                {
                    s_cameraWarned = true;
                    Debug.LogWarning("[怪物血条] 场景里没有 MainCamera，血条不会朝向镜头（功能不受影响）。");
                }

                return;
            }

            transform.rotation = camera.transform.rotation;
        }

        // 功能：取共用的 1×1 白色 Sprite（第一次调用时生成并缓存）。
        // 入参：无
        // 返回：Sprite（永不为 null）
        // 步骤：
        //   1) 已经生成过就直接返回
        //   2) 建一张 1×1 白纹理，设成 Point 过滤避免模糊
        //   3) 用它创建 Sprite 并缓存
        // 上游：Bind
        // 下游：SpriteRenderer.sprite
        private static Sprite GetWhiteSprite()
        {
            if (s_whiteSprite != null)
            {
                return s_whiteSprite;
            }

            Texture2D texture = new Texture2D(1, 1, TextureFormat.RGBA32, false);
            texture.SetPixel(0, 0, Color.white);
            texture.filterMode = FilterMode.Point;
            texture.Apply();

            s_whiteSprite = Sprite.Create(texture, new Rect(0f, 0f, 1f, 1f), new Vector2(0.5f, 0.5f), 1f);
            return s_whiteSprite;
        }
    }
}
```

```csharp
        // 功能：按快照里的血量与存活状态，驱动血条、伤害飘字、受击与死亡动画。
        // 入参：slot 槽位；entity 这一帧的快照
        // 返回：无
        // 步骤：
        //   1) 第一次见到这个单位：只记基线（血量/存活/满血血条），不飘字
        //   2) 之后掉血 ⇒ 飘一个伤害数字 + 触发受击动画
        //   3) 血条比例跟着血量走
        //   4) 刚死的这一刻 ⇒ 触发死亡动画并记下时刻（隐藏由调用方延迟处理）
        // 上游：LateUpdate 遍历实体时
        // 下游：DamagePopup.Show、Animator.SetTrigger、MonsterHealthBar.SetRatio
        // 注意：首帧必须只记基线 —— 否则单位一进场就会飘一个「满血」数字
        private void ApplyCombatFeedback(Slot slot, EntitySnapshot entity)
        {
            if (slot.HealthBar == null)
            {
                slot.HealthBar = slot.Go.AddComponent<MonsterHealthBar>();
                slot.HealthBar.Bind(entity.MaxHp);
            }

            if (!slot.HpKnown)
            {
                slot.HpKnown = true;
                slot.LastHp = entity.Hp;
                slot.AliveKnown = entity.Alive;
                return;
            }

            if (entity.Hp < slot.LastHp)
            {
                int amount = slot.LastHp - entity.Hp;
                Vector3 popupAt = slot.Transform.position + Vector3.up * 1.4f;
                DamagePopup.Show(popupAt, amount);

                if (slot.HasTakeDamage)
                {
                    slot.Animator.SetTrigger(TakeDamageParameter);
                }
            }

            slot.LastHp = entity.Hp;

            if (entity.MaxHp > 0)
            {
                slot.HealthBar.SetRatio((float)entity.Hp / entity.MaxHp);
            }

            if (slot.AliveKnown && !entity.Alive)
            {
                slot.DiedAt = Time.time;

                if (slot.HasDie)
                {
                    slot.Animator.SetTrigger(DieParameter);
                }
            }

            slot.AliveKnown = entity.Alive;
        }
```

```csharp
        // 功能：播一次受击动画（左/右交替）。
        // 入参：无
        // 返回：无
        // 步骤：
        //   1) 没有动画器 ⇒ 返回
        //   2) 第一次调用时确认两个受击 Trigger 是否存在，缺了只告警一次并放弃
        //   3) 按交替标记 SetTrigger，然后翻转标记
        // 上游：RuntimeNetDriver（玩家掉血时）
        // 下游：Animator.SetTrigger
        // 注意：受击是**表现**，玩家的真实血量以服务端快照为准
        public void PlayHit()
        {
            if (m_animator == null)
            {
                return;
            }

            if (!m_hitTriggerReady)
            {
                if (m_hitTriggerWarned)
                {
                    return;
                }

                m_hitTriggerWarned = true;

                if (!ValidateParameter(HitLeftTrigger, AnimatorControllerParameterType.Trigger)
                    || !ValidateParameter(HitRightTrigger, AnimatorControllerParameterType.Trigger))
                {
                    Debug.LogWarning("[PlayerAnimation] controller 里缺少受击 Trigger（" + HitLeftTrigger + " / " +
                                     HitRightTrigger + "），玩家受击时不会播动作。", this);
                    return;
                }

                m_hitTriggerReady = true;
            }

            m_animator.SetTrigger(m_nextHitIsLeft ? HitLeftTrigger : HitRightTrigger);
            m_nextHitIsLeft = !m_nextHitIsLeft;
        }
```

```csharp
        // 功能：盯住本地英雄的血量 —— 掉血时飘伤害数字 + 播受击，并把最新血量留给 HUD。
        // 入参：hero 本地英雄的快照（可为 null）
        // 返回：无
        // 步骤：
        //   1) 没有数据就返回（还没进房 / 英雄还没生成）
        //   2) 首帧只记基线，避免一进场就飘一个「满血」数字
        //   3) 掉血 ⇒ DamagePopup + PlayHit；随后把最新血量写进 HeroHp/HeroMaxHp（HUD 读）
        // 上游：RefreshPresentation
        // 下游：DamagePopup.Show、PlayerAnimation.PlayHit、HeroHp / HeroMaxHp
        // 注意：只处理「血变少」；治疗不飘字（本版不做治疗数字）
        private void TrackHeroHealth(EntitySnapshot hero)
        {
            if (hero == null)
            {
                return;
            }

            HeroHp = hero.Hp;
            HeroMaxHp = hero.MaxHp;

            if (!m_heroHpKnown)
            {
                m_heroHpKnown = true;
                m_heroLastHp = hero.Hp;
                return;
            }

            if (hero.Hp < m_heroLastHp)
            {
                int amount = m_heroLastHp - hero.Hp;

                if (m_playerAnimation == null)
                {
```
