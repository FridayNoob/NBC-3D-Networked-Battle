## 一、子任务与设计约束

- 新增：`NetRuntimeStats`、`CountingTransport`（Framework.Net，**纯 C#，禁 Unity API**）、`NetHudPanel`（Game.UI）。

- 不改：`ITransport`、两个传输实现、`NetSession`。

## 二、功能概述

`CountingTransport` 在 `Send` / `FrameReceived` 两处累加业务字节 → `NetRuntimeStats` 用 4 个 0.25 秒采样点算 1 秒带宽并保存模式/帧号/RTT →
`NetHudPanel` 每 0.25 秒读一次快照写 4 行文本。Editor 窗口是「建会话方」，负责写模式/帧号/RTT，并保证与 HUD **同源**。

## 三、接口与入参出参

| 单元 | 入参 | 出参 | 异常 |
| --- | --- | --- | --- |
| `NetRuntimeStats.AddSent/AddReceived` | 字节数 | 无 | 负数/0 直接忽略 |
| `NetRuntimeStats.SetMode/SetRtt/SetFrame` | 模式 / ms / 帧号 | 无 | 无 |

| `NetHudPanel.Update` | — | 写 4 行文本 | 无 |

## 四、逐段讲解

1. **为什么用 `Stopwatch` 不用 `Time`**：`Framework.Net` 的 asmdef 是 `noEngineReferences: true`，引用 `UnityEngine` 会导致整个程序集编不过。
2. **为什么「读时惰性推进」**：面板隐藏时没人调 `Update`，若靠外部 Tick 推进滑窗，带宽会永久卡在旧值；惰性推进在 `AddSent/AddReceived/Snapshot/Set*` 里都会走一次，所以不受影响。
3. **为什么超过一整个窗口就清空**：`steps >= 4` 说明中间没人采样（比如面板关了很久），追历史没有意义，直接清 0 更接近真实。
4. **先计数再抛事件**：`OnInnerFrameReceived` 里先 `AddReceived` 再 `handler(payload)`，订阅者抛异常也不会漏计。
5. **`FrameReceived` 不能只转发**：装饰器要自己重新抛事件，否则订阅者收到的是内层的委托链、位置错乱。
6. **模式由「建会话方」写**：今天只有 Editor 建会话；以后运行时启动器接上，同一个 `SetMode` 就自动生效，HUD 一行不用改。
7. **F1 只切 `Box`**：直接 `SetActive(false)` 自己会让 `Update` 停掉 —— 那就再也按不回来了（这是最容易犯的一个错）。

```csharp
// 这个文件里有：
//   NetSyncMode（枚举）、NetStatsSnapshot（结构）、NetRuntimeStats（静态统计）

using System;
using System.Diagnostics;

namespace NBC.Framework.Net
{
    // 当前同步模式。离线 = 没有会话；两个在线模式与 Editor 窗口的「锁步 / 快照」开关一一对应。
    public enum NetSyncMode
    {
        // 没有连接（或已断开）
        Offline = 0,

        // 状态同步：服务端按 tick 广播 WorldSnapshot
        StateSync = 1,

        // 帧同步：服务端按帧广播 LockstepFrame，客户端自己步进
        Lockstep = 2
    }

    // 一次读取拿到的全部统计（避免面板分几次读、数字来自不同时刻）
    public readonly struct NetStatsSnapshot
    {
        public readonly NetSyncMode Mode;
        public readonly long TotalBytesOut;
        public readonly long TotalBytesIn;
        public readonly float BytesPerSecondOut;
        public readonly float BytesPerSecondIn;
        public readonly int RttMs;
        public readonly int Frame;

        public NetStatsSnapshot(NetSyncMode mode, long totalBytesOut, long totalBytesIn,
            float bytesPerSecondOut, float bytesPerSecondIn, int rttMs, int frame)
        {
            Mode = mode;
            TotalBytesOut = totalBytesOut;
            TotalBytesIn = totalBytesIn;
            BytesPerSecondOut = bytesPerSecondOut;
            BytesPerSecondIn = bytesPerSecondIn;
            RttMs = rttMs;
            Frame = frame;
        }
    }

    // 主要功能：累加收发字节、用 4 个采样点算 1 秒带宽、保存模式 / RTT / 帧号，供 HUD 只读。
    // 主要数据成员：4 个采样槽（收发各一组）、当前累计、累计总量、上次采样时刻。
    // 主要外部方法：AddSent / AddReceived（统计点调）、SetMode / SetRtt / SetFrame（建会话方调）、Reset、Snapshot（HUD 调）。
    // 谁在用它：CountingTransport（写字节）、Editor 窗口或运行时启动器（写模式/帧/RTT）、NetHudPanel（读）。
    // 注意：本程序集 noEngineReferences=true ⇒ 只能用 Stopwatch，不能用 Time.unscaledTime。
    public static class NetRuntimeStats
    {
        // 采样点数：4 个 × 0.25 秒 = 1 秒窗口
        private const int SlotCount = 4;

        // 每个采样点的时长（秒）
        private const double SlotSeconds = 0.25;

        // 整个窗口的时长（秒）
        private const float WindowSeconds = SlotCount * (float)SlotSeconds;

        // 收发可能在不同线程被调，统一加锁（临界区很小，不值得上无锁结构）
        private static readonly object Gate = new object();

        // 上行 / 下行的环形采样槽
        private static readonly int[] SlotsOut = new int[SlotCount];
        private static readonly int[] SlotsIn = new int[SlotCount];

        // 下一个要覆盖的槽位
        private static int m_cursor;

        // 当前这一段里累计的字节
        private static int m_currentOut;
        private static int m_currentIn;

        // 累计总量（不随窗口清空）
        private static long m_totalOut;
        private static long m_totalIn;

        // 上次采样的 Stopwatch 时刻（0 = 还没开始）
        private static long m_lastSampleTicks;

        // 会话状态
        private static NetSyncMode m_mode = NetSyncMode.Offline;
        private static int m_rttMs = -1;
        private static int m_frame = -1;

        // 入参：bytes 本次发送的业务字节数
        // 返回：无
        // 步骤：1) 推进采样窗口；2) 当前段与总量都加上
        // 上游：CountingTransport.Send
        // 下游：无
        public static void AddSent(int bytes)
        {
            if (bytes <= 0)
            {
                return;
            }

            lock (Gate)
            {
                Advance();
                m_currentOut += bytes;
                m_totalOut += bytes;
            }
        }

        // 入参：bytes 本次收到的业务字节数
        // 返回：无
        // 步骤：同 AddSent，走下行通道
        // 上游：CountingTransport.FrameReceived
        // 下游：无
        public static void AddReceived(int bytes)
        {
            if (bytes <= 0)
            {
                return;
            }

            lock (Gate)
            {
                Advance();
                m_currentIn += bytes;
                m_totalIn += bytes;
            }
        }

        // 入参：mode 当前模式
        // 返回：无
        // 上游：建会话的一方（Editor 窗口 / 运行时启动器）
        // 下游：Snapshot
        public static void SetMode(NetSyncMode mode)
        {
            lock (Gate)
            {
                Advance();
                m_mode = mode;
            }
        }

        // 入参：rttMs 往返延迟（毫秒，<0 表示未知）
        // 返回：无
        // 上游：建会话的一方
        // 下游：Snapshot
        public static void SetRtt(int rttMs)
        {
            lock (Gate)
            {
                Advance();
                m_rttMs = rttMs;
            }
        }

        // 入参：frame 当前帧号（状态同步 = 快照 tick；帧同步 = 已步进帧；<0 表示未知）
        // 返回：无
        // 上游：建会话的一方
        // 下游：Snapshot
        public static void SetFrame(int frame)
        {
            lock (Gate)
            {
                Advance();
                m_frame = frame;
            }
        }

        // 入参：无
        // 返回：无
        // 步骤：清空槽位、当前段、总量与模式（连断开 / 换局都该调）
        // 上游：断开会话、换局
        // 下游：Snapshot
        public static void Reset()
        {
            lock (Gate)
            {
                Array.Clear(SlotsOut, 0, SlotCount);
                Array.Clear(SlotsIn, 0, SlotCount);
                m_cursor = 0;
                m_currentOut = 0;
                m_currentIn = 0;
                m_totalOut = 0;
                m_totalIn = 0;
                m_lastSampleTicks = 0;
                m_mode = NetSyncMode.Offline;
                m_rttMs = -1;
                m_frame = -1;
            }
        }

        // 入参：无
        // 返回：一次一致的全部统计
        // 步骤：1) 先惰性推进窗口（保证数字是最新的）；2) 4 槽求和 ÷ 1 秒 = B/s；3) 打包返回
        // 上游：NetHudPanel（每 0.25 秒一次）
        // 下游：无
        public static NetStatsSnapshot Snapshot()
        {
            lock (Gate)
            {
                Advance();

                float outBps = Sum(SlotsOut) / WindowSeconds;
                float inBps = Sum(SlotsIn) / WindowSeconds;

                return new NetStatsSnapshot(m_mode, m_totalOut, m_totalIn, outBps, inBps, m_rttMs, m_frame);
            }
        }

        // 入参：slots 采样槽
        // 返回：槽内字节之和
        // 上游：Snapshot
        // 下游：无
        private static int Sum(int[] slots)
        {
            int sum = 0;

            for (int i = 0; i < slots.Length; i++)
            {
                sum += slots[i];
            }

            return sum;
        }

        // 入参：无
        // 返回：无
        // 步骤：
        //   1) 首次调用只记时刻，不开窗
        //   2) 算经过了多少个 0.25 秒
        //   3) 超过一整个窗口 ⇒ 直接清空（不追历史），时刻对齐到现在
        //   4) 否则逐个推进槽位：当前段入槽、当前清零、游标前进
        // 上游：AddSent / AddReceived / 各 Set / Snapshot
        // 下游：SlotsOut、SlotsIn
        // 注意：调用者必须已持有 Gate
        private static void Advance()
        {
            long now = Stopwatch.GetTimestamp();

            if (m_lastSampleTicks == 0L)
            {
                m_lastSampleTicks = now;
                return;
            }

            double elapsed = (now - m_lastSampleTicks) / (double)Stopwatch.Frequency;

            if (elapsed < SlotSeconds)
            {
                return;
            }

            int steps = (int)(elapsed / SlotSeconds);

            if (steps >= SlotCount)
            {
                Array.Clear(SlotsOut, 0, SlotCount);
                Array.Clear(SlotsIn, 0, SlotCount);
                m_cursor = 0;
                m_currentOut = 0;
                m_currentIn = 0;
                m_lastSampleTicks = now;
                return;
            }

            for (int i = 0; i < steps; i++)
            {
                SlotsOut[m_cursor] = m_currentOut;
                SlotsIn[m_cursor] = m_currentIn;
                m_currentOut = 0;
                m_currentIn = 0;
                m_cursor = (m_cursor + 1) % SlotCount;
            }

            m_lastSampleTicks += (long)(steps * SlotSeconds * Stopwatch.Frequency);
        }
    }
}
```

```csharp
// 这个文件里有：
//   CountingTransport（类）：给任意 ITransport 加一层收发字节统计

using System;

namespace NBC.Framework.Net
{
    // 主要功能：包住任意 ITransport，把发送 / 收到的 payload 字节数交给 NetRuntimeStats，其余行为原样转发。
    // 主要数据成员：m_inner（被包的传输层）。
    // 主要外部方法：ITransport 全套（Connect / Send / Pump / Close / IsConnected / 两个事件）。
    // 谁在用它：建 transport 的地方（今天 Editor 窗口，以后运行时启动器）。
    // 注意：统计的是**业务字节**（payload 长度），不含 TCP 头与分帧头 ⇒ 比抓包工具的数字小，这是预期。
    public sealed class CountingTransport : ITransport
    {
        // 被包的传输层（真实的 TcpTransport，或测试用的 SimulatedTransport）
        private readonly ITransport m_inner;

        // 已经 Dispose 就不再转发（避免二次释放）
        private bool m_disposed;

        // 入参：inner 被包的传输层
        // 返回：无
        // 上游：建 transport 的地方
        // 下游：无
        public CountingTransport(ITransport inner)
        {
            if (inner == null)
            {
                throw new ArgumentNullException(nameof(inner), "[CountingTransport] 被包的传输层不能为 null。");
            }

            m_inner = inner;
            m_inner.FrameReceived += OnInnerFrameReceived;
            m_inner.Closed += OnInnerClosed;
        }

        // 入参：无
        // 返回：底层当前状态（连接中 / 已连接 / 已关闭）
        public ETransportState State
        {
            get { return m_inner.State; }
        }

        // 入参：无
        // 返回：底层自带的收发统计（原样转发；本装饰器只额外记字节，不改它的统计）
        public TransportStats Stats
        {
            get { return m_inner.Stats; }
        }

        // 入参：无
        // 返回：底层的人话描述（日志用）
        public string Description
        {
            get { return m_inner.Description; }
        }

        // 入参：无
        // 返回：底层是否已连接
        public bool IsConnected
        {
            get { return m_inner.IsConnected; }
        }

        // 收到一帧时抛出（先计数再抛，保证统计不漏）
        public event Action<byte[]> FrameReceived;

        // 连接关闭时抛出
        public event Action<string> Closed;

        // 入参：host / port
        // 返回：无（连接是异步的，结果看 Closed 或 IsConnected）
        public void Connect(string host, int port)
        {
            m_inner.Connect(host, port);
        }

        // 入参：payload 业务负载
        // 返回：无
        // 步骤：1) 先记字节数；2) 再交给底层发送
        // 上游：NetSession
        // 下游：ITransport.Send
        public void Send(byte[] payload)
        {
            if (payload != null)
            {
                NetRuntimeStats.AddSent(payload.Length);
            }

            m_inner.Send(payload);
        }

        // 入参：无
        // 返回：无
        public void Pump()
        {
            m_inner.Pump();
        }

        // 入参：无
        // 返回：无
        public void Close()
        {
            m_inner.Close();
        }

        // 入参：无
        // 返回：无
        public void Dispose()
        {
            if (m_disposed)
            {
                return;
            }

            m_disposed = true;
            m_inner.FrameReceived -= OnInnerFrameReceived;
            m_inner.Closed -= OnInnerClosed;
            m_inner.Dispose();
        }

        // 入参：payload 收到的一帧
        // 返回：无
        // 步骤：1) 记字节数；2) 再抛给订阅者（订阅者抛异常也不会漏计）
        // 上游：被包的传输层
        // 下游：FrameReceived 订阅者
        private void OnInnerFrameReceived(byte[] payload)
        {
            if (payload != null)
            {
                NetRuntimeStats.AddReceived(payload.Length);
            }

            Action<byte[]> handler = FrameReceived;

            if (handler != null)
            {
                handler(payload);
            }
        }

        // 入参：reason 关闭原因
        // 返回：无
        private void OnInnerClosed(string reason)
        {
            Action<string> handler = Closed;

            if (handler != null)
            {
                handler(reason);
            }
        }
    }
}
```

```csharp
// 这个文件里有：
//   NetHudPanel（类）：屏幕右上角的网络状态面板（模式 / 带宽 / RTT / 帧号）

using NBC.Framework.Net;
using NBC.Framework.UI;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace NBC.Game.UI
{
    // 主要功能：每 0.25 秒读一次 NetRuntimeStats，把模式、上下行带宽、RTT、帧号写成文本；F1 显隐。
    // 主要数据成员：四个文本控件、m_box（可见性容器）、m_elapsed（刷新计时）。
    // 主要外部方法：无，只有 Unity 的 Update 回调。
    // 谁在用它：UIManager.ShowPanel("NetHudPanel", UILayer.Bot) 打开；演示/联调时用。
    public sealed class NetHudPanel : BasePanel
    {
        // 模式文本的控件名（prefab 里的 GameObject 名字，必须一致）
        public const string ModeControl = "Text_Mode";

        // 带宽文本的控件名
        public const string SpeedControl = "Text_Speed";

        // RTT 文本的控件名
        public const string RttControl = "Text_Rtt";

        // 帧号文本的控件名
        public const string FrameControl = "Text_Frame";

        // 可见性容器（F1 隐藏它；隐藏自己会让 Update 停掉，F1 就再也按不回来）
        public const string BoxControl = "Box";

        // 刷新间隔（秒）：4 次/秒足够跟手，又不必每帧刷文本
        private const float RefreshInterval = 0.25f;

        // 显示 / 隐藏的快捷键
        private const KeyCode ToggleKey = KeyCode.F1;

        private TMP_Text m_mode;
        private TMP_Text m_speed;
        private TMP_Text m_rtt;
        private TMP_Text m_frame;

        // F1 切换的对象（没有就用自己）
        private GameObject m_box;

        // 距离上次刷新累积的时间
        private float m_elapsed;

        // 入参：无（BasePanel.Initialize 里调）
        // 返回：无
        // 步骤：1) 取四个必需文本；2) 取可选的 Box（取不到就整块面板一起隐藏）
        // 上游：BasePanel.Initialize
        // 下游：BasePanel.RequireControl / GetControl
        protected override void OnInit()
        {
            m_mode = RequireControl<TMP_Text>(ModeControl);
            m_speed = RequireControl<TMP_Text>(SpeedControl);
            m_rtt = RequireControl<TMP_Text>(RttControl);
            m_frame = RequireControl<TMP_Text>(FrameControl);

            TMP_Text boxText = GetControl<TMP_Text>(BoxControl);

            if (boxText != null)
            {
                m_box = boxText.gameObject;
            }

            LayoutElement layout = GetControl<LayoutElement>(BoxControl);

            if (m_box == null && layout != null)
            {
                m_box = layout.gameObject;
            }

            if (m_box == null)
            {
                m_box = gameObject;
            }
        }

        // 入参：无（引擎每帧调）
        // 返回：无
        // 步骤：
        //   1) F1 切换显隐（只切 Box，UI 面板本身保持启用，否则读不到按键）
        //   2) 计时到 0.25 秒才刷新一次文本
        //   3) 读一次快照，四个字段分别写文本
        // 上游：Unity 引擎
        // 下游：NetRuntimeStats.Snapshot
        private void Update()
        {
            if (Input.GetKeyDown(ToggleKey) && m_box != null)
            {
                m_box.SetActive(!m_box.activeSelf);
            }

            m_elapsed += Time.unscaledDeltaTime;

            if (m_elapsed < RefreshInterval)
            {
                return;
            }

            m_elapsed = 0f;

            NetStatsSnapshot snapshot = NetRuntimeStats.Snapshot();

            m_mode.text = DescribeMode(snapshot.Mode);
            m_speed.text = FormatSpeed(snapshot);
            m_rtt.text = snapshot.RttMs < 0 ? "RTT --" : "RTT " + snapshot.RttMs + " ms";
            m_frame.text = snapshot.Frame < 0 ? "帧 --" : "帧 " + snapshot.Frame;
        }

        // 入参：mode 模式枚举
        // 返回：给人看的中文
        private static string DescribeMode(NetSyncMode mode)
        {
            switch (mode)
            {
                case NetSyncMode.StateSync:
                    return "状态同步";
                case NetSyncMode.Lockstep:
                    return "帧同步";
                default:
                    return "离线";
            }
        }

        // 入参：snapshot 统计快照
        // 返回：形如「↑ 1.2 KB/s  ↓ 3.4 KB/s」的文本
        private static string FormatSpeed(NetStatsSnapshot snapshot)
        {
            return "↑ " + ToKilobytes(snapshot.BytesPerSecondOut) + "  ↓ " + ToKilobytes(snapshot.BytesPerSecondIn);
        }

        // 入参：bytesPerSecond 字节每秒
        // 返回：保留一位小数的 KB/s 文本
        private static string ToKilobytes(float bytesPerSecond)
        {
            float kb = bytesPerSecond / 1024f;
            return kb.ToString("0.0") + " KB/s";
        }
    }
}
```

```csharp
using NBC.Framework.Net;       // R004：CountingTransport / NetRuntimeStats / NetSyncMode

// ② ConnectSession：建 transport 后包一层，并先清零
m_transport = new TcpTransport();
ITransport counting = new CountingTransport(m_transport);
NetRuntimeStats.Reset();
m_session = new NetSession(counting, /* …其余参数不变… */);

// ③ 握手完成回调里：模式跟着窗口的锁步开关
NetRuntimeStats.SetMode(m_lockstepMode ? NetSyncMode.Lockstep : NetSyncMode.StateSync);

// ④ SetLockstepMode：切换时更新模式（未连接时不谎报在线）
if (m_session != null) { NetRuntimeStats.SetMode(on ? NetSyncMode.Lockstep : NetSyncMode.StateSync); }

// ⑤ FeedPresentationLayer 末尾：每帧写 RTT 与帧号（两种模式语义不同）
NetRuntimeStats.SetRtt(m_session.RttMs);
NetRuntimeStats.SetFrame(m_lockstepMode
    ? (client == null ? -1 : client.LastSteppedFrame)
    : (int)m_session.World.ServerTick);

// ⑥ CloseSession：断开清零
NetRuntimeStats.Reset();
```
