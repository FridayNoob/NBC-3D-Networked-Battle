## 二、功能概述

每个观察者只收到「自己的实体 + 3×3 邻域内的实体」；同格玩家共用一份 payload；找不到英雄的会话兜底发全量。

## 三、接口与入参出参

| 成员 | 入参 | 出参 |
| --- | --- | --- |
| `AoiGrid.CellOf(mm)` | 坐标（毫米） | 格子下标（向下取整） |
| `AoiGrid.IsVisible(vx,vz,tx,tz)` | 观察者/目标坐标 | bool |
| `DungeonBattle.ToSnapshot(filter)` | 过滤委托（null=全量） | `WorldSnapshot` |
| `RoomBattleService.AoiCellSizeMm` | 静态配置（默认 10000） | — |

## 四、逐段讲解

1. **格子计算**：非负直接整除；负数先取绝对值、向上取整再取反 —— 保证 -1mm 与 1mm 分处相邻格（用 0 附近塌成一格）。
2. **可见性**：两个轴的格子差都 ≤ 1（3×3）。
3. **快照过滤**：`ToSnapshot()` 保留原签名（内部传 null，行为不变），新增重载接收委托。
4. **分组发送**：按英雄格子分桶，每桶一份 payload；过滤条件 = 自身实体 or `IsVisible`；发送用 `long sessionId`（`SessionId` 是 long，踩过一次）。
5. **兜底**：英雄还没建出来的会话单独发全量，避免整屏空白。

### `Server/NBC.Server.Game/Aoi/AoiGrid.cs`

```csharp
// 这个文件里有：
//   AoiGrid（类）：九宫格兴趣管理 —— 判断"某个观察者该不该看到某个实体"。
//
// 它只决定**发不发**，不参与任何战斗判定；服务端的模拟永远看全世界。

using System;

namespace NBC.Server.Game.Aoi
{
    // 主要功能：按固定格子把平面切开，用"3×3 邻域"判断可见性。
    // 主要数据成员：CellSizeMm（格子边长，毫米）。
    // 主要外部方法：CellOf（坐标→格子）、IsVisible（观察者→目标是否在邻域内）。
    // 谁在用它：RoomBattleService.BroadcastSnapshot（按观察者过滤快照）。
    // 注意：坐标可以是负数，除法必须向下取整（不是朝零取整），否则 0 附近会塌成一格。
    public sealed class AoiGrid
    {
        // 默认格子边长（毫米）：10 米。demo 场地 10×10 米 ⇒ 等于全可见，行为不突变
        public const int DefaultCellSizeMm = 10_000;

        // 邻域半径（格）：1 = 3×3
        public const int NeighbourhoodRadiusCells = 1;

        // 格子边长（毫米）
        public int CellSizeMm { get; }

        // 功能：构造。
        // 入参：cellSizeMm 格子边长（毫米，必须 > 0；非法值回落到默认）
        // 返回：无
        // 步骤：夹住非法值
        // 上游：RoomBattleService
        // 下游：CellOf / IsVisible
        public AoiGrid(int cellSizeMm = DefaultCellSizeMm)
        {
            CellSizeMm = cellSizeMm > 0 ? cellSizeMm : DefaultCellSizeMm;
        }

        // 功能：把坐标换成格子下标（向下取整）。
        // 入参：mm 坐标（毫米，可为负）
        // 返回：格子下标
        // 步骤：非负直接整除；负数先转正再向上取整后取反
        // 上游：IsVisible
        // 下游：无
        public int CellOf(int mm)
        {
            if (mm >= 0)
            {
                return mm / CellSizeMm;
            }

            return -((-mm + CellSizeMm - 1) / CellSizeMm);
        }

        // 功能：判断目标是否在观察者的邻域内。
        // 入参：viewerXmm/viewerZmm 观察者坐标；targetXmm/targetZmm 目标坐标
        // 返回：true = 该发给这个观察者
        // 步骤：算两侧格子下标 → 两个轴的差都 <= 半径
        // 上游：RoomBattleService.BroadcastSnapshot
        // 下游：CellOf
        public bool IsVisible(int viewerXmm, int viewerZmm, int targetXmm, int targetZmm)
        {
            int dx = Math.Abs(CellOf(viewerXmm) - CellOf(targetXmm));
            int dz = Math.Abs(CellOf(viewerZmm) - CellOf(targetZmm));
            return dx <= NeighbourhoodRadiusCells && dz <= NeighbourhoodRadiusCells;
        }
    }
}
```

public WorldSnapshot ToSnapshot(Func<BattleEntity, bool> include)
{
    var snapshot = new WorldSnapshot { ServerTick = (int)Tick };

    for (int i = 0; i < _entities.Count; i++)
    {
        if (include == null || include(_entities[i]))
        {
            snapshot.Entities.Add(_entities[i].ToSnapshot());
        }
    }

    return snapshot;

### `RoomBattleService.cs`（新增 AOI 字段 + 重写 BroadcastSnapshot）

```csharp
// AOI 格子边长（毫米）：默认 10 米；测试可以调小来验证「远处不发」
public static int AoiCellSizeMm = AoiGrid.DefaultCellSizeMm;

private static AoiGrid? s_aoi;
private static int s_aoiCellMm;

private static AoiGrid GetAoiGrid()
{
    if (s_aoi == null || s_aoiCellMm != AoiCellSizeMm)
    {
        s_aoiCellMm = AoiCellSizeMm;
        s_aoi = new AoiGrid(AoiCellSizeMm);
    }

    return s_aoi;
}

private sealed class AoiGroup
{
    public int CenterXmm;
    public int CenterZmm;
    public readonly List<long> SessionIds = new();
    public readonly HashSet<int> SelfEntityIds = new();
}

private int BroadcastSnapshot(DungeonBattle battle, Room room)
{
    AoiGrid grid = GetAoiGrid();
    Dictionary<long, AoiGroup> groups = new();
    List<long> blindSessionIds = new();

    for (int i = 0; i < room.Seats.Count; i++)
    {
        RoomSeat seat = room.Seats[i];
        BattleEntity? hero = battle.FindHeroOfPlayer(seat.Session.PlayerId);

        if (hero == null)
        {
            blindSessionIds.Add(seat.Session.SessionId);
            continue;
        }

        int cx = grid.CellOf(hero.PosXmm);
        int cz = grid.CellOf(hero.PosZmm);
        long key = ((long)cx << 32) ^ (uint)cz;

        if (!groups.TryGetValue(key, out AoiGroup group))
        {
            group = new AoiGroup { CenterXmm = hero.PosXmm, CenterZmm = hero.PosZmm };
            groups[key] = group;
        }

        group.SessionIds.Add(seat.Session.SessionId);
        group.SelfEntityIds.Add(hero.Id);
    }

    int sent = 0;

    foreach (AoiGroup group in groups.Values)
    {
        byte[] payload = new ServerMessage
        {
            Snapshot = battle.ToSnapshot(e => group.SelfEntityIds.Contains(e.Id)
                || grid.IsVisible(group.CenterXmm, group.CenterZmm, e.PosXmm, e.PosZmm)),
        }.ToByteArray();

        for (int i = 0; i < group.SessionIds.Count; i++)
        {
            if (_transport.Send(group.SessionIds[i], payload))
            {
                sent++;
            }
        }
    }

    if (blindSessionIds.Count > 0)
    {
        byte[] full = new ServerMessage { Snapshot = battle.ToSnapshot() }.ToByteArray();

        for (int i = 0; i < blindSessionIds.Count; i++)
        {
            if (_transport.Send(blindSessionIds[i], full))
            {
                sent++;
            }
        }
    }

    SnapshotsSent += sent;
    return sent;
}
```
