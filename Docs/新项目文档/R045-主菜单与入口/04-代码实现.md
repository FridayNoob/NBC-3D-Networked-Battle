## 一、子任务与约束

| # | 子任务 | 约束 |
| --- | --- | --- |
| 1 | `MainMenuState` 纯逻辑 | 不引用 UnityEngine，可 EditMode 直测 |
| 2 | `MainMenuView` 运行时 UGUI | 不写业务判断，只画和收点击 |
| 3 | `MainMenuController` 串流程 | 连接失败必须给原因，不能静默 |
| 4 | Login 场景挂 `[MainMenu]` | 只加一个物体，不改相机与灯 |
| 5 | EditMode 用例 | 覆盖昵称规范化、连点保护、失败提示 |

## 二、功能概述

`Login` 场景里出现标题、昵称输入框、剑士卡片与「开始游戏 / 设置 / 退出」三个按钮。点开始后建一个常驻会话物体连服，3 秒内成功就切 `World`，失败就在按钮下方写明原因并可重试。

## 三、接口与入参出参

| 成员 | 入参 | 出参 | 约定 |
| --- | --- | --- | --- |
| `MainMenuState.NormalizeName` | `string raw` | `string` | 空/空白 →「玩家」；最长 12 字 |
| `MainMenuState.BeginConnect` | 无 | `bool` | 已连接或连接中返回 false（连点保护） |
| `MainMenuState.Fail` | `string reason` | 无 | 空原因用默认文案 |
| `MainMenuState.Succeed` | 无 | 无 | 之后 `CanStart` 保持 false |
| `MainMenuView.Bind` | 三个 `Action` | 无 | 重复绑定会先清旧监听 |
| `MainMenuController.StartGame` | 无 | 无 | 昵称存 `PlayerPrefs` 后再连 |
| `MainMenuController.State` | 无 | `MainMenuState` | 排查/测试读取 |

## 四、逐段讲解

1. **为什么逻辑与界面分开**：昵称规范化和「能不能点」是纯判断，放 `MainMenuState` 就能用单测钉死；`MainMenuView` 只负责画。
2. **昵称规范化**：`Replace("\r"/"\n")` + `Trim()` + 截断 12 字，空输入回「玩家」。放到逻辑层是为了不改界面也能测。
3. **连点保护**：`BeginConnect` 自检 `CanStart`，控制器不再判第二次；按钮同时也被置灰（双保险）。
4. **会话物体常驻**：`[MenuSession]` + `DontDestroyOnLoad`，`AutoJoin=false`、`BindPresentation=false`，进 `World` 后仍是同一条连接。
5. **失败可重试**：失败时销毁这次建的会话物体，下次点开始是全新连接；提示写在按钮下方，不用翻 Console。
6. **切场景**：优先 `SceneFlow.LoadAsync`（编辑器内走 SceneManager），没有 `SceneFlow` 才直接 `SceneManager.LoadScene("World")`。
7. **退出**：`#if UNITY_EDITOR` 里停播放，构建版 `Application.Quit()`。
8. **场景挂载**：`Login.unity` 里只新增 `[MainMenu]`（Transform + `MainMenuController`），视图由控制器运行时 `AddComponent` 建出来。

// 这个文件里有：
// MainMenuState：主菜单的纯逻辑状态（昵称规范化、按钮可用性、提示文案），不引用任何 UnityEngine 类型。

using System;

namespace NBC.Game.Menu
{
    /// <summary>
    /// 主菜单的纯逻辑状态。
    /// <para>主要功能：把「昵称怎么算合法」「什么时候能点开始」「按钮下方显示什么」三件事收在一起，方便单测。</para>
    /// <para>主要数据成员：m_playerName（规范化后的昵称）、m_connecting（连接中）、m_succeeded（已连上）、m_hint（提示）。</para>
    /// <para>主要外部方法：BeginConnect / Fail / Succeed 三个状态转移，CanStart / IsConnecting 两个查询。</para>
    /// <para>谁在用它：MainMenuController（读状态改界面）、EditMode 用例（直接断言状态）。</para>
    /// </summary>
    public sealed class MainMenuState
    {
        /// <summary>昵称留空时用的名字。</summary>
        public const string DefaultName = "玩家";

        /// <summary>昵称最多几个字（超出截断）。</summary>
        public const int MaxNameLength = 12;

        /// <summary>连接失败且没拿到服务端原因时的默认提示。</summary>
        public const string DefaultFailureHint = "连不上服务器，请确认服务端已启动（端口 7778）";

        private string m_playerName = DefaultName;
        private bool m_connecting;
        private bool m_succeeded;
        private string m_hint = string.Empty;

        /// <summary>规范化后的昵称（空输入会变成「玩家」）。</summary>
        public string PlayerName
        {
            get { return m_playerName; }
        }

        /// <summary>现在能不能点「开始游戏」（连接中或已连上都不行）。</summary>
        public bool CanStart
        {
            get { return !m_connecting && !m_succeeded; }
        }

        /// <summary>是否正在连接。</summary>
        public bool IsConnecting
        {
            get { return m_connecting; }
        }

        /// <summary>是否已经连上并准备切场景。</summary>
        public bool IsSucceeded
        {
            get { return m_succeeded; }
        }

        /// <summary>按钮下方的提示（空串表示不显示）。</summary>
        public string Hint
        {
            get { return m_hint; }
        }

        /// <summary>
        /// 把玩家输入的名字规范化。
        /// </summary>
        /// <param name="raw">输入框里的原始文本，允许 null。</param>
        /// <returns>去掉首尾空白与换行、长度 1~12 的名字；空输入返回「玩家」。</returns>
        /// <remarks>
        /// 把这件事放在纯逻辑里，是为了让「留空 / 只有空格 / 超长」三种情况都能用单测覆盖，
        /// 不必为了验一个 trim 去进 Play 模式。
        /// </remarks>
        public static string NormalizeName(string raw)
        {
            if (string.IsNullOrEmpty(raw))
            {
                return DefaultName;
            }

            string cleaned = raw.Replace("\r", string.Empty).Replace("\n", string.Empty).Trim();

            if (cleaned.Length == 0)
            {
                return DefaultName;
            }

            if (cleaned.Length > MaxNameLength)
            {
                cleaned = cleaned.Substring(0, MaxNameLength);
            }

            return cleaned;
        }

        /// <summary>
        /// 记下昵称。
        /// </summary>
        /// <param name="raw">输入框里的原始文本。</param>
        /// <returns>规范化后的昵称。</returns>
        public string SetPlayerName(string raw)
        {
            m_playerName = NormalizeName(raw);
            return m_playerName;
        }

        /// <summary>
        /// 开始连接：按钮变灰、提示清空。
        /// </summary>
        /// <returns>这次调用是否真的开始了连接（已经在连接或已连上时返回 false）。</returns>
        public bool BeginConnect()
        {
            if (!CanStart)
            {
                return false;
            }

            m_connecting = true;
            m_succeeded = false;
            m_hint = string.Empty;
            return true;
        }

        /// <summary>
        /// 连接失败：恢复按钮、记下失败原因。
        /// </summary>
        /// <param name="reason">人话原因；为空时用默认文案。</param>
        public void Fail(string reason)
        {
            m_connecting = false;
            m_succeeded = false;
            m_hint = string.IsNullOrEmpty(reason) ? DefaultFailureHint : reason;
        }

        /// <summary>
        /// 连接成功：按钮保持不可点，提示改成「正在进入世界…」。
        /// </summary>
        public void Succeed()
        {
            m_connecting = false;
            m_succeeded = true;
            m_hint = "已连接，正在进入世界…";
        }
    }
}

        private Canvas m_canvas;

        private Button m_startButton;
        private Text m_hint;

        /// <summary>
        /// 把三个按钮接到回调上。
        /// </summary>
        /// <param name="onStart">点「开始游戏」。</param>
        /// <param name="onSettings">点「设置」。</param>
        /// <param name="onQuit">点「退出」。</param>
        public void Bind(Action onStart, Action onSettings, Action onQuit)
        {
            BuildIfNeeded();

            m_startButton.onClick.RemoveAllListeners();
            m_startButton.onClick.AddListener(() => { if (onStart != null) { onStart(); } });

            // 设置与退出按钮在 Build 里各存了一个 Button 字段
            m_settingsButton.onClick.RemoveAllListeners();
            m_settingsButton.onClick.AddListener(() => { if (onSettings != null) { onSettings(); } });

            m_quitButton.onClick.RemoveAllListeners();
            m_quitButton.onClick.AddListener(() => { if (onQuit != null) { onQuit(); } });
        }

        private Button m_settingsButton;
        private Button m_quitButton;

        /// <summary>读输入框里的原始文本。</summary>
        /// <returns>输入框文本；视图还没建好时返回空串。</returns>
        public string ReadName()
        {
            return m_nameInput != null ? m_nameInput.text : string.Empty;
        }

        /// <summary>把名字写进输入框（启动时回填上次昵称）。</summary>
        /// <param name="name">要显示的昵称。</param>
        public void SetName(string name)
        {
            BuildIfNeeded();

            if (m_nameInput != null)
            {
                m_nameInput.text = name ?? string.Empty;
            }
        }

        /// <summary>改开始按钮上的文字（空闲「开始游戏」、连接中「连接中…」）。</summary>
        /// <param name="text">新文字。</param>
        public void SetStartLabel(string text)
        {
            if (m_startLabel != null)
            {
                m_startLabel.text = text;
            }
        }

        /// <summary>设开始按钮能不能点。</summary>
        /// <param name="interactable">true 可点。</param>
        public void SetStartInteractable(bool interactable)
        {
            if (m_startButton != null)
            {
                m_startButton.interactable = interactable;
            }
        }

        /// <summary>设置按钮下方的提示（空串表示隐藏）。</summary>
        /// <param name="hint">提示文字。</param>
        public void SetHint(string hint)
        {
            if (m_hint != null)

                m_hint.text = hint ?? string.Empty;

        // 功能：第一次用到时把界面搭出来（重复调用无副作用）。

        // 返回：无。
        // 步骤：1) 已有 Canvas 就直接返回 2) 建 Canvas 与缩放器 3) 摆标题、昵称、卡片、按钮、提示。

            if (m_canvas != null)

            }

            canvasObject.transform.SetParent(transform, false);

            m_quitButton = CreateButton(canvasObject.transform, "QuitButton", "退出",

        // 入参：parent 父节点。
        // 返回：无。
        private static void CreateBackground(Transform parent)

            GameObject go = new GameObject("Background");
            go.transform.SetParent(parent, false);

            RectTransform rect = go.AddComponent<RectTransform>();
            rect.anchorMin = Vector2.zero;
            rect.anchorMax = Vector2.one;
            rect.offsetMin = Vector2.zero;
            rect.offsetMax = Vector2.zero;

        {
            GameObject go = new GameObject("NameInput");
            go.transform.SetParent(parent, false);

            RectTransform rect = go.AddComponent<RectTransform>();
            rect.anchorMin = new Vector2(0.5f, 0.5f);
            rect.anchorMax = new Vector2(0.5f, 0.5f);

            rect.anchoredPosition = position;
            rect.sizeDelta = size;

            Image background = go.AddComponent<Image>();
            background.color = new Color(1f, 1f, 1f, 0.94f);

            GameObject textObject = new GameObject("Text");
            textObject.transform.SetParent(go.transform, false);
            RectTransform textRect = textObject.AddComponent<RectTransform>();
            textRect.anchorMin = Vector2.zero;
            textRect.anchorMax = Vector2.one;

            text.fontSize = 34;
            text.color = new Color(0.08f, 0.08f, 0.1f, 1f);

            text.supportRichText = false;

            InputField input = go.AddComponent<InputField>();
            input.targetGraphic = background;
            input.textComponent = text;
            input.lineType = InputField.LineType.SingleLine;
            input.characterLimit = MainMenuState.MaxNameLength;
            return input;
        }

        {
            GameObject go = new GameObject(name);
            go.transform.SetParent(parent, false);

            RectTransform rect = go.AddComponent<RectTransform>();
            rect.anchorMin = new Vector2(0.5f, 0.5f);
            rect.anchorMax = new Vector2(0.5f, 0.5f);
            rect.pivot = new Vector2(0.5f, 0.5f);
            rect.anchoredPosition = position;
            rect.sizeDelta = size;

            Image image = go.AddComponent<Image>();
            image.color = new Color(0.22f, 0.42f, 0.72f, 1f);

            Button button = go.AddComponent<Button>();
            button.targetGraphic = image;

            labelText = CreateText(go.transform, "Label", label, 36,
                Vector2.zero, size, TextAnchor.MiddleCenter);
            return button;
        }

        // 功能：建一段居中的文字。
        // 入参：parent 父节点；name 物体名；content 内容；fontSize 字号；position 偏移；size 尺寸；alignment 对齐。
        // 返回：文字组件。
        private static Text CreateText(Transform parent, string name, string content, int fontSize,
            Vector2 position, Vector2 size, TextAnchor alignment)
        {
            GameObject go = new GameObject(name);
            go.transform.SetParent(parent, false);

            RectTransform rect = go.AddComponent<RectTransform>();
            rect.anchorMin = new Vector2(0.5f, 0.5f);
            rect.anchorMax = new Vector2(0.5f, 0.5f);
            rect.pivot = new Vector2(0.5f, 0.5f);
            rect.anchoredPosition = position;
            rect.sizeDelta = size;

            Text text = go.AddComponent<Text>();
            text.font = LoadBuiltinFont();
            text.fontSize = fontSize;
            text.text = content;
            text.alignment = alignment;
            text.color = Color.white;
            text.raycastTarget = false;
            text.horizontalOverflow = HorizontalWrapMode.Overflow;
            text.verticalOverflow = VerticalWrapMode.Overflow;
            return text;
        }

        // 功能：确保场景里有一个 EventSystem，否则按钮收不到点击。
        // 入参：无。
        // 返回：无。
        private static void EnsureEventSystem()
        {
            if (UnityEngine.Object.FindObjectOfType<EventSystem>() != null)
            {
                return;
            }

            GameObject go = new GameObject("EventSystem");
            go.AddComponent<EventSystem>();
            go.AddComponent<StandaloneInputModule>();
        }

        // 功能：取一个内置字体给 UGUI 用。
        // 入参：无。
        // 返回：内置字体；两个名字都取不到时返回 null（文字不显示，但不会崩）。
        private static Font LoadBuiltinFont()
        {
            Font font = Resources.GetBuiltinResource<Font>(BuiltinFontName);

            if (font == null)
            {
                font = Resources.GetBuiltinResource<Font>(LegacyBuiltinFontName);
            }

            return font;
        }
    }
}

// 这个文件里有：
// MainMenuController：主菜单的流程控制（读昵称、点开始连服、成功切 World、失败提示、设置占位、退出）。

using NBC.Game.Net;

        }

        // 功能：启动时建界面、回填昵称、绑按钮。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 拿或加 View 2) 读 PlayerPrefs 回填 3) 绑三个回调 4) 按状态刷一次文字。
        private void Start()
        {
            m_view = GetComponent<MainMenuView>();

            if (m_view == null)
            {
                m_view = gameObject.AddComponent<MainMenuView>();
            }

            string saved = PlayerPrefs.GetString(PlayerNameKey, string.Empty);
            m_state.SetPlayerName(saved);
            m_view.SetName(m_state.PlayerName);
            m_view.Bind(StartGame, ToggleSettings, QuitGame);
            Refresh();
        }

        // 功能：点「开始游戏」——记昵称、建常驻会话、开始连服。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 状态机拒绝（正在连/已连上）就直接返回 2) 规范化并存昵称 3) 找或建会话物体 4) 起协程等连接。
        // 上游/下游：View 的按钮回调；下游 StartCoroutine(ConnectRoutine)。
        public void StartGame()
        {
            if (!m_state.BeginConnect())
            {
                return;
            }

            m_state.SetPlayerName(m_view.ReadName());
            PlayerPrefs.SetString(PlayerNameKey, m_state.PlayerName);
            PlayerPrefs.Save();

            EnsureSession();
            Refresh();

            StartCoroutine(ConnectRoutine());
        }

        // 功能：设置图标的占位行为（真设置留给 R047）。
        // 入参：无。
        // 返回：无。
        private void ToggleSettings()
        {
            m_view.SetHint("设置将在 R047 做（音量 / 画质 / 鼠标灵敏度）");
        }

        // 功能：退出游戏（编辑器里停止播放，构建版退出进程）。
        // 入参：无。
        // 返回：无。
        private void QuitGame()
        {
#if UNITY_EDITOR
            UnityEditor.EditorApplication.isPlaying = false;
#else
            Application.Quit();
#endif
        }

        // 功能：等连接结果（成功切场景 / 失败提示）。
        // 入参：无。
        // 返回：协程。
        // 步骤：1) 每帧看 IsOnline 2) 超时或拿到失败原因就失败 3) 成功就切 World。
        private IEnumerator ConnectRoutine()
        {
            float deadline = Time.realtimeSinceStartup + Mathf.Max(0.5f, m_connectTimeoutSeconds);

            while (Time.realtimeSinceStartup < deadline)
            {
                if (m_driver != null && m_driver.IsOnline)
                {
                    m_state.Succeed();
                    Refresh();
                    SwitchToWorld();
                    yield break;
                }

                if (m_driver != null && !string.IsNullOrEmpty(m_driver.FailureReason))
                {
                    FailAndCleanup(m_driver.FailureReason);
                    yield break;
                }

                yield return null;
            }

            string reason = m_driver != null && !string.IsNullOrEmpty(m_driver.LastError)
                ? m_driver.LastError
                : "连不上服务器（" + Host + ":" + Port + "），请确认服务端已启动";
            FailAndCleanup(reason);
        }

        // 功能：失败后恢复界面，并把这次建的会话物体销毁（下次点开始是全新连接）。
        // 入参：reason 失败原因。
        // 返回：无。
        private void FailAndCleanup(string reason)
        {
            m_state.Fail(reason);
            Refresh();

            if (m_sessionObject != null)
            {
                Destroy(m_sessionObject);
                m_sessionObject = null;
                m_driver = null;
            }
        }

        // 功能：切到 World（已经连上，会话物体常驻过去）。
        // 入参：无。
        // 返回：无。
        private void SwitchToWorld()
        {
            if (m_switching)
            {
                return;
            }

            m_switching = true;

            if (m_logDiagnostics)
            {
                Debug.Log("[主菜单] 已连接，进入世界（昵称 " + m_state.PlayerName + "）");
            }

            if (SceneFlow.Instance != null)
            {
                SceneFlow.Instance.LoadAsync(GameScene.World, OnSceneLoaded);
                return;
            }

            SceneManager.LoadScene("World");
        }

        // 功能：场景加载回调（只记日志；失败时把原因写在提示行）。
        // 入参：ok 是否成功；message 失败原因。
        // 返回：无。
        private void OnSceneLoaded(bool ok, string message)
        {
            if (ok)
            {
                return;
            }

            m_switching = false;
            m_state.Fail("世界场景加载失败：" + message);
            Refresh();
        }

        // 功能：找或建常驻会话物体，并把驱动配成「只连服、不进房、不接管表现」。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 先找同名物体（重复点开始要复用）2) 没有就新建并 DontDestroyOnLoad 3) 配连接参数与两个开关。
        private void EnsureSession()
        {
            if (m_sessionObject != null)
            {
                return;
            }

            GameObject existing = GameObject.Find(SessionObjectName);

            if (existing != null)
            {
                m_sessionObject = existing;
            }
            else
            {
                m_sessionObject = new GameObject(SessionObjectName);
                DontDestroyOnLoad(m_sessionObject);
            }

            m_driver = m_sessionObject.GetComponent<RuntimeNetDriver>();

            if (m_driver == null)
            {
                m_driver = m_sessionObject.AddComponent<RuntimeNetDriver>();
            }

            m_driver.ConfigureSession(Host, Port, DungeonId);
            m_driver.AutoJoin = false;
            m_driver.BindPresentation = false;
        }

        // 功能：把状态刷到界面上（按钮文字、可用性、提示）。
        // 入参：无。
        // 返回：无。
        private void Refresh()
        {
            if (m_view == null)
            {
                return;
            }

            m_view.SetStartLabel(m_state.IsConnecting ? "连接中…" : "开始游戏");
            m_view.SetStartInteractable(m_state.CanStart);
            m_view.SetHint(m_state.Hint);
        }
    }
}

## 七、风险与幻觉自查

- `RuntimeNetDriver` 的 `ConfigureSession` 会写入 host/port/dungeonId；`AutoJoin`/`BindPresentation` 是属性，不是字段。
- 视图里 `out _` 丢弃参数用的是 C# 7 语法，Unity 2022 支持。
- 场景 YAML 手写了 `[MainMenu]` 与脚本 meta 的 guid，**必须在 Unity 里打开 Login 场景确认脚本没有丢**（脚本丢失会显示 Missing）。
- 测试未跑：写文档时 Unity 编辑器没有启动（MCP 超时），05 里会如实写「待跑」，并在启动后补真数字。
