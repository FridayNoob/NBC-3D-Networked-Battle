## 二、功能概述

玩家进副本后：左上角出现 BOSS 血条与最近目标血条，右上角出现最多三条任务进度；死亡时黑幕从预制体加载，显示标题、倒计时与状态行。

## 三、接口与入参出参

| 类型 | 方法 | 入参 | 出参 | 约定 |
| --- | --- | --- | --- | --- |

## 四、逐段讲解

1. **控件怎么拿到**：`BasePanel.CollectControls()` 按 **GameObject 名字** 收集 `UIBehaviour` 子组件，所以 prefab 里每个要用的节点都挂了 `Image` 或 `TMP_Text`，名字与常量逐字一致；`RectTransform` 不是 `UIBehaviour`，不能当控件。
2. **面板怎么定位**：两块 HUD 面板根上挂 `KeepPrefabTransform`（`Client/Assets/_Project/Framework.UI/KeepPrefabTransform.cs`），`UIManager.AttachToLayer` 才不会把锚点铺满、把位置清零；黑幕面板不挂，铺满整屏正是要的效果。
3. **首帧没有预制体实例**：`UIManager.ShowPanel` 是异步加载、只经 `onShown` 回调给结果；`DeathScreenOverlay.Ensure()` 被 `PlayerDeathFlow` 同步调用，所以先返回占位实例、把 `Show/SetCountdown/SetStatusText` 的参数存成静态待定值，真面板一到立刻补一遍。
4. **谁是最该打的目标**：`RuntimeHudDriver.UpdateTargets` 取英雄快照做原点，BOSS（`Kind==2`）单独一条，其余活怪按平方距离取最近；平方距离避免开方，也不会溢出（毫米级坐标 `long` 计算）。
5. **任务进度从哪来**：联机路径没有 `QuestRuntime`（它只在离线 `BattleSession` 上），所以读 `NetSession.LastQuestStates` 的全量条目，按 `1=已接取 / 2=可交付` 过滤，再查 `QuestConfig` 拿标题、`QuestConditionConfig` 拿条件说明，进度用 `NetSession.FindProgress(conditionId)`。
6. **为什么只在进副本时 `Bind`**：`RuntimeNetDriver` 跨场景常驻，HUD 驱动挂在同一个物体上；`m_hudBound` 保证一局只 `Bind` 一次，回世界 `CloseHud` 收起并把标记清掉，下一局重新接。

```csharp
// 这个文件里有：
//   HudTargetPanel（类）：副本 HUD 的「BOSS 血条 + 当前目标」面板，只按数值改填充与文字。

using NBC.Framework.UI;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace NBC.Game.UI
{
    // 主要功能：把 BOSS 与当前目标的「名字 + 当前/最大血」画成两条血条；显隐由调用方决定。
    // 主要数据成员：m_bossRoot/m_targetRoot（整行显隐）、两条 Fill（填充图）、四个文字。
    // 主要外部方法：SetBoss（BOSS 行）、SetTarget（目标行）。
    // 谁在用它：RuntimeHudDriver 每帧按快照喂数据；prefab 在 Client/Assets/_Project/UI/HudTargetPanel.prefab。
    public sealed class HudTargetPanel : BasePanel
    {
        // 面板名必须等于 prefab 文件名（收集器是 AddressByFileName）
        public new const string PanelName = "HudTargetPanel";

        public const string BossRootControl = "BossRoot";
        public const string BossLabelControl = "BossLabel";
        public const string BossValueControl = "BossValue";
        public const string BossFillControl = "BossFill";
        public const string TargetRootControl = "TargetRoot";
        public const string TargetLabelControl = "TargetLabel";
        public const string TargetValueControl = "TargetValue";
        public const string TargetFillControl = "TargetFill";

        // 两条血条行与文字（BossRoot/TargetRoot 各带一张背景图，所以能当控件取到）
        private GameObject m_bossRoot;
        private GameObject m_targetRoot;
        private TMP_Text m_bossLabel;
        private TMP_Text m_bossValue;
        private Image m_bossFill;
        private TMP_Text m_targetLabel;
        private TMP_Text m_targetValue;
        private Image m_targetFill;

        // 功能：取出 prefab 里的控件（两行各四个）。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 两行根各取一张 Image 当显隐开关 2) 文字与填充图逐个取。
        // 上游：BasePanel.Initialize。
        // 下游：RequireControl / GetControl。
        protected override void OnInit()
        {
            m_bossRoot = RequireControl<Image>(BossRootControl).gameObject;
            m_targetRoot = RequireControl<Image>(TargetRootControl).gameObject;

            m_bossLabel = GetControl<TMP_Text>(BossLabelControl);
            m_bossValue = GetControl<TMP_Text>(BossValueControl);
            m_bossFill = GetControl<Image>(BossFillControl);

            m_targetLabel = GetControl<TMP_Text>(TargetLabelControl);
            m_targetValue = GetControl<TMP_Text>(TargetValueControl);
            m_targetFill = GetControl<Image>(TargetFillControl);
        }

        // 功能：设置 BOSS 行（显隐、名字、血条）。
        // 入参：visible 是否显示；label 名字；hp 当前血；maxHp 最大血。
        // 返回：无。
        // 步骤：1) 切换整行显隐 2) 隐藏就直接返回 3) 写名字与血条。
        // 上游：RuntimeHudDriver.Update。
        public void SetBoss(bool visible, string label, int hp, int maxHp)
        {
            if (m_bossRoot != null && m_bossRoot.activeSelf != visible)
            {
                m_bossRoot.SetActive(visible);
            }

            if (!visible)
            {
                return;
            }

            if (m_bossLabel != null)
            {
                m_bossLabel.text = label ?? string.Empty;
            }

            ApplyBar(m_bossFill, m_bossValue, hp, maxHp);
        }

        // 功能：设置目标行（显隐、名字、血条）。
        // 入参：visible 是否显示；label 名字；hp 当前血；maxHp 最大血。
        // 返回：无。
        // 步骤：同 SetBoss。
        // 上游：RuntimeHudDriver.Update。
        public void SetTarget(bool visible, string label, int hp, int maxHp)
        {
            if (m_targetRoot != null && m_targetRoot.activeSelf != visible)
            {
                m_targetRoot.SetActive(visible);
            }

            if (!visible)
            {
                return;
            }

            if (m_targetLabel != null)
            {
                m_targetLabel.text = label ?? string.Empty;
            }

            ApplyBar(m_targetFill, m_targetValue, hp, maxHp);
        }

        // 功能：把当前/最大血写进一条填充图与一个数字。
        // 入参：fill 填充图（可为空）；value 数字（可为空）；hp 当前血；maxHp 最大血。
        // 返回：无。
        // 步骤：1) maxHp <= 0 时比例按 0 2) hp 钳到 0..maxHp 3) 写 fillAmount 与「当前/最大」。
        private static void ApplyBar(Image fill, TMP_Text value, int hp, int maxHp)
        {
            float ratio = 0f;

            if (maxHp > 0)
            {
                int clamped = Mathf.Clamp(hp, 0, maxHp);
                ratio = (float)clamped / maxHp;

                if (value != null)
                {
                    value.text = clamped + "/" + maxHp;
                }
            }
            else if (value != null)
            {
                value.text = "0/0";
            }

            if (fill != null)
            {
                fill.fillAmount = ratio;
            }
        }
    }
}
```

```csharp
// 这个文件里有：
//   HudQuestPanel（类）：副本 HUD 的任务追踪面板，把进行中的任务拼成最多三行文字。

using System.Collections.Generic;
using System.Text;
using NBC.Framework.UI;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace NBC.Game.UI
{
    // 主要功能：把任务本里「进行中」的任务拼成最多三行「标题 · 详情」，没有就整块隐藏。
    // 主要数据成员：m_root（整块显隐）、m_tracker（一段多行文字）、m_rows（复用缓冲，避免每帧分配）。
    // 主要外部方法：SetQuest（喂任务本）。
    // 谁在用它：RuntimeHudDriver 每 0.2 秒喂一次；prefab 在 Client/Assets/_Project/UI/HudQuestPanel.prefab。
    public sealed class HudQuestPanel : BasePanel
    {
        // 面板名必须等于 prefab 文件名
        public new const string PanelName = "HudQuestPanel";

        public const string RootControl = "Root";
        public const string TrackerControl = "Text_Tracker";

        // 任务行在 prefab 里能看到的最大条数
        public const int MaxRows = 3;

        private GameObject m_root;
        private TMP_Text m_tracker;
        private readonly List<QuestRow> m_rows = new List<QuestRow>();
        private readonly StringBuilder m_builder = new StringBuilder(256);

        // 功能：取 prefab 里的控件。
        // 入参：无。
        // 返回：无。
        // 上游：BasePanel.Initialize。
        protected override void OnInit()
        {
            m_root = RequireControl<Image>(RootControl).gameObject;
            m_tracker = RequireControl<TMP_Text>(TrackerControl);
        }

        // 功能：按任务本刷新追踪行。
        // 入参：model 任务本（可为 null）。
        // 返回：无。
        // 步骤：1) 取进行中的行 2) 转给 SetRows。
        // 上游：离线调试路径（RuntimeHudDriver 直接用 SetRows）。
        public void SetQuest(QuestPanelModel model)
        {
            m_rows.Clear();

            if (model != null)
            {
                model.CopyTrackingRows(m_rows);
            }

            SetRows(m_rows);
        }

        // 功能：直接按行刷新追踪区（联机时行由 RuntimeHudDriver 拼）。
        // 入参：rows 要显示的行，最多 MaxRows 条。
        // 返回：无。
        // 步骤：1) 没行就隐藏整块 2) 有行就显示并拼「标题 · 详情」。
        // 上游：RuntimeHudDriver.UpdateQuest。
        public void SetRows(IReadOnlyList<QuestRow> rows)
        {
            if (rows == null || rows.Count == 0)
            {
                if (m_root != null && m_root.activeSelf)
                {
                    m_root.SetActive(false);
                }

                return;
            }

            if (m_root != null && !m_root.activeSelf)
            {
                m_root.SetActive(true);
            }

            m_builder.Length = 0;

            int count = Mathf.Min(MaxRows, rows.Count);

            for (int i = 0; i < count; i++)
            {
                QuestRow row = rows[i];

                if (i > 0)
                {
                    m_builder.Append('\n');
                }

                m_builder.Append(row.Title);

                if (!string.IsNullOrEmpty(row.Detail))
                {
                    m_builder.Append(" · ");
                    m_builder.Append(row.Detail);
                }
            }

            if (m_tracker != null)
            {
                m_tracker.text = m_builder.ToString();
            }
        }
    }
}
```

```csharp
// 这个文件里有：
//   DeathScreenOverlay（类）：死亡时的黑幕面板（标题 + 倒计时 + 状态行），视觉来自预制体。

using NBC.Framework.UI;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace NBC.Game.UI
{
    // 主要功能：玩家死亡时盖一层黑幕、显示标题与倒计时；复活后关掉。
    // 主要数据成员：m_backdrop（背景图，同时是显隐开关）、m_title/m_countdown/m_status（三段文字）、m_isStub（占位实例）。
    // 主要外部方法：Ensure（静态入口）、Show、SetCountdown、SetStatusText、Hide。
    // 谁在用它：PlayerDeathFlow 在死亡/复活流程里调；DungeonResultPanel 结算前调 Hide。
    // 注意：面板是异步加载的，第一帧还拿不到预制体实例 —— 这时 Ensure 返回一个占位实例，
    //       先把 Show/SetCountdown 的参数记下来，等真面板到了再补一遍（见 OnPanelShown）。
    public sealed class DeathScreenOverlay : BasePanel
    {
        // 面板名必须等于 prefab 文件名
        public new const string PanelName = "DeathScreenOverlay";

        public const string BackdropControl = "Img_Backdrop";
        public const string TitleControl = "Text_Title";
        public const string CountdownControl = "Text_Countdown";
        public const string StatusControl = "Text_Status";

        // 全屏黑幕的透明度：0.85 让玩家还能隐约看到世界，但不刺眼
        public const float BackdropAlpha = 0.85f;

        private static DeathScreenOverlay s_instance;
        private static bool s_panelRequested;

        // 占位期记下的「待应用状态」：真面板一到就照着补上
        private static bool s_wantShow;
        private static string s_wantTitle;
        private static string s_wantStatus;
        private static float s_wantCountdown = -1f;

        private Image m_backdrop;
        private TMP_Text m_title;
        private TMP_Text m_countdown;
        private TMP_Text m_status;
        private bool m_showing;
        private bool m_isStub;

        // 功能：现在黑幕是不是显示着。
        // 入参：无。
        // 返回：true 表示玩家能看到黑幕（占位实例返回「想显示」的状态）。
        public bool IsShowing
        {
            get { return m_isStub ? s_wantShow : m_showing; }
        }

        // 功能：取黑幕面板（没有就请求加载预制体）。
        // 入参：无。
        // 返回：面板实例或占位实例，保证非空。
        // 步骤：1) 已拿到真面板直接返回 2) 没有就建占位 3) 请求一次加载。
        // 上游：PlayerDeathFlow；下游：UIManager.ShowPanel。
        public static DeathScreenOverlay Ensure()
        {
            if (s_instance != null && !s_instance.m_isStub)
            {
                return s_instance;
            }

            if (s_instance == null)
            {
                GameObject stub = new GameObject("[DeathScreenOverlay]");
                s_instance = stub.AddComponent<DeathScreenOverlay>();
                s_instance.m_isStub = true;
            }

            if (!s_panelRequested)
            {
                s_panelRequested = true;
                UIManager ui = UIManager.Instance;

                if (ui != null)
                {
                    ui.ShowPanel(PanelName, UILayer.Top, OnPanelShown);
                }
            }

            return s_instance;
        }

        // 功能：真面板加载好之后的回调：接管状态、把占位期记下的内容补上。
        // 入参：panel 加载好的面板。
        // 返回：无。
        // 步骤：1) 类型不对就返回 2) 换成真实例、藏掉占位 3) 应用待定状态。
        // 上游：UIManager.ShowPanel 的 onShown。
        private static void OnPanelShown(BasePanel panel)
        {
            DeathScreenOverlay real = panel as DeathScreenOverlay;

            if (real == null)
            {
                return;
            }

            DeathScreenOverlay stub = s_instance;
            real.m_isStub = false;
            s_instance = real;

            if (stub != null && stub != real)
            {
                stub.Hide();
                Object.Destroy(stub.gameObject);
            }

            if (s_wantShow)
            {
                real.Show(s_wantTitle, s_wantStatus);

                if (s_wantCountdown >= 0f)
                {
                    real.SetCountdown(s_wantCountdown);
                }
            }
        }

        // 功能：取 prefab 里的控件。
        // 入参：无。
        // 返回：无。
        // 上游：BasePanel.Initialize。
        protected override void OnInit()
        {
            m_backdrop = RequireControl<Image>(BackdropControl);
            m_title = GetControl<TMP_Text>(TitleControl);
            m_countdown = GetControl<TMP_Text>(CountdownControl);
            m_status = GetControl<TMP_Text>(StatusControl);

            Color color = m_backdrop.color;
            color.a = BackdropAlpha;
            m_backdrop.color = color;
            m_backdrop.raycastTarget = false;
        }

        // 功能：打开黑幕并写入标题与提示。
        // 入参：title 大标题；hint 状态行（可为空）。
        // 返回：无。
        // 步骤：1) 占位实例只记状态 2) 真面板显示背景、写标题、清倒计时。
        public void Show(string title, string hint)
        {
            s_wantShow = true;
            s_wantTitle = title;
            s_wantStatus = hint;

            if (m_isStub)
            {
                return;
            }

            m_showing = true;

            if (m_backdrop != null)
            {
                m_backdrop.gameObject.SetActive(true);
            }

            if (m_title != null)
            {
                m_title.text = title ?? string.Empty;
            }

            if (m_countdown != null)
            {
                m_countdown.text = string.Empty;
            }

            SetStatusText(hint);
        }

        // 功能：更新倒计时数字。
        // 入参：remaining 剩余秒（向上取整显示）。
        // 返回：无。
        public void SetCountdown(float remaining)
        {
            s_wantCountdown = remaining;

            if (!m_isStub && m_countdown != null)
            {
                m_countdown.text = Mathf.CeilToInt(Mathf.Max(0f, remaining)).ToString();
            }
        }

        // 功能：更新状态行文字。
        // 入参：text 要显示的字（null 当空串）。
        // 返回：无。
        public void SetStatusText(string text)
        {
            s_wantStatus = text;

            if (!m_isStub && m_status != null)
            {
                m_status.text = text ?? string.Empty;
            }
        }

        // 功能：关掉黑幕（面板实例保留，下次直接复用）。
        // 入参：无。
        // 返回：无。
        public void Hide()
        {
            s_wantShow = false;
            s_wantCountdown = -1f;
            m_showing = false;

            if (!m_isStub && m_backdrop != null)
            {
                m_backdrop.gameObject.SetActive(false);
            }
        }
    }
}
```

```csharp
// 这个文件里有：
//   RuntimeHudDriver（类）：副本里唯一的 HUD 大脑，按快照挑 BOSS 与目标、按任务状态拼追踪行。

using System.Collections.Generic;
using NBC.Framework.UI;
using NBC.Game.Config;
using NBC.Game.Net;
using NBC.Protocol;
using UnityEngine;

namespace NBC.Game.UI
{
    // 主要功能：把「谁在打我 / 我在打谁 / 我还有哪些任务」翻译成三个面板上的数字与文字。
    // 主要数据成员：m_feed（表现层快照源）、m_session（任务权威）、三块面板、m_rows（复用行缓冲）。
    // 主要外部方法：Bind（进副本时接线）、CloseAll（回 World 时收起）。
    // 谁在用它：RuntimeNetDriver 在进房成功后 Bind、返回世界前 CloseAll。
    public sealed class RuntimeHudDriver : MonoBehaviour
    {
        // BOSS 在快照里的 Kind（协议约定：1 普通怪、2 BOSS、0 英雄）
        private const int KindHero = 0;
        private const int KindBoss = 2;

        // 服务端任务阶段（与 Server/NBC.Server.Game/IQuestStateStore.cs 的 EQuestStage 数值对齐）
        private const int StageAccepted = 1;
        private const int StageCompleted = 2;

        // 任务行刷新间隔，秒：任务不是每帧都变，省掉每帧拼字符串
        private const float QuestRefreshSeconds = 0.25f;

        // 追踪区最多显示几条任务
        private const int MaxQuestRows = HudQuestPanel.MaxRows;

        private PresentationFeed m_feed;
        private NetSession m_session;
        private HudTargetPanel m_targetPanel;
        private HudQuestPanel m_questPanel;
        private QuestConfig m_questConfig;
        private QuestConditionConfig m_conditionConfig;
        private MonsterConfig m_monsterConfig;
        private readonly List<QuestRow> m_rows = new List<QuestRow>();
        private float m_questTimer;

        // 功能：接上快照源与会话，并打开 HUD 面板。
        // 入参：feed 表现层快照源；session 联机会话。
        // 返回：无。
        // 步骤：1) 记住来源与配置表 2) 打开两个面板（失败只告警）3) 立刻刷新一次。
        // 上游：RuntimeNetDriver.OnJoinSucceeded；下游：UIManager.ShowPanel。
        public void Bind(PresentationFeed feed, NetSession session)
        {
            m_feed = feed;
            m_session = session;
            m_questConfig = ConfigMgr.Instance.Get<QuestConfig>();
            m_conditionConfig = ConfigMgr.Instance.Get<QuestConditionConfig>();
            m_monsterConfig = ConfigMgr.Instance.Get<MonsterConfig>();

            UIManager ui = UIManager.Instance;

            if (ui != null)
            {
                if (m_targetPanel == null)
                {
                    ui.ShowPanel(HudTargetPanel.PanelName, UILayer.Bot, panel => m_targetPanel = panel as HudTargetPanel);
                }

                if (m_questPanel == null)
                {
                    ui.ShowPanel(HudQuestPanel.PanelName, UILayer.Bot, panel => m_questPanel = panel as HudQuestPanel);
                }
            }

            if (m_targetPanel == null)
            {
                Debug.LogWarning("[HUD] 打不开 " + HudTargetPanel.PanelName + "（预制体或地址缺失）");
            }

            if (m_questPanel == null)
            {
                Debug.LogWarning("[HUD] 打不开 " + HudQuestPanel.PanelName + "（预制体或地址缺失）");
            }

            UpdateTargets();
            UpdateQuest();
        }

        // 功能：收起 HUD（回 World 时调）。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 两条血条隐藏 2) 关掉两块面板。
        // 上游：RuntimeNetDriver.ReturnToWorldRoutine。
        public void CloseAll()
        {
            m_feed = null;
            m_session = null;

            if (m_targetPanel != null)
            {
                m_targetPanel.SetBoss(false, null, 0, 0);
                m_targetPanel.SetTarget(false, null, 0, 0);
            }

            UIManager ui = UIManager.Instance;

            if (ui != null)
            {
                ui.HidePanel(HudTargetPanel.PanelName);
                ui.HidePanel(HudQuestPanel.PanelName);
            }
        }

        // 功能：每帧刷血条，每 0.25 秒刷任务行。
        // 入参：无。
        // 返回：无。
        private void Update()
        {
            if (m_feed == null || m_session == null)
            {
                return;
            }

            UpdateTargets();

            m_questTimer -= Time.unscaledDeltaTime;

            if (m_questTimer <= 0f)
            {
                m_questTimer = QuestRefreshSeconds;
                UpdateQuest();
            }
        }

        // 功能：按当前快照挑 BOSS 与最近的目标，喂给面板。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 找英雄位置 2) 遍历实体找 BOSS 与最近的目标 3) 写面板。
        // 上游：Update / Bind。
        private void UpdateTargets()
        {
            if (m_targetPanel == null)
            {
                return;
            }

            IReadOnlyList<EntitySnapshot> entities = m_feed.View != null ? m_feed.View.Entities : null;
            SnapshotView view = m_feed.View;

            if (entities == null || entities.Count == 0)
            {
                m_targetPanel.SetBoss(false, null, 0, 0);
                m_targetPanel.SetTarget(false, null, 0, 0);
                return;
            }

            EntitySnapshot hero = view != null ? view.FindHero(m_session.PlayerId) : null;
            EntitySnapshot boss = null;
            EntitySnapshot nearest = null;
            long nearestDistanceSq = long.MaxValue;

            for (int i = 0; i < entities.Count; i++)
            {
                EntitySnapshot entity = entities[i];

                if (!entity.Alive || entity.Kind == KindHero)
                {
                    continue;
                }

                if (entity.Kind == KindBoss)
                {
                    if (boss == null)
                    {
                        boss = entity;
                    }

                    continue;
                }

                if (hero == null)
                {
                    continue;
                }

                long dx = entity.PosXMm - hero.PosXMm;
                long dz = entity.PosZMm - hero.PosZMm;
                long distanceSq = dx * dx + dz * dz;

                if (distanceSq < nearestDistanceSq)
                {
                    nearestDistanceSq = distanceSq;
                    nearest = entity;
                }
            }

            if (boss != null)
            {
                m_targetPanel.SetBoss(true, NameOf(boss), boss.Hp, boss.MaxHp);
            }
            else
            {
                m_targetPanel.SetBoss(false, null, 0, 0);
            }

            if (nearest != null)
            {
                m_targetPanel.SetTarget(true, NameOf(nearest), nearest.Hp, nearest.MaxHp);
            }
            else
            {
                m_targetPanel.SetTarget(false, null, 0, 0);
            }
        }

        // 功能：把「已接取 / 可交付」的任务拼成最多三行。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 取任务状态全量 2) 逐条过滤阶段并查表拼标题与进度 3) 交给面板。
        // 上游：Update / Bind。
        private void UpdateQuest()
        {
            if (m_questPanel == null)
            {
                return;
            }

            m_rows.Clear();
            QuestStateSync states = m_session.LastQuestStates;

            if (states != null && m_questConfig != null)
            {
                for (int i = 0; i < states.Entries.Count && m_rows.Count < MaxQuestRows; i++)
                {
                    QuestStateEntry entry = states.Entries[i];

                    if (entry.State != StageAccepted && entry.State != StageCompleted)
                    {
                        continue;
                    }

                    Config_Quest quest;

                    if (!m_questConfig.TryGet(entry.QuestId, out quest))
                    {
                        continue;
                    }

                    m_rows.Add(new QuestRow
                    {
                        QuestId = entry.QuestId,
                        Title = "[" + entry.QuestId + "] " + quest.name,
                        Detail = BuildDetail(quest)
                    });
                }
            }

            m_questPanel.SetRows(m_rows);
        }

        // 功能：把一条任务的完成条件拼成一行进度文字。
        // 入参：quest 任务配置行。
        // 返回：形如「击杀怪物 2/3」；没有条件时返回空串。
        // 步骤：1) 没有条件表就返回空 2) 逐条取进度 3) 用「；」拼起来。
        private string BuildDetail(Config_Quest quest)
        {
            if (m_conditionConfig == null || quest.conditionIds == null || quest.conditionIds.Length == 0)
            {
                return string.Empty;
            }

            System.Text.StringBuilder builder = new System.Text.StringBuilder(64);

            for (int i = 0; i < quest.conditionIds.Length; i++)
            {
                int conditionId = quest.conditionIds[i];
                Config_QuestCondition condition;

                if (!m_conditionConfig.TryGet(conditionId, out condition))
                {
                    continue;
                }

                ConditionProgressEntry progress = m_session.FindProgress(conditionId);

                if (builder.Length > 0)
                {
                    builder.Append('；');
                }

                builder.Append(string.IsNullOrEmpty(condition.note) ? "条件 " + conditionId : condition.note);
                builder.Append(' ');
                builder.Append(progress.Current);
                builder.Append('/');
                builder.Append(condition.requiredCount);
            }

            return builder.ToString();
        }

        // 功能：取实体的显示名（怪物查配置表，查不到就用编号）。
        // 入参：entity 快照条目。
        // 返回：给玩家看的名字。
        private string NameOf(EntitySnapshot entity)
        {
            if (m_monsterConfig != null)
            {
                Config_Monster monster;

                if (m_monsterConfig.TryGet(entity.ConfigId, out monster) && !string.IsNullOrEmpty(monster.name))
                {
                    return monster.name;
                }
            }

            return "怪物 " + entity.ConfigId;
        }
    }
}
```

## 七、风险与幻觉自查

| 风险 | 自查结果 |
| --- | --- |
| `ShowPanel` 返回值 | **不是**返回值，只有 `onShown` 回调 —— 第一版写成 `as HudTargetPanel` 编译报 CS0039，已改回调 |
| `NetSession.Quests` | 不存在；`Quests` 在离线 `BattleSession` 上。联机改用 `LastQuestStates` + `FindProgress` |
