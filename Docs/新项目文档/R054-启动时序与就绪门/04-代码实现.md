## 二、功能概述

玩家按 E 进副本后：驱动每帧问「HUD 能不能挂了」，编排链负责按顺序等资源层、设配置来源、等三张表，最后一步才打开 HUD。中途任何一步没好就原地等，超过 8 秒打一条 `[Boot] 失败：第 N 步（…）`，验收者据此判断是资源慢还是配置慢。

## 三、接口与入参出参

见 03 第三节（逐字一致）。要点：

- `BattleBootSequence.Tick(float deltaSeconds)`：`deltaSeconds` 传负值按 0 计；
- `Completed` 与 `Failed` 互斥；`Cancel` 之后两者都不再触发；
- `BattleConfigBootstrap.Ensure(Action<string> onFailed = null)`：`onFailed` 可空，为空时失败静默（由调用方决定是否提示）。

## 四、逐段讲解

1. `BattleReadiness` 的 `AssetsReady` 走 `AssetManager.IsInitialized`：**判断而不是加载**，所以可以每帧问。
2. `HudTablesReady` 只查三张表（任务、任务条件、怪物）——HUD 真正会读的；`AllTablesReady` 留给需要「全表就绪」的场合。
3. `BattleConfigBootstrap.Ensure` 先 `Source == null` 判断再 `SetSource`，最后只在 `LoadedTableCount == 0` 时 `PreloadAll`，因此 `HpBarDemo` 与编排同时调用也只加载一次。
4. `BattleBootSequence` 把「四个判断/动作」全部从构造函数注入，内部不碰任何单例 —— 这是 7 条 EditMode 用例能跑的前提。
5. 第 3 步（等表）满足时**同帧**调 `Finish()`，避免「表齐了但还要等下一帧才挂 HUD」的空窗；`Finish` 先置完成标记再执行 `bindHud`，防止动作里重入。
6. `RuntimeNetDriver.EnsureHud` 只在第一次（`m_boot == null`）构造编排并 `Begin()`，之后每帧 `Tick`；`BindHud` 保留原有的「拿组件/挂组件 + Bind」。
7. `CloseHud` 先 `Cancel()` 再置 `m_boot = null`：回 World 后这次编排作废，下次进副本重新从第 1 步开始。
8. `RuntimeHudDriver.Bind` 的提前返回改用 `BattleReadiness.AssetsReady`；`ResolveConfigs` 用 `BattleReadiness.ConfigSourceReady` 判断「来源没设」，语义不变但判断收口到一处。

```csharp
// 这个文件里有：
//   BattleReadiness（静态类）：判断战斗表现层所需的资源层与配置表是否就绪，并说明缺什么。

using NBC.Framework.Asset;
using NBC.Game.Config;

namespace NBC.Game.Bootstrap
{
    // 主要功能：把「现在能不能让表现层干活」收敛成几个只读判断，避免每个消费者各写一套。
    // 主要数据成员：HudTables（HUD 真正依赖的三张表名）；其余都是即时计算，不缓存状态。
    // 主要外部方法：AssetsReady / ConfigSourceReady / HudTablesReady / HudReady / DescribeMissing。
    // 谁在用它：RuntimeNetDriver（能否挂 HUD）、RuntimeHudDriver（能否请求面板与查表）、BattleBootSequence（编排各步）。
    // 注意：这里只回答「现在是不是就绪」，不做任何加载与等待 —— 加载由启动方负责，等待由调用方按帧重试。
    public static class BattleReadiness
    {
        // HUD 真正要用的三张表（表名不带 Config 后缀，和 GameTables.All 里的写法一致）
        private static readonly string[] HudTables = { "Quest", "QuestCondition", "Monster" };

        // 功能：资源层（AssetManager）是否已经装好加载器并初始化。
        // 入参：无。
        // 返回：true = 可以加载资源；false = 还没装好。
        // 步骤：1) 取单例 2) 判空 3) 读 IsInitialized。
        // 上游：所有需要加载预制体/配置的调用方；下游：AssetManager.IsInitialized。
        public static bool AssetsReady
        {
            get
            {
                AssetManager assets = AssetManager.Instance;
                return assets != null && assets.IsInitialized;
            }
        }

        // 功能：配置表来源是否已经设置（没设来源时任何 Preload 都会立刻失败）。
        // 入参：无。
        // 返回：true = 有来源；false = 还没 SetSource。
        // 步骤：1) 取单例 2) 判空 3) 读 Source。
        // 上游：BattleConfigBootstrap、RuntimeHudDriver；下游：ConfigMgr.Source。
        public static bool ConfigSourceReady
        {
            get
            {
                ConfigMgr mgr = ConfigMgr.Instance;
                return mgr != null && mgr.Source != null;
            }
        }

        // 功能：GameTables.All 里的表是否都已加载进内存。
        // 入参：无。
        // 返回：true = 全部加载完；false = 还有表在加载或加载失败。
        // 步骤：1) 取单例 2) 比 LoadedTableCount 与 All.Length。
        // 上游：BattleBootSequence 判「表好了没有」；下游：ConfigMgr.LoadedTableCount。
        public static bool AllTablesReady
        {
            get
            {
                ConfigMgr mgr = ConfigMgr.Instance;
                return mgr != null && mgr.LoadedTableCount >= GameTables.All.Length;
            }
        }

        // 功能：HUD 要用的三张表（Quest / QuestCondition / Monster）是否都在内存里。
        // 入参：无。
        // 返回：true = 三张全有；false = 缺任意一张。
        // 步骤：1) 取单例 2) 逐张 IsLoaded 3) 有一张缺就返回 false。
        // 上游：BattleBootSequence、RuntimeHudDriver.ResolveConfigs；下游：ConfigMgr.IsLoaded。
        public static bool HudTablesReady
        {
            get
            {
                ConfigMgr mgr = ConfigMgr.Instance;

                if (mgr == null)
                {
                    return false;
                }

                for (int i = 0; i < HudTables.Length; i++)
                {
                    if (!mgr.IsLoaded(HudTables[i]))
                    {
                        return false;
                    }
                }

                return true;
            }
        }

        // 功能：HUD 能不能挂了（资源层就绪 + 三张表都在）。
        // 入参：无。
        // 返回：true = 可以 Bind；false = 还得等。
        // 步骤：1) AssetsReady 2) HudTablesReady 3) 两者都真才返回 true。
        // 上游：RuntimeNetDriver；下游：AssetsReady、HudTablesReady。
        public static bool HudReady
        {
            get { return AssetsReady && HudTablesReady; }
        }

        // 功能：说清现在缺什么（给日志用，不要拿它做判断）。
        // 入参：无。
        // 返回：缺项的人话描述；都齐了返回空串。
        // 步骤：1) 依次判资源层、配置来源、HUD 三张表 2) 第一个缺的写成一句话返回。
        // 上游：BattleBootSequence 失败时、驱动诊断日志；下游：无。
        public static string DescribeMissing()
        {
            if (!AssetsReady)
            {
                return "资源层还没装好";
            }

            if (!ConfigSourceReady)
            {
                return "配置表来源还没设置";
            }

            if (!HudTablesReady)
            {
                return "任务/怪物配置表还没加载完";
            }

            return string.Empty;
        }
    }
}
```

```csharp
// 这个文件里有：
//   BattleConfigBootstrap（静态类）：进副本时把配置表来源接上并预加载，重复调用无害。

using System;
using NBC.Game.Config;

namespace NBC.Game.Bootstrap
{
    // 主要功能：一处负责「配置表从哪读、什么时候读」，避免每个消费者各自 SetSource。
    // 主要数据成员：无（全部状态由 ConfigMgr 单例持有）。
    // 主要外部方法：Ensure（设来源 + 空表时预加载全部）。
    // 谁在用它：HpBarDemoBehaviour（进副本装配点）、BattleBootSequence（第二步）。
    // 注意：本方法幂等 —— 已有来源不覆盖、已有表不再加载；失败只通过 onFailed 上报，不抛异常。
    public static class BattleConfigBootstrap
    {
        // 功能：确保配置表来源已设置，并在「一张表都没有」时预加载全部表。
        // 入参：onFailed 预加载失败时的原因回调，可空。
        // 返回：无（加载是异步的，结果稍后通过 ConfigMgr 的表缓存体现）。
        // 步骤：1) 取单例 2) 没来源就设成走资源层的 AssetConfigSource 3) 已有表就返回 4) 否则 GameTables.PreloadAll。
        // 上游：HpBarDemoBehaviour.Start、BattleBootSequence；下游：ConfigMgr.SetSource、GameTables.PreloadAll。
        public static void Ensure(Action<string> onFailed = null)
        {
            ConfigMgr mgr = ConfigMgr.Instance;

            if (mgr == null)
            {
                if (onFailed != null)
                {
                    onFailed("ConfigMgr 单例不可用");
                }

                return;
            }

            if (mgr.Source == null)
            {
                mgr.SetSource(new AssetConfigSource());
            }

            if (mgr.LoadedTableCount > 0)
            {
                return;
            }

            GameTables.PreloadAll(null, onFailed);
        }
    }
}
```

### 5.3 `Client/Assets/_Project/Game/Bootstrap/BattleBootSequence.cs`

```csharp
// 这个文件里有：
//   EBootStep（枚举）、BattleBootSequence（类）：进副本装表现层的四步编排，带超时与失败原因。

using System;

namespace NBC.Game.Bootstrap
{
    // 战斗表现层启动的四个阶段（顺序即执行顺序）。
    public enum EBootStep
    {
        // 等资源层装好（HpBarDemo 里 await InstallAsync 那一段）
        WaitAssets = 0,

        // 设配置表来源（缺来源时任何 Preload 都会失败）
        PrepareConfigs = 1,

        // 等 HUD 要用的配置表加载完
        WaitTables = 2,

        // 打开 HUD 面板并接上数据源
        BindHud = 3,

        // 四步都完成
        Done = 4
    }

    // 主要功能：把「进副本要做哪几件有先后的事」写成一条可观察的步骤链，每步带超时与失败原因。
    // 主要数据成员：m_step（当前步）、m_stepElapsed（当前步已耗时，秒）、m_stepTimeoutSeconds（单步超时，秒）。
    // 主要外部方法：Begin / Tick / Cancel；事件 Completed / Failed。
    // 谁在用它：RuntimeNetDriver（每次进副本 Begin 一次，Update 里 Tick）。
    // 注意：本类不直接碰 Unity 与单例，全部依赖由构造函数注入 —— 这样 EditMode 测试能拿假条件跑。
    public sealed class BattleBootSequence
    {
        // 单步默认超时：8 秒还没好就判失败（资源加载慢于这个值基本就是出问题了）
        public const float DefaultStepTimeoutSeconds = 8f;

        // 读「资源层好了吗」
        private readonly Func<bool> m_assetsReady;

        // 读「配置来源设了吗」
        private readonly Func<bool> m_configSourceReady;

        // 读「HUD 要用的表齐了吗」
        private readonly Func<bool> m_tablesReady;

        // 第二步要执行的动作（设来源 + 发起预加载）
        private readonly Action m_prepareConfigs;

        // 第四步要执行的动作（打开面板、接上数据源）
        private readonly Action m_bindHud;

        // 日志出口（可空；空则不输出）
        private readonly Action<string> m_log;

        // 单步超时秒数
        private readonly float m_stepTimeoutSeconds;

        // 当前步骤
        private EBootStep m_step = EBootStep.WaitAssets;

        // 当前步骤已经耗时多少秒
        private float m_stepElapsed;

        // 是否在跑（Begin 之后、完成/取消之前为 true）
        private bool m_running;

        // 是否已经完成过一次
        private bool m_finished;

        // 功能：注入四个判断/动作与日志出口。
        // 入参：assetsReady/configSourceReady/tablesReady 三个只读判断；prepareConfigs/bindHud 两个动作；log 日志出口可空；stepTimeoutSeconds 单步超时秒数。
        // 返回：无（构造即就绪，调用 Begin 才开始跑）。
        // 步骤：1) 逐项存下依赖 2) 超时取正数（传 0 或负数时回落到默认值）。
        // 上游：RuntimeNetDriver、EditMode 测试。
        public BattleBootSequence(
            Func<bool> assetsReady,
            Func<bool> configSourceReady,
            Func<bool> tablesReady,
            Action prepareConfigs,
            Action bindHud,
            Action<string> log = null,
            float stepTimeoutSeconds = DefaultStepTimeoutSeconds)
        {
            m_assetsReady = assetsReady;
            m_configSourceReady = configSourceReady;
            m_tablesReady = tablesReady;
            m_prepareConfigs = prepareConfigs;
            m_bindHud = bindHud;
            m_log = log;
            m_stepTimeoutSeconds = stepTimeoutSeconds > 0f ? stepTimeoutSeconds : DefaultStepTimeoutSeconds;
        }

        // 四步都走完时触发（只触发一次）
        public event Action Completed;

        // 某一步超时时触发，参数是人话原因
        public event Action<string> Failed;

        // 当前步骤
        public EBootStep Step
        {
            get { return m_step; }
        }

        // 是否正在跑
        public bool IsRunning
        {
            get { return m_running; }
        }

        // 是否已经跑完过
        public bool IsFinished
        {
            get { return m_finished; }
        }

        // 功能：从头开始跑这条链（重复调用会重置到第一步）。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 复位步骤与计时 2) 置运行中 3) 打一条开始日志。
        // 上游：RuntimeNetDriver.EnsureHud；下游：无。
        public void Begin()
        {
            m_step = EBootStep.WaitAssets;
            m_stepElapsed = 0f;
            m_running = true;
            m_finished = false;
            Log("[Boot] 开始装表现层（第 1 步：等资源层）");
        }

        // 功能：推进一次（由调用方每帧驱动）。
        // 入参：deltaSeconds 本帧耗时（秒）。
        // 返回：无。
        // 步骤：1) 没在跑就返回 2) 累加耗时并判超时 3) 按当前步的条件决定前进还是继续等。
        // 上游：RuntimeNetDriver.Update；下游：m_prepareConfigs、m_bindHud、Completed、Failed。
        public void Tick(float deltaSeconds)
        {
            if (!m_running)
            {
                return;
            }

            m_stepElapsed += deltaSeconds > 0f ? deltaSeconds : 0f;

            if (m_stepElapsed > m_stepTimeoutSeconds)
            {
                Fail("第 " + ((int)m_step + 1) + " 步（" + NameOf(m_step) + "）超过 " + m_stepTimeoutSeconds + " 秒还没完成");
                return;
            }

            switch (m_step)
            {
                case EBootStep.WaitAssets:
                    if (m_assetsReady != null && m_assetsReady())
                    {
                        Enter(EBootStep.PrepareConfigs);
                    }

                    break;

                case EBootStep.PrepareConfigs:
                    if (m_configSourceReady != null && m_configSourceReady())
                    {
                        Enter(EBootStep.WaitTables);
                    }

                    break;

                case EBootStep.WaitTables:
                    if (m_tablesReady != null && m_tablesReady())
                    {
                        // 表刚齐：同一次 Tick 里就走到最后一步，避免「等下一帧才挂 HUD」这类一帧空窗
                        Enter(EBootStep.BindHud);
                        Finish();
                    }

                    break;

                case EBootStep.BindHud:
                    Finish();
                    break;
            }
        }

        // 功能：收尾 —— 挂 HUD、标记完成、触发 Completed（只会执行一次）。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 先置完成状态（避免动作里重入）2) 执行 m_bindHud 3) 打日志并触发 Completed。
        // 上游：Tick（表齐那一刻、BindHud 步）；下游：m_bindHud、Completed。
        private void Finish()
        {
            if (m_finished)
            {
                return;
            }

            m_step = EBootStep.Done;
            m_running = false;
            m_finished = true;
            m_stepElapsed = 0f;

            if (m_bindHud != null)
            {
                m_bindHud();
            }

            Log("[Boot] 表现层就绪");
            Completed?.Invoke();
        }

        // 功能：取消（回世界、切场景时调），之后不再触发任何回调。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 清运行标记 2) 打一条日志。
        // 上游：RuntimeNetDriver.CloseHud；下游：无。
        public void Cancel()
        {
            if (!m_running)
            {
                return;
            }

            m_running = false;
            Log("[Boot] 已取消（第 " + ((int)m_step + 1) + " 步：" + NameOf(m_step) + "）");
        }

        // 功能：步骤名的中文说法（日志与失败原因用）。
        // 入参：step 步骤。
        // 返回：中文名字。
        // 步骤：1) 按枚举返回固定文案。
        // 上游：Tick、Cancel；下游：无。
        public static string NameOf(EBootStep step)
        {
            switch (step)
            {
                case EBootStep.WaitAssets:
                    return "等资源层";
                case EBootStep.PrepareConfigs:
                    return "设配置来源";
                case EBootStep.WaitTables:
                    return "等配置表";
                case EBootStep.BindHud:
                    return "挂 HUD";
                default:
                    return "完成";
            }
        }

        // 功能：切到下一步，并在进入「设配置来源」时执行一次准备动作。
        // 入参：next 目标步骤。
        // 返回：无。
        // 步骤：1) 记新步骤并清零计时 2) 打日志 3) 轮到准备配置时执行一次 m_prepareConfigs。
        // 上游：Tick；下游：m_prepareConfigs。
        private void Enter(EBootStep next)
        {
            m_step = next;
            m_stepElapsed = 0f;
            Log("[Boot] 进入第 " + ((int)next + 1) + " 步：" + NameOf(next));

            if (next == EBootStep.PrepareConfigs && m_prepareConfigs != null)
            {
                m_prepareConfigs();
            }
        }

        // 功能：判超时失败并停止。
        // 入参：reason 人话原因。
        // 返回：无。
        // 步骤：1) 停跑 2) 打错误日志 3) 触发 Failed。
        // 上游：Tick；下游：Failed 订阅方。
        private void Fail(string reason)
        {
            m_running = false;
            Log("[Boot] 失败：" + reason);
            Failed?.Invoke(reason);
        }

        // 功能：写一条日志（没注入日志出口时什么都不做）。
        // 入参：message 文本。
        // 返回：无。
        // 步骤：1) 判空 2) 转调注入的日志动作。
        // 上游：Begin/Enter/Fail/Cancel/Tick；下游：注入的日志出口。
        private void Log(string message)
        {
            if (m_log != null)
            {
                m_log(message);
            }
        }
    }
}
```

```csharp
// 这个文件里有：
//   BattleBootSequenceTests（测试类）：编排链的顺序、超时、取消与回调次数。

using System.Collections.Generic;
using NBC.Game.Bootstrap;
using NUnit.Framework;

namespace NBC.Tests.EditMode.Game
{
    // 主要功能：用假条件驱动 BattleBootSequence，验证它「按顺序前进、只触发一次、超时会失败」。
    // 主要数据成员：三个可切换的就绪标志、两个动作计数器、日志收集。
    // 主要外部方法：SetUp（复位）、七条 TestXxx 用例。
    // 谁在用它：Test Runner（EditMode）。
    public sealed class BattleBootSequenceTests
    {
        // 假资源层就绪标志
        private bool m_assetsReady;

        // 假配置来源标志
        private bool m_sourceReady;

        // 假配置表就绪标志
        private bool m_tablesReady;

        // PrepareConfigs 被调用了几次
        private int m_prepareCalls;

        // BindHud 被调用了几次
        private int m_bindCalls;

        // Completed 被触发了几次
        private int m_completedCalls;

        // Failed 原因（没失败时为 null）
        private string m_failure;

        // 日志（按顺序）

        // 功能：每条用例前把全部状态复位。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 三个标志置 false 2) 计数清零 3) 清空日志。
        // 上游：NUnit；下游：无。

        public void SetUp()
        {
            m_assetsReady = false;
            m_sourceReady = false;
            m_tablesReady = false;
            m_prepareCalls = 0;
            m_bindCalls = 0;
            m_completedCalls = 0;
            m_failure = null;
            m_logs.Clear();

        // 功能：Begin 之后应停在第一步。
        // 入参：无。
        // 返回：无。

            Assert.IsTrue(sequence.IsRunning, "Begin 之后应在运行中");
            Assert.IsFalse(sequence.IsFinished, "刚 Begin 不该算完成");

        // 功能：依次满足条件后四步应依次走完，并在最后触发一次 Completed。
        // 入参：无。

        // 步骤：1) Begin 2) 逐帧打开资源/来源/表 3) 断言 Step=Done、BindHud 一次、Completed 一次。
        // 上游：测试驱动；下游：Tick。
        [Test]
        public void StepsAdvanceInOrder_WhenConditionsBecomeTrue()
        {
            BattleBootSequence sequence = NewSequence();

            Assert.AreEqual(EBootStep.PrepareConfigs, sequence.Step, "资源好了应该去设来源");

        // 入参：无。

        // 上游：测试驱动；下游：Cancel、Tick。
        [Test]
        public void Cancel_StopsCallbacks()

            m_tablesReady = true;

        // 功能：取消后再 Begin 应能从第一步重新跑（回世界再进副本的场景）。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 跑到第二步 2) Cancel 3) Begin 4) 断言 Step 回到 WaitAssets 且跑完仍只有一次 BindHud。
        // 上游：测试驱动；下游：Begin、Tick。
        [Test]
        public void Begin_AfterCancel_RestartsFromFirstStep()
        {
            BattleBootSequence sequence = NewSequence();
            sequence.Begin();
            m_assetsReady = true;
            sequence.Tick(0.1f);
            sequence.Cancel();
            sequence.Begin();

            Assert.AreEqual(EBootStep.WaitAssets, sequence.Step, "重新 Begin 应回到第一步");

            m_sourceReady = true;
            m_tablesReady = true;
            sequence.Tick(0.1f);
            sequence.Tick(0.1f);
            sequence.Tick(0.1f);

            Assert.AreEqual(1, m_bindCalls, "重跑一遍只该挂一次 HUD");
            Assert.AreEqual(1, m_completedCalls, "重跑一遍只该完成一次");
        }

        // 功能：日志应按步输出，便于现场定位卡在哪一步。
        // 入参：无。
        // 返回：无。
        // 步骤：1) 走完 2) 断言日志里出现「第 1 步」与「表现层就绪」。
        // 上游：测试驱动；下游：注入的日志出口。
        [Test]
        public void Log_ContainsStepMarkers()
        {
            BattleBootSequence sequence = NewSequence();
            RunToDone(sequence);

            Assert.IsTrue(m_logs.Exists(line => line.Contains("第 1 步")), "日志应出现第 1 步");
            Assert.IsTrue(m_logs.Exists(line => line.Contains("表现层就绪")), "日志应出现就绪");
        }

        // 功能：建一条测试用的链（条件读本类字段，动作只计数）。
        // 入参：timeoutSeconds 单步超时秒数。
        // 返回：建好的编排链。
        // 步骤：1) 用本类字段构造四个依赖 2) 接上 Completed/Failed 计数 3) 返回。
        // 上游：各用例；下游：BattleBootSequence 构造。
        private BattleBootSequence NewSequence(float timeoutSeconds = 8f)
        {
            BattleBootSequence sequence = new BattleBootSequence(
                () => m_assetsReady,
                () => m_sourceReady,
                () => m_tablesReady,
                () => m_prepareCalls++,
                () => m_bindCalls++,
                line => m_logs.Add(line),
                timeoutSeconds);

            sequence.Completed += () => m_completedCalls++;
            sequence.Failed += reason => m_failure = reason;
            return sequence;
        }

        // 功能：把条件全部置真并推进到完成（供「已完成之后」类用例复用）。
        // 入参：sequence 目标链。
        // 返回：无。
        // 步骤：1) 三个标志置真 2) Begin 3) Tick 四次走完。
        // 上游：测试驱动；下游：Begin、Tick。
        private void RunToDone(BattleBootSequence sequence)
        {
            m_assetsReady = true;
            m_sourceReady = true;
            m_tablesReady = true;
            sequence.Begin();
            sequence.Tick(0.1f);
            sequence.Tick(0.1f);
            sequence.Tick(0.1f);
            sequence.Tick(0.1f);
        }
    }
}
```
